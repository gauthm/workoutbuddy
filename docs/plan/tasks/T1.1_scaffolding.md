### `server.js` (structure complète)

```javascript
import express from 'express'
import path from 'path'
import { fileURLToPath } from 'url'
import { getDb } from './src/db/database.js'
import { migrate } from './src/db/migrate.js'
import { seedExercises } from './src/seed/seed.js'

const __dirname = path.dirname(fileURLToPath(import.meta.url))
const app = express()

app.use(express.json())
app.use(express.static(path.join(__dirname, 'public')))

// ── Montage des routes (à compléter tâche par tâche) ──────────
// import exercisesRouter from './src/routes/exercises.js'
// app.use('/api/v1/exercises', exercisesRouter)

// ── Fallback SPA ──────────────────────────────────────────────
app.get('*', (req, res) => {
  if (!req.path.startsWith('/api/')) {
    res.sendFile(path.join(__dirname, 'public', 'index.html'))
  }
})

// ── Démarrage ─────────────────────────────────────────────────
async function bootstrap() {
  const db = getDb()
  migrate(db)
  seedExercises(db)

  // Lire host et port depuis les settings (table créée par migrate)
  const portRow = db.prepare("SELECT value FROM settings WHERE key='network_port'").get()
  const hostRow = db.prepare("SELECT value FROM settings WHERE key='network_host'").get()
  const port = portRow ? JSON.parse(portRow.value) : 3000
  const host = hostRow ? JSON.parse(hostRow.value) : '127.0.0.1'

  app.listen(port, host, () => {
    console.log(`WorkoutBuddy démarré → http://${host}:${port}`)
  })
}

bootstrap().catch(err => {
  console.error('Erreur au démarrage :', err.message)
  process.exit(1)
})

export { app } // pour supertest
```

### `scripts/download-vendors.js`

```javascript
import fs from 'fs'
import path from 'path'
import { fileURLToPath } from 'url'

const __dirname = path.dirname(fileURLToPath(import.meta.url))
const VENDOR_DIR = path.join(__dirname, '..', 'public', 'vendor')

const FILES = [
  {
    url: 'https://cdn.jsdelivr.net/npm/vue@3.5.13/dist/vue.esm-browser.prod.js',
    dest: 'vue.esm-browser.prod.js',
  },
  {
    url: 'https://cdn.jsdelivr.net/npm/chart.js@4.4.4/dist/chart.umd.min.js',
    dest: 'chart.umd.min.js',
  },
]

async function download() {
  fs.mkdirSync(VENDOR_DIR, { recursive: true })
  for (const { url, dest } of FILES) {
    const destPath = path.join(VENDOR_DIR, dest)
    if (fs.existsSync(destPath)) {
      console.log(`✓ Déjà présent : ${dest}`)
      continue
    }
    console.log(`Téléchargement : ${url}`)
    const res = await fetch(url)
    if (!res.ok) throw new Error(`HTTP ${res.status} pour ${url}`)
    const text = await res.text()
    fs.writeFileSync(destPath, text, 'utf8')
    console.log(`✓ Enregistré : ${dest}`)
  }
  console.log('Vendors prêts.')
}

download().catch(err => {
  console.error('Erreur download-vendors :', err.message)
  process.exit(1)
})
```

## Étapes d'implémentation

1. Créer tous les fichiers listés ci-dessus avec leur contenu exact.
2. Créer les dossiers vides avec `.gitkeep` :
   `data/`, `public/vendor/`, `public/exercises/images/`
3. Créer les dossiers manquants pour les tâches futures (vides) :
   `src/db/migrations/`, `src/routes/`, `src/services/`, `src/seed/`,
   `tests/unit/`, `tests/integration/`, `public/js/components/`,
   `public/locales/`, `public/icons/`, `public/sounds/`
4. Créer des stubs vides pour que `server.js` n'échoue pas à l'import :
   - `src/db/database.js` → exporte `getDb()` qui retourne `null` pour l'instant
   - `src/db/migrate.js` → exporte `migrate()` vide
   - `src/seed/seed.js` → exporte `seedExercises()` vide
5. Lancer `npm install`
6. Lancer `node scripts/download-vendors.js`
7. Lancer `npm start` → vérifier le message dans le terminal

**Points d'attention :**
- `"type": "module"` dans `package.json` → tous les fichiers `.js`
  utilisent `import`/`export`, pas `require()`.
- `getDb()` retourne `null` dans ce stub — `server.js` appellera
  `migrate(null)` et `seedExercises(null)` sans planter car ces fonctions
  sont vides. Ce sera corrigé en T1.2.
- Ne pas oublier `export { app }` à la fin de `server.js` pour supertest.
