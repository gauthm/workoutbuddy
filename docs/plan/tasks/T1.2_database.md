```sql
CREATE TABLE IF NOT EXISTS schema_version (
  version    INTEGER PRIMARY KEY,

-- ── Exercices ─────────────────────────────────────────────────
CREATE TABLE IF NOT EXISTS exercise (
```

## Étapes d'implémentation

1. Créer `src/db/database.js` avec le singleton.
2. Créer `src/db/migrations/001_schema.sql` (coller le SQL ci-dessus).
3. Créer `src/db/migrations/002_default_settings.sql` (coller les INSERT).
4. Créer `src/db/migrate.js` avec la logique décrite dans les signatures.
5. Mettre à jour `server.js` : remplacer les imports stub par les vrais imports.

**Points d'attention :**
- `db.exec(sql)` exécute plusieurs statements d'un coup — c'est la bonne
  méthode pour les migrations (contrairement à `db.prepare().run()`).
- Le `UNIQUE INDEX` sur `equipment_advice` utilise `WHERE status = 'active'`
  (index partiel SQLite) — syntaxe correcte, ne pas modifier.
- Les colonnes JSON (`primary_muscles`, etc.) sont de type `TEXT` en SQL.
  Les valeurs par défaut `'[]'` sont des strings, pas des tableaux.

```javascript
import { describe, it, before } from 'node:test'
import assert from 'node:assert/strict'
import Database from 'better-sqlite3'
import { migrate } from '../../src/db/migrate.js'

function createMemoryDb() {
  const db = new Database(':memory:')
  db.pragma('foreign_keys = ON')
  return db
}

describe('T1.2 — Migrations', () => {
  it('applique les 2 migrations sur une DB vierge', () => {
    const db = createMemoryDb()
    migrate(db)
    const versions = db.prepare('SELECT version FROM schema_version ORDER BY version').all()
    assert.deepEqual(versions.map(r => r.version), [1, 2])
  })

  it('est idempotent — re-exécuter migrate() ne génère pas d\'erreur', () => {
    const db = createMemoryDb()
    migrate(db)
    assert.doesNotThrow(() => migrate(db))
  })

  it('crée la table exercise', () => {
    const db = createMemoryDb()
    migrate(db)
    const row = db.prepare(
      "SELECT name FROM sqlite_master WHERE type='table' AND name='exercise'"
    ).get()
    assert.ok(row)
  })

  it('insère les settings par défaut', () => {
    const db = createMemoryDb()
    migrate(db)
    const row = db.prepare("SELECT value FROM settings WHERE key='unit'").get()
    assert.equal(JSON.parse(row.value), 'kg')
  })

  it('active les foreign keys', () => {
    const db = createMemoryDb()
    db.pragma('foreign_keys = ON')
    migrate(db)
    // Tenter d'insérer une session_exercise avec session_id invalide doit échouer
    migrate(db)
    assert.throws(() => {
      db.prepare(
        "INSERT INTO session_exercise (session_id, exercise_id) VALUES (9999, 'x')"
      ).run()
    })
  })
})
```
