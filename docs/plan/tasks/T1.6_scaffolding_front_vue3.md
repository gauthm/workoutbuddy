```javascript
const ROUTE_DEFS = [

  { re: /^#\/equipment$/,                  name: 'EquipmentView',         params: [] },
  { re: /^#\/programs\/new$/,              name: 'ProgramCreateView',     params: [] },
  { re: /^#\/programs\/([^/]+)\/edit$/,    name: 'ProgramEditView',       params: ['id'] },
  { re: /^#\/programs\/([^/]+)$/,          name: 'ProgramDetailView',     params: ['id'] },
  { re: /^#\/programs$/,                   name: 'ProgramListView',       params: [] },
  { re: /^#\/session\/history$/,           name: 'SessionHistoryView',    params: [] },
  { re: /^#\/session\/([^/]+)$/,           name: 'SessionDetailView',     params: ['id'] },
  { re: /^#\/session$/,                    name: 'SessionActiveView',     params: [] },
  { re: /^#\/calendar$/,                   name: 'CalendarView',          params: [] },
  { re: /^#\/stats\/exercise\/([^/]+)$/,   name: 'ExerciseProgressView',  params: ['id'] },
  { re: /^#\/stats\/bodyweight$/,          name: 'BodyWeightView',        params: [] },
  { re: /^#\/stats$/,                      name: 'StatsDashboardView',    params: [] },
  { re: /^#\/coach$/,                      name: 'CoachView',             params: [] },
  { re: /^#\/settings$/,                   name: 'SettingsView',          params: [] },
]

function matchRoute(hash) {
  const h = (!hash || hash === '' || hash === '#') ? '#/' : hash
  for (const def of ROUTE_DEFS) {
    const m = h.match(def.re)
    if (m) {
      const params = {}
      def.params.forEach((name, i) => { params[name] = decodeURIComponent(m[i + 1] || '') })
      return { name: def.name, params }
    }
  }
  return { name: 'DashboardView', params: {} }  // fallback
}

export const router = reactive({
  currentRoute: { name: 'DashboardView', params: {} },

  push(hash) {
    window.location.hash = hash
  },

  back() {
    window.history.back()
  },

  replace(hash) {
    // Remplace sans ajouter d'entrée dans l'historique
    const url = window.location.href.split('#')[0] + hash
    window.history.replaceState(null, '', url)
    this._update()
  },

  _update() {
    const route = matchRoute(window.location.hash)
    this.currentRoute.name   = route.name
    this.currentRoute.params = route.params
  },

  init() {
    window.addEventListener('hashchange', () => this._update())
    this._update()
  }
})
```

```javascript
import { reactive } from '/vendor/vue.esm-browser.prod.js'

let currentLocale = 'fr'
const locales = {}

async function loadLocale(lang) {
  if (locales[lang]) return
  try {
    const res = await fetch(`/locales/${lang}.json`)
    if (!res.ok) throw new Error(`HTTP ${res.status}`)
    locales[lang] = await res.json()
  } catch (e) {
    console.warn(`[i18n] Impossible de charger la locale "${lang}"`)
    locales[lang] = {}
  }
}

export const i18n = reactive({
  locale: 'fr',
  async setLocale(lang) {
    await loadLocale(lang)
    this.locale = lang
    currentLocale = lang
    document.documentElement.lang = lang
  }
})

// t('key') ou t('key', { name: 'John' }) → interpolation {{name}}
export function t(key, params = {}) {
  const dict = locales[currentLocale] || locales['fr'] || {}
  let str = dict[key]
  if (str === undefined) {
    // Fallback sur fr si locale courante ne contient pas la clé
    str = locales['fr']?.[key] ?? key
  }
  // Interpolation : remplacer {{param}} par la valeur
  return str.replace(/\{\{(\w+)\}\}/g, (_, k) => params[k] ?? `{{${k}}}`)
}

// Initialisation : charger fr dès le démarrage
loadLocale('fr')
```

```javascript
const BASE = '/api/v1'

async function request(method, path, body = undefined) {
  const opts = {
    method,
    headers: { 'Content-Type': 'application/json' }
  }
  if (body !== undefined) opts.body = JSON.stringify(body)
  const res = await fetch(`${BASE}${path}`, opts)
  if (res.status === 204) return null
  const data = await res.json()
  if (!res.ok) {

    err.status = res.status
    err.detail = data.detail

}
```

```javascript
// Formules de RM estimé — miroir client de src/services/rmCalculator.js
// Utiliser uniquement pour l'affichage (calcul autoritaire sur le serveur)

export const FORMULAS = {
  epley:    (w, r) => r === 1 ? w : w * (1 + r / 30),
  brzycki:  (w, r) => r >= 37 ? w : w * 36 / (37 - r),
  lander:   (w, r) => 100 * w / (101.3 - 2.67123 * r),
  oconner:  (w, r) => w * (1 + 0.025 * r),
  lombardi: (w, r) => w * Math.pow(r, 0.10),
}

export function estimate1RM(weightKg, reps, formula = 'epley') {
  if (!weightKg || weightKg <= 0 || !reps || reps <= 0) return 0
  if (reps === 1) return weightKg
  const fn = FORMULAS[formula] || FORMULAS.epley
  return Math.round(fn(weightKg, reps) * 100) / 100
}

export function rmTable(estimated1rm, formula = 'epley') {
  if (!estimated1rm || estimated1rm <= 0) return []
  const result = []
  for (let n = 1; n <= 15; n++) {
    const w = n === 1
      ? estimated1rm
      : Math.round((estimated1rm / (1 + n / 30)) * 10) / 10
    result.push({ reps: n, weight_kg: w })
  }
  return result
}
```

```javascript
export function kgToLb(kg) {
  return Math.round(kg * 2.20462 * 10) / 10
}

export function lbToKg(lb) {
  return Math.round(lb / 2.20462 * 10) / 10
}

// Formate un poids selon l'unité active
// unit: 'kg' | 'lb'
export function formatWeight(kg, unit = 'kg', decimals = 1) {
  if (kg === null || kg === undefined) return '—'
  const val = unit === 'lb' ? kgToLb(kg) : kg
  return `${parseFloat(val.toFixed(decimals))} ${unit}`
}
```

## `public/js/utils/date.js`

```javascript
// Retourne la date du jour au format YYYY-MM-DD
export function isoDate(d = new Date()) {
  return d.toISOString().slice(0, 10)
}

// Retourne la semaine ISO au format YYYY-Www (ex: "2025-W27")
export function isoWeek(dateStr) {
  const d = new Date(dateStr)
  d.setHours(0, 0, 0, 0)
  d.setDate(d.getDate() + 3 - ((d.getDay() + 6) % 7))
  const week1 = new Date(d.getFullYear(), 0, 4)
  const weekNum = 1 + Math.round(
    ((d - week1) / 86400000 - 3 + (week1.getDay() + 6) % 7) / 7
  )
  return `${d.getFullYear()}-W${String(weekNum).padStart(2, '0')}`
}

// Semaine ISO précédente
export function previousIsoWeek(weekStr) {
  // weekStr: "2025-W27"
  const [year, w] = weekStr.split('-W').map(Number)
  if (w === 1) {
    // Trouver la dernière semaine de l'année précédente
    const dec28 = new Date(year - 1, 11, 28)
    return isoWeek(isoDate(dec28))
  }
  return `${year}-W${String(w - 1).padStart(2, '0')}`
}

// Nombre de jours entre deux dates YYYY-MM-DD
export function daysBetween(dateA, dateB) {
  const a = new Date(dateA)
  const b = new Date(dateB)
  return Math.round(Math.abs(b - a) / 86400000)
}

// Formate une date YYYY-MM-DD pour l'affichage
export function formatDate(dateStr, locale = 'fr-FR') {
  if (!dateStr) return '—'
  return new Date(dateStr).toLocaleDateString(locale, {
    day: 'numeric', month: 'short', year: 'numeric'
  })
}

// Formate une durée en secondes → "1h 23min" ou "45min"
export function formatDuration(seconds) {
  if (!seconds || seconds < 0) return '—'
  const h = Math.floor(seconds / 3600)
  const m = Math.floor((seconds % 3600) / 60)
  if (h > 0) return `${h}h ${m}min`
  return `${m}min`
}
```

## `public/js/timer-worker.js`

```javascript
// Web Worker — tourne en arrière-plan même si l'onglet est en arrière-plan
// Implémentation complète en T2.5 — ce fichier est un stub fonctionnel minimal

let restInterval = null
let sessionInterval = null
let restRemaining = 0
let sessionElapsed = 0

self.onmessage = function(e) {
  const { type, payload } = e.data

  if (type === 'START_REST') {
    if (restInterval) clearInterval(restInterval)
    restRemaining = payload.duration_seconds
    restInterval = setInterval(() => {
      restRemaining--
      self.postMessage({ type: 'REST_TICK', remaining: restRemaining })
      if (restRemaining <= 0) {
        clearInterval(restInterval)
        restInterval = null
        self.postMessage({ type: 'REST_COMPLETE' })
      }
    }, 1000)
  }

  if (type === 'STOP_REST') {
    if (restInterval) { clearInterval(restInterval); restInterval = null }
    self.postMessage({ type: 'REST_STOPPED' })
  }

  if (type === 'START_SESSION') {
    if (sessionInterval) clearInterval(sessionInterval)
    sessionElapsed = payload.elapsed_seconds || 0
    sessionInterval = setInterval(() => {
      sessionElapsed++
      self.postMessage({ type: 'SESSION_TICK', elapsed: sessionElapsed })
    }, 1000)
  }

  if (type === 'STOP_SESSION') {
    if (sessionInterval) { clearInterval(sessionInterval); sessionInterval = null }
  }
}
```

## `public/js/app.js`

```javascript
import { createApp } from '/vendor/vue.esm-browser.prod.js'
import { router } from './router.js'
import { store } from './store.js'
import { i18n, t } from './i18n.js'
import { settings as settingsApi } from './api.js'
import AppShell from './components/layout/AppShell.js'

async function init() {
  // Charger les settings depuis le serveur
  try {
    const data = await settingsApi.get()
    store.setSettings(data)
    // Appliquer la langue
    const lang = data.language || 'fr'
    await i18n.setLocale(lang)
    // Appliquer le thème
    document.documentElement.setAttribute('data-theme', data.theme || 'dark')
  } catch (e) {
    console.warn('[app] Impossible de charger les settings :', e.message)
    await i18n.setLocale('fr')
  }

  // Créer et monter l'app Vue
  const app = createApp(AppShell)
  app.provide('router', router)
  app.provide('store', store)
  app.provide('t', t)

  router.init()
  app.mount('#app')
}

init()
```

  avec `import`. L'utiliser directement dans les composants graphiques (T4.x).

  composants enfants d'AppShell (qui fait `provide`).
