## Pattern commun à tous les composants

Chaque composant suit ce pattern de chargement :

```javascript
setup() {
  const state = reactive({
    loading: false,
    error: null,
    data: null,        // ou [] selon le cas
  })

  async function load() {
    state.loading = true
    state.error = null
    try {
      state.data = await api.xxx.yyy()
    } catch (err) {
      state.error = err.message
    } finally {
      state.loading = false
    }
  }

  onMounted(load)
  return { state, load, t }
}
```

Template :
```html
<LoadingSpinner v-if="state.loading" />
<ErrorBanner v-else-if="state.error" :message="state.error" :retry="load" />
<EmptyState v-else-if="!state.data?.length" ... />
<div v-else> ... contenu ... </div>
```

```javascript
import { reactive, onMounted, inject } from '/vendor/vue.esm-browser.prod.js'
import { exercises as api } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader      from '../layout/PageHeader.js'
import LoadingSpinner  from '../common/LoadingSpinner.js'
import ErrorBanner     from '../common/ErrorBanner.js'
import EmptyState      from '../common/EmptyState.js'
import ExerciseCard    from './ExerciseCard.js'
import ExerciseFilter  from './ExerciseFilter.js'

export default {
  name: 'ExerciseListView',
  components: { PageHeader, LoadingSpinner, ErrorBanner, EmptyState, ExerciseCard, ExerciseFilter },

  setup() {
    const router = inject('router')

    const state = reactive({
      loading: false,
      error: null,
      exercises: [],
      filters: { muscle: '', equipment: '', movement: '', q: '' },
    })

    async function load() {
      state.loading = true
      state.error = null
      try {
        // Construire les params en excluant les valeurs vides
        const params = {}
        if (state.filters.muscle)    params.muscle    = state.filters.muscle
        if (state.filters.equipment) params.equipment = state.filters.equipment
        if (state.filters.movement)  params.movement  = state.filters.movement
        if (state.filters.q)         params.q         = state.filters.q
        const data = await api.list(params)
        state.exercises = data.exercises
      } catch (err) {
        state.error = err.message
      } finally {
        state.loading = false
      }
    }

    function onFilterChange(newFilters) {
      Object.assign(state.filters, newFilters)
      load()
    }

    onMounted(load)

    return { state, load, onFilterChange, router, t }
  },

  template: `
    <div class="view">
      <PageHeader :title="t('exercises.title')">
        <template #actions>
          <button class="btn btn--primary" style="padding:0.4rem 0.75rem;font-size:0.85rem"
            @click="router.push('#/exercises/new')">
            +
          </button>
        </template>
      </PageHeader>

      <div class="view__content">
        <ExerciseFilter :filters="state.filters" @change="onFilterChange" />

        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner   v-else-if="state.error"
          :message="state.error" :retry="load" />
        <EmptyState    v-else-if="state.exercises.length === 0"
          icon="🔍" :message="t('exercises.no_results')" />
        <div v-else class="exercise-list">
          <ExerciseCard
            v-for="ex in state.exercises"
            :key="ex.id"
            :exercise="ex"
            @click="router.push('#/exercises/' + ex.id)"
          />
        </div>
      </div>
    </div>
  `
}
```

## `ExerciseFilter.js`

```javascript
import { reactive } from '/vendor/vue.esm-browser.prod.js'
import { t } from '../../i18n.js'

const MUSCLES = [
  'chest','back','shoulders','biceps','triceps',
  'quads','hamstrings','glutes','calves','core','traps','forearms'
]
const EQUIPMENT = [
  'barbell','ez_bar','dumbbell_fixed','dumbbell_adjustable','kettlebell',
  'resistance_band','bench','rack','pullup_bar','cable_machine','other'
]
const MOVEMENTS = [
  'push_horizontal','push_vertical','pull_horizontal','pull_vertical',
  'squat','hinge','lunge','core','other'
]

export default {
  name: 'ExerciseFilter',
  props: {
    filters: { type: Object, required: true }
    // { muscle, equipment, movement, q }
  },
  emits: ['change'],

  setup(props, { emit }) {
    const local = reactive({ ...props.filters })

    function update(key, value) {
      local[key] = value
      emit('change', { ...local })
    }

    function clearAll() {
      local.muscle = ''; local.equipment = ''; local.movement = ''; local.q = ''
      emit('change', { ...local })
    }

    const hasFilters = () => local.muscle || local.equipment || local.movement || local.q

    return { local, update, clearAll, hasFilters, t, MUSCLES, EQUIPMENT, MOVEMENTS }
  },

  template: `
    <div class="exercise-filter">
      <div class="exercise-filter__search">
        <input
          class="form-input"
          type="search"
          :placeholder="t('exercises.search')"
          :value="local.q"
          @input="update('q', $event.target.value)"
        />
      </div>
      <div class="exercise-filter__chips">
        <select class="form-select form-select--compact"
          :value="local.muscle" @change="update('muscle', $event.target.value)">
          <option value="">{{ t('exercises.filter_muscle') }}</option>
          <option v-for="m in MUSCLES" :key="m" :value="m">{{ t('muscles.' + m) }}</option>
        </select>
        <select class="form-select form-select--compact"
          :value="local.movement" @change="update('movement', $event.target.value)">
          <option value="">{{ t('exercises.filter_movement') }}</option>
          <option v-for="mv in MOVEMENTS" :key="mv" :value="mv">{{ t('movement.' + mv) }}</option>
        </select>
        <button v-if="hasFilters()" class="btn btn--secondary" style="padding:0.4rem 0.75rem"
          @click="clearAll()">✕</button>
      </div>
    </div>
  `
}
```

```javascript
import { reactive, onMounted, inject } from '/vendor/vue.esm-browser.prod.js'
import { exercises as api } from '../../api.js'
import { estimate1RM, rmTable } from '../../utils/rm.js'
import { t } from '../../i18n.js'
import PageHeader     from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner    from '../common/ErrorBanner.js'

export default {
  name: 'ExerciseDetailView',
  props: { id: { type: String, required: true } },
  components: { PageHeader, LoadingSpinner, ErrorBanner },

  setup(props) {
    const router = inject('router')
    const store  = inject('store')

    const state = reactive({
      loading: false, error: null,
      exercise: null, history: [],
      imageIndex: 0,
    })

    async function load() {
      state.loading = true; state.error = null
      try {
        const [ex, hist] = await Promise.all([
          api.get(props.id),
          api.history(props.id, { limit: 5 })
        ])
        state.exercise = ex
        state.history  = hist.history
      } catch (err) {
        state.error = err.message
      } finally {
        state.loading = false
      }
    }

    onMounted(load)

    // RM estimé depuis la dernière performance
    function lastRM() {
      if (!state.history?.length) return null
      const lastSession = state.history[0]
      const bestSet = lastSession.sets.reduce((best, s) =>
        (s.is_completed && s.reps > 0 && s.weight_kg > estimate1RM(best?.weight_kg||0, best?.reps||1))
          ? s : best
      , null)
      if (!bestSet) return null
      const formula = store.getSetting('rm_formula', 'epley')
      return { rm: estimate1RM(bestSet.weight_kg, bestSet.reps, formula), set: bestSet }
    }

    return { state, load, router, t, lastRM }
  },

  template: `
    <div class="view">
      <PageHeader
        :title="state.exercise?.name_fr || '…'"
        :show-back="true"
      >
        <template #actions v-if="state.exercise?.is_custom">
          <button class="btn btn--secondary"
            style="padding:0.4rem 0.75rem;font-size:0.85rem"
            @click="router.push('#/exercises/' + state.exercise.id + '/edit')">
            {{ t('common.edit') }}
          </button>
        </template>
      </PageHeader>

        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner v-else-if="state.error" :message="state.error" :retry="load" />

        <template v-else-if="state.exercise">

              :alt="state.exercise.name_fr"
              class="exercise-detail__img"
              @error="$event.target.style.display='none'"
            />
            <div v-if="state.exercise.image_paths.length > 1" class="exercise-detail__dots">
              <span
                v-for="(_, i) in state.exercise.image_paths" :key="i"

            </div>

          <!-- Muscles -->

            <p>{{ state.exercise.primary_muscles.map(m => t('muscles.' + m) || m).join(', ') }}</p>
```

## `ExerciseForm.js`

```javascript
import { reactive, onMounted, inject } from '/vendor/vue.esm-browser.prod.js'
import { exercises as api } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader    from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner   from '../common/ErrorBanner.js'

const MUSCLES = ['chest','back','shoulders','biceps','triceps',
  'quads','hamstrings','glutes','calves','core','traps','forearms','lower_back']
const MOVEMENTS = [
  'push_horizontal','push_vertical','pull_horizontal','pull_vertical',
  'squat','hinge','lunge','carry','core','rotation','other'
]

export default {
  name: 'ExerciseForm',
  // Prop id : présente si édition, absente si création
  props: { id: { type: String, default: null } },
  components: { PageHeader, LoadingSpinner, ErrorBanner },

  setup(props) {
    const router = inject('router')
    const store  = inject('store')

    const isEdit = !!props.id
    const state = reactive({
      loading: isEdit, error: null, saving: false,
      form: {
        name_fr: '', name_en: '', category: 'strength',
        movement_pattern: 'other', force: '', mechanic: '',
        primary_muscles: [], secondary_muscles: [],
        equipment_required: [], instructions_fr: [], instructions_en: [],
        is_unilateral: false,
      }
    })

    async function load() {
      if (!isEdit) return
      try {
        const ex = await api.get(props.id)
        if (!ex.is_custom) {
          state.error = 'Impossible de modifier un exercice intégré.'
          return
        }
        Object.assign(state.form, {
          name_fr: ex.name_fr, name_en: ex.name_en,
          category: ex.category, movement_pattern: ex.movement_pattern,
          force: ex.force || '', mechanic: ex.mechanic || '',
          primary_muscles: [...ex.primary_muscles],
          secondary_muscles: [...ex.secondary_muscles],
          equipment_required: [...ex.equipment_required],
          is_unilateral: ex.is_unilateral,
        })
      } catch (err) {
        state.error = err.message
      } finally {
        state.loading = false
      }
    }

    function toggleMuscle(list, muscle) {
      const idx = list.indexOf(muscle)
      if (idx >= 0) list.splice(idx, 1)
      else list.push(muscle)
    }

    async function save() {
      if (!state.form.name_fr.trim()) {
        store.showToast('Le nom français est obligatoire.', 'error')
        return
      }
      if (!state.form.primary_muscles.length) {
        store.showToast('Choisir au moins un muscle primaire.', 'error')
        return
      }
      state.saving = true
      try {
        if (isEdit) {
          await api.update(props.id, state.form)
          store.showToast('Exercice modifié.', 'success')
        } else {
          await api.create(state.form)
          store.showToast('Exercice créé.', 'success')
        }
        router.back()
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        state.saving = false
      }
    }

    onMounted(load)
    return { state, save, toggleMuscle, isEdit, router, t, MUSCLES, MOVEMENTS }
  },

  template: `
    <div class="view">
      <PageHeader
        :title="isEdit ? t('exercises.edit_title') : t('exercises.new_title')"
        :show-back="true"
      />
      <div class="view__content" style="padding:0 1rem">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner v-else-if="state.error" :message="state.error" />

        <form v-else @submit.prevent="save" class="exercise-form">
          <div class="form-group">
            <label class="form-label">Nom (français) *</label>
            <input class="form-input" v-model="state.form.name_fr" required />
          </div>
          <div class="form-group">
            <label class="form-label">Nom (anglais)</label>
            <input class="form-input" v-model="state.form.name_en" />
          </div>
          <div class="form-group">
            <label class="form-label">{{ t('exercises.filter_movement') }}</label>
            <select class="form-select" v-model="state.form.movement_pattern">
              <option v-for="mv in MOVEMENTS" :key="mv" :value="mv">
                {{ t('movement.' + mv) }}
              </option>
            </select>
          </div>
          <div class="form-group">
            <label class="form-label">{{ t('exercises.primary') }} *</label>
            <div class="muscle-chips">
              <button
                v-for="m in MUSCLES" :key="m" type="button"
                class="chip"
                :class="{ 'chip--active': state.form.primary_muscles.includes(m) }"
                @click="toggleMuscle(state.form.primary_muscles, m)"
              >{{ t('muscles.' + m) }}</button>
            </div>
          </div>
          <div class="form-group">
            <label class="form-label">{{ t('exercises.secondary') }}</label>
            <div class="muscle-chips">
              <button
                v-for="m in MUSCLES" :key="m" type="button"
                class="chip"
                :class="{ 'chip--active': state.form.secondary_muscles.includes(m) }"
                @click="toggleMuscle(state.form.secondary_muscles, m)"
              >{{ t('muscles.' + m) }}</button>
            </div>
          </div>
          <div class="form-group">
            <label class="form-label">
              <input type="checkbox" v-model="state.form.is_unilateral" />
              Exercice unilatéral (un bras / une jambe)
            </label>
          </div>
          <button type="submit" class="btn btn--primary btn--full" :disabled="state.saving">
            {{ state.saving ? t('common.loading') : t('common.save') }}
          </button>
        </form>
      </div>
    </div>
  `
}
```

```javascript
export default {
  name: 'EquipmentView',
  components: {
    PageHeader, LoadingSpinner, ErrorBanner, EmptyState,
    EquipmentItemRow, EquipmentForm, ConfirmModal
  },

  setup() {
    const store = inject('store')

    const state = reactive({
      loading: false, error: null, items: [],
      showForm: false, editItem: null,
      deleteTarget: null,
    })

    async function load() {
      state.loading = true; state.error = null
      try {
        const data = await api.list()
        state.items = data.items
      } catch (err) {

    async function onSaved() {
      closeForm()
      await load()
      store.showToast('Équipement sauvegardé.', 'success')
    }

    function confirmDelete(item) { state.deleteTarget = item }

    async function doDelete() {
      if (!state.deleteTarget) return
      try {
        await api.delete(state.deleteTarget.id)
        store.showToast('Supprimé.', 'success')
        state.deleteTarget = null
        await load()
      } catch (err) {
        store.showToast(err.message, 'error')
      }
    }

    onMounted(load)

    return {
      state, load, t,
      openAdd, openEdit, closeForm, onSaved,
      confirmDelete, doDelete,
    }
  },

  template: `
    <div class="view">
      <PageHeader :title="t('equipment.title')">
        <template #actions>
          <button class="btn btn--primary" style="padding:0.4rem 0.75rem;font-size:0.85rem"
            @click="openAdd()">+</button>
        </template>
      </PageHeader>

      <div class="view__content">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner   v-else-if="state.error" :message="state.error" :retry="load" />
        <EmptyState    v-else-if="!state.items.length"
          icon="🏋️" :message="t('equipment.no_equipment')">
          <button class="btn btn--primary" style="margin-top:1rem" @click="openAdd()">
            {{ t('equipment.add') }}
          </button>
        </EmptyState>
        <div v-else>
          <EquipmentItemRow
            v-for="item in state.items" :key="item.id"
            :item="item"
            @edit="openEdit(item)"
            @delete="confirmDelete(item)"
          />
        </div>
      </div>

      <!-- Formulaire ajout/édition (bottom sheet) -->
      <div v-if="state.showForm" class="modal-overlay" @click.self="closeForm()">
        <div class="modal">
          <EquipmentForm
            :item="state.editItem"
            @saved="onSaved"
            @cancel="closeForm()"
          />
        </div>
      </div>

      <!-- Confirmation suppression -->
      <ConfirmModal
        v-if="state.deleteTarget"
        :title="t('equipment.delete_confirm')"
        :body="state.deleteTarget.name"
        :danger="true"
        @confirm="doDelete()"
        @cancel="state.deleteTarget = null"
      />
    </div>
  `
}
```

## `EquipmentForm.js`

```javascript
import { reactive, watch, inject } from '/vendor/vue.esm-browser.prod.js'
import { equipment as api } from '../../api.js'
import { t } from '../../i18n.js'

const CATEGORIES = [
  'barbell','ez_bar','plate','dumbbell_fixed','dumbbell_adjustable',
  'kettlebell','resistance_band','bench','rack','pullup_bar',
  'cable_machine','smith_machine','other'
]

const CATEGORY_LABELS = {
  barbell:'Barre droite', ez_bar:'Barre EZ', plate:'Disque',
  dumbbell_fixed:'Haltère fixe', dumbbell_adjustable:'Haltère réglable',
  kettlebell:'Kettlebell', resistance_band:'Élastique',
  bench:'Banc', rack:'Rack/Cage', pullup_bar:'Barre de traction',
  cable_machine:'Machine câble / poulie', smith_machine:'Smith machine',
  other:'Autre'
}

export default {
  name: 'EquipmentForm',
  props: {
    item: { type: Object, default: null }  // null = création, objet = édition
  },
  emits: ['saved','cancel'],

  setup(props, { emit }) {
    const store = inject('store')

    const blank = () => ({
      category: 'barbell', name: '', weight_kg: '',
      count: 1, weight_min_kg: '', weight_max_kg: '',
      weight_step_kg: '', resistance_min_kg: '', resistance_max_kg: '', notes: ''
    })

    const form = reactive(props.item ? {
      category:          props.item.category,
      name:              props.item.name,
      weight_kg:         props.item.weight_kg ?? '',
      count:             props.item.count ?? 1,
      weight_min_kg:     props.item.weight_min_kg ?? '',
      weight_max_kg:     props.item.weight_max_kg ?? '',
      weight_step_kg:    props.item.weight_step_kg ?? '',
      resistance_min_kg: props.item.resistance_min_kg ?? '',
      resistance_max_kg: props.item.resistance_max_kg ?? '',
      notes:             props.item.notes ?? '',
    } : blank())

    const saving = reactive({ value: false })

    async function save() {
      if (!form.name.trim()) {
        store.showToast('Le nom est obligatoire.', 'error')
        return
      }
      // Validation plaques
      if (form.category === 'plate' && form.count % 2 !== 0) {
        store.showToast('Les disques doivent être en quantité paire.', 'error')
        return
      }
      saving.value = true
      try {
        // Nettoyer les champs vides → null
        const body = {}
        for (const [k, v] of Object.entries(form)) {
          body[k] = (v === '' ? null : v)
        }
        if (props.item) {
          await api.update(props.item.id, body)
        } else {
          await api.create(body)
        }
        emit('saved')
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        saving.value = false
      }
    }

    // Afficher les champs selon la catégorie
    const showWeight       = () => ['barbell','ez_bar','dumbbell_fixed','kettlebell'].includes(form.category)
    const showCount        = () => ['plate','dumbbell_fixed','kettlebell'].includes(form.category)
    const showAdjustable   = () => form.category === 'dumbbell_adjustable'
    const showResistance   = () => form.category === 'resistance_band'

    return {
      form, saving, save, t,
      CATEGORIES, CATEGORY_LABELS,
      showWeight, showCount, showAdjustable, showResistance,
      isEdit: !!props.item,
    }
  },

  template: `
    <form @submit.prevent="save">
      <h2 class="modal__title">
        {{ isEdit ? t('common.edit') : t('equipment.add') }}
      </h2>
      <div class="form-group">
        <label class="form-label">{{ t('equipment.category') }}</label>
        <select class="form-select" v-model="form.category">
          <option v-for="c in CATEGORIES" :key="c" :value="c">{{ CATEGORY_LABELS[c] }}</option>
        </select>
      </div>
      <div class="form-group">
        <label class="form-label">{{ t('equipment.name') }}</label>
        <input class="form-input" v-model="form.name" required />
      </div>
      <div v-if="showWeight()" class="form-group">
        <label class="form-label">Poids (kg)</label>
        <input class="form-input" type="number" step="0.5" min="0" v-model="form.weight_kg" />
      </div>
      <div v-if="showCount()" class="form-group">
        <label class="form-label">
          {{ t('equipment.count') }}
          <span v-if="form.category === 'plate'" style="color:var(--color-text-muted);font-size:0.8rem">
            (nombre total, doit être pair)
          </span>
        </label>
        <input class="form-input" type="number" min="1" v-model="form.count" />
      </div>
      <template v-if="showAdjustable()">
        <div class="form-group">
          <label class="form-label">Poids minimum (kg)</label>
          <input class="form-input" type="number" step="0.5" min="0" v-model="form.weight_min_kg" />
        </div>
        <div class="form-group">
          <label class="form-label">Poids maximum (kg)</label>
          <input class="form-input" type="number" step="0.5" min="0" v-model="form.weight_max_kg" />
        </div>
        <div class="form-group">
          <label class="form-label">Incrément (kg)</label>
          <input class="form-input" type="number" step="0.5" min="0.5" v-model="form.weight_step_kg" />
        </div>
      </template>
      <template v-if="showResistance()">
        <div class="form-group">
          <label class="form-label">Résistance min (kg équiv.)</label>
          <input class="form-input" type="number" step="0.5" min="0" v-model="form.resistance_min_kg" />
        </div>
        <div class="form-group">
          <label class="form-label">Résistance max (kg équiv.)</label>
          <input class="form-input" type="number" step="0.5" min="0" v-model="form.resistance_max_kg" />
        </div>
      </template>
      <div class="modal__actions">
        <button type="button" class="btn btn--secondary" @click="$emit('cancel')">
          {{ t('common.cancel') }}
        </button>
        <button type="submit" class="btn btn--primary" :disabled="saving.value">
          {{ saving.value ? t('common.loading') : t('common.save') }}
        </button>
      </div>
    </form>
  `
}
```
