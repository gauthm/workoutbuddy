const SETS_REPS = {
  strength: {
    compound:  { sets: 5, rmin: 3, rmax: 5,  rest: 180 },
    isolation: { sets: 3, rmin: 6, rmax: 8,  rest: 90  },
  },

  weight_loss: {
    compound:  { sets: 3, rmin: 12, rmax: 15, rest: 60 },

  },
  general_fitness: {
    compound:  { sets: 3, rmin: 8, rmax: 12, rest: 90  },
    isolation: { sets: 3, rmin: 10, rmax: 15, rest: 75 },
  },
}

const DEFAULT_MODEL = {
  strength: 'linear', hypertrophy: 'double_progression',
  endurance: 'double_progression', weight_loss: 'double_progression',
  general_fitness: 'double_progression',
}

function selectSplit(daysPerWeek, goal, level) {
  if (daysPerWeek === 2) return 'full_body_2'
  if (daysPerWeek === 3 && level === 'beginner') return 'full_body_3'
  if (daysPerWeek === 3) return 'push_pull_legs'
  if (daysPerWeek === 4) return 'upper_lower'
  if (daysPerWeek === 5) return 'upper_lower_x5'
  return 'ppl_x2'  // 6 jours
}

function isEquipmentAvailable(equipmentRequired, equipmentItems) {
  if (!equipmentRequired || equipmentRequired.length === 0) return true
  const available = new Set(equipmentItems.map(i => i.category))
  // Cas spéciaux : dumbbell_fixed et dumbbell_adjustable couvrent 'dumbbell'
  if (available.has('dumbbell_fixed') || available.has('dumbbell_adjustable')) {
    available.add('dumbbell_fixed')
    available.add('dumbbell_adjustable')
  }
  return equipmentRequired.every(req => {
    if (req === 'dumbbell_fixed' || req === 'dumbbell_adjustable') {
      return available.has('dumbbell_fixed') || available.has('dumbbell_adjustable')
    }
    return available.has(req)
  })
}

export function generateProgram(params, allExercises, equipmentItems) {
  // params: { goal, level, days_per_week, duration_weeks?, priority_muscles[], name? }
  // allExercises: Exercise[] depuis la DB
  // equipmentItems: EquipmentItem[] depuis la DB

  const { goal, level, days_per_week, duration_weeks, priority_muscles = [] } = params

  const splitKey = selectSplit(days_per_week, goal, level)
  const templateDefs = SPLIT_TEMPLATES[splitKey]

  const templates = []

  for (const tdef of templateDefs) {
    // Booster les muscles prioritaires : les mettre en premier
    let muscles = [...tdef.muscles]
    for (const pm of [...priority_muscles].reverse()) {
      if (muscles.includes(pm)) {
        muscles = [pm, ...muscles.filter(m => m !== pm)]
      }
    }

    const exercises = []
    const usedExercises = new Set()

    for (const muscle of muscles) {
      // Candidats pour ce muscle
      const candidates = allExercises.filter(ex =>
        ex.primary_muscles.includes(muscle) &&
        isEquipmentAvailable(ex.equipment_required, equipmentItems) &&
        !usedExercises.has(ex.id)
      )
      if (candidates.length === 0) continue

      // Préférer les composés pour la force, mix sinon
      const preferred = candidates.filter(e => e.mechanic === 'compound')
      const selected = preferred.length > 0 ? preferred[0] : candidates[0]

      const mechanic = selected.mechanic === 'compound' ? 'compound' : 'isolation'
      const cfg = SETS_REPS[goal]?.[mechanic] || SETS_REPS.general_fitness.compound

      // Réduire les séries pour les débutants
      const sets = level === 'beginner' ? Math.max(2, cfg.sets - 1) : cfg.sets

      exercises.push({
        exercise_id:   selected.id,
        target_sets:   sets,
        target_reps_min: cfg.rmin,
        target_reps_max: cfg.rmax,
        rest_seconds:  cfg.rest,
        progression_model: 'inherit',
      })
      usedExercises.add(selected.id)

      // Si muscle prioritaire ET hypertrophie → ajouter un exercice isolation
      if (priority_muscles.includes(muscle) && goal === 'hypertrophy') {
        const isolations = candidates.filter(e =>
          e.mechanic === 'isolation' && !usedExercises.has(e.id)
        )
        if (isolations.length > 0) {
          const iso = isolations[0]
          const isoCfg = SETS_REPS[goal].isolation
          exercises.push({
            exercise_id:     iso.id,
            target_sets:     level === 'beginner' ? 2 : isoCfg.sets,
            target_reps_min: isoCfg.rmin,
            target_reps_max: isoCfg.rmax,
            rest_seconds:    isoCfg.rest,
            progression_model: 'inherit',
          })
          usedExercises.add(iso.id)
        }
      }
    }

    if (exercises.length > 0) {
      templates.push({ name: tdef.name, day_of_week: null, exercises })
    }
  }

  return {
    name: params.name ||
      `${level === 'beginner' ? 'Débutant' : level === 'intermediate' ? 'Intermédiaire' : 'Avancé'} — ${goal} ${days_per_week}j`,
    goal,
    level,
    days_per_week,
    duration_weeks: duration_weeks || null,
    progression_model: DEFAULT_MODEL[goal] || 'double_progression',
    use_exercise_rotation: 0,
    deload_auto_enabled: 0,
    templates,
  }
}

import express from 'express'
import { getDb } from '../db/database.js'
import { generateProgram } from '../services/programGenerator.js'

const router = express.Router()
const db = getDb()

function parseProgram(row) {
  if (!row) return null
  return {
    ...row,
    is_active:              row.is_active === 1,
    use_exercise_rotation:  row.use_exercise_rotation === 1,
    deload_auto_enabled:    row.deload_auto_enabled === 1,
    is_in_deload:           row.is_in_deload === 1,
  }
}

// ── GET /programs ─────────────────────────────────────────────
router.get('/', (req, res) => {
  try {
    let sql = 'SELECT * FROM program'
    const params = []
    if (req.query.active !== undefined) {
      sql += ' WHERE is_active = ?'
      params.push(parseInt(req.query.active))
    }
    sql += ' ORDER BY created_at DESC'
    const rows = db.prepare(sql).all(...params)
    res.json({ programs: rows.map(parseProgram) })
  } catch (err) {
    console.error('[GET /programs]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── GET /programs/:id ─────────────────────────────────────────
router.get('/:id', (req, res) => {
  try {
    const program = db.prepare('SELECT * FROM program WHERE id = ?').get(req.params.id)
    if (!program) return res.status(404).json({ error: 'PROGRAM_NOT_FOUND' })

    const templates = db.prepare(
      'SELECT * FROM workout_template WHERE program_id = ? ORDER BY order_index ASC'
    ).all(program.id)

    for (const tpl of templates) {
      tpl.exercises = db.prepare(
        'SELECT * FROM template_exercise WHERE workout_template_id = ? ORDER BY order_index ASC'
      ).all(tpl.id)
    }

    res.json({ ...parseProgram(program), templates })
  } catch (err) {
    console.error('[GET /programs/:id]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── POST /programs ────────────────────────────────────────────
router.post('/', (req, res) => {
  try {
    const {
      name, description, goal, level, days_per_week, duration_weeks,
      use_exercise_rotation, rotation_frequency_weeks, progression_model,
      deload_auto_enabled, deload_frequency_weeks,
      deload_load_reduction_pct, deload_volume_reduction_pct,
    } = req.body

    // Validations
    if (!name?.trim()) return res.status(400).json({ error: 'MISSING_NAME' })
    const VALID_GOALS = ['strength','hypertrophy','endurance','weight_loss','general_fitness']
    if (!VALID_GOALS.includes(goal)) return res.status(400).json({ error: 'INVALID_GOAL' })
    const VALID_LEVELS = ['beginner','intermediate','advanced']
    if (!VALID_LEVELS.includes(level)) return res.status(400).json({ error: 'INVALID_LEVEL' })
    const dpw = parseInt(days_per_week)
    if (!dpw || dpw < 2 || dpw > 6) return res.status(400).json({ error: 'INVALID_DAYS_PER_WEEK' })

    const result = db.prepare(`
      INSERT INTO program
        (name, description, goal, level, days_per_week, duration_weeks,
         use_exercise_rotation, rotation_frequency_weeks, progression_model,
         deload_auto_enabled, deload_frequency_weeks,
         deload_load_reduction_pct, deload_volume_reduction_pct)
      VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?)
    `).run(
      name.trim(), description || null, goal, level, dpw,
      duration_weeks || null,
      use_exercise_rotation ? 1 : 0,
      rotation_frequency_weeks || 1,
      progression_model || 'double_progression',
      deload_auto_enabled ? 1 : 0,
      deload_frequency_weeks || 6,
      deload_load_reduction_pct || 20,
      deload_volume_reduction_pct || 40,
    )

    const created = db.prepare('SELECT * FROM program WHERE id = ?').get(result.lastInsertRowid)
    res.status(201).json(parseProgram(created))
  } catch (err) {
    console.error('[POST /programs]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── PUT /programs/:id ─────────────────────────────────────────
router.put('/:id', (req, res) => {
  try {
    const existing = db.prepare('SELECT id FROM program WHERE id = ?').get(req.params.id)
    if (!existing) return res.status(404).json({ error: 'PROGRAM_NOT_FOUND' })

    const UPDATABLE = [
      'name','description','goal','level','days_per_week','duration_weeks',
      'use_exercise_rotation','rotation_frequency_weeks','progression_model',
      'deload_auto_enabled','deload_frequency_weeks',
      'deload_load_reduction_pct','deload_volume_reduction_pct',
    ]
    const updates = []; const values = []
    for (const f of UPDATABLE) {
      if (req.body[f] !== undefined) {
        updates.push(`${f} = ?`)
        values.push(req.body[f])
      }
    }
    if (updates.length === 0) return res.status(400).json({ error: 'NO_FIELDS_TO_UPDATE' })
    updates.push("updated_at = datetime('now')")
    values.push(req.params.id)
    db.prepare(`UPDATE program SET ${updates.join(', ')} WHERE id = ?`).run(...values)

    const updated = db.prepare('SELECT * FROM program WHERE id = ?').get(req.params.id)
    res.json(parseProgram(updated))
  } catch (err) {
    console.error('[PUT /programs/:id]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── DELETE /programs/:id ──────────────────────────────────────
router.delete('/:id', (req, res) => {
  try {
    const existing = db.prepare('SELECT * FROM program WHERE id = ?').get(req.params.id)
    if (!existing) return res.status(404).json({ error: 'PROGRAM_NOT_FOUND' })

    const inProgress = db.prepare(
      "SELECT COUNT(*) as n FROM session WHERE program_id = ? AND status = 'in_progress'"
    ).get(req.params.id)
    if (inProgress.n > 0) {
      return res.status(409).json({ error: 'CONFLICT', detail: 'Une séance est en cours.' })
    }

    db.prepare('DELETE FROM program WHERE id = ?').run(req.params.id)
    res.status(204).end()
  } catch (err) {
    console.error('[DELETE /programs/:id]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

router.post('/:id/activate', (req, res) => {

      equipmentItems

    // Persister en DB dans une transaction
    const programId = db.transaction(() => {
      const pResult = db.prepare(`
        INSERT INTO program
          (name, goal, level, days_per_week, duration_weeks, progression_model,
           use_exercise_rotation, deload_auto_enabled)
        VALUES (?,?,?,?,?,?,0,0)
      `).run(
        draft.name, draft.goal, draft.level, draft.days_per_week,
        draft.duration_weeks, draft.progression_model
      )
      const pid = pResult.lastInsertRowid

      for (let ti = 0; ti < draft.templates.length; ti++) {
        const tpl = draft.templates[ti]
        const tResult = db.prepare(`
          INSERT INTO workout_template (program_id, name, order_index)
          VALUES (?, ?, ?)
        `).run(pid, tpl.name, ti)
        const tid = tResult.lastInsertRowid

        for (let ei = 0; ei < tpl.exercises.length; ei++) {
          const ex = tpl.exercises[ei]
          const teResult = db.prepare(`
            INSERT INTO template_exercise
              (workout_template_id, exercise_id, order_index,
               target_sets, target_reps_min, target_reps_max, rest_seconds, progression_model)

      'SELECT * FROM workout_template WHERE program_id = ? ORDER BY order_index'
    ).all(programId)

      tpl.exercises = db.prepare(

      "SELECT workout_template_id FROM session WHERE program_id = ? AND status = 'completed' ORDER BY completed_at DESC LIMIT 1"

    let nextTemplate
    if (!lastSession || !lastSession.workout_template_id) {
      nextTemplate = templates[0]
    } else {
      const lastIdx = templates.findIndex(t => t.id === lastSession.workout_template_id)
      nextTemplate = templates[(lastIdx + 1) % templates.length]
    }

    const templateExercises = db.prepare(
      'SELECT te.*, e.name_fr, e.name_en, e.equipment_required, e.is_unilateral FROM template_exercise te JOIN exercise e ON te.exercise_id = e.id WHERE te.workout_template_id = ? ORDER BY te.order_index'
    ).all(nextTemplate.id)

    const equipmentItems = db.prepare('SELECT * FROM equipment_item').all()

    // Pour chaque exercice, calculer la charge cible depuis progression_state
    const exercises = templateExercises.map(te => {
      const state = db.prepare(
        'SELECT * FROM progression_state WHERE template_exercise_id = ?'
      ).get(te.id)

      const equipReq = JSON.parse(te.equipment_required || '[]')

      // Pour l'instant : retourner la charge courante + context
      const currentWeight = state?.current_weight_kg || 0
