## `ProgramListView.js`

```javascript
import { reactive, onMounted, inject } from '/vendor/vue.esm-browser.prod.js'
import { programs as api } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader     from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner    from '../common/ErrorBanner.js'
import EmptyState     from '../common/EmptyState.js'
import ProgramCard    from './ProgramCard.js'

export default {
  name: 'ProgramListView',
  components: { PageHeader, LoadingSpinner, ErrorBanner, EmptyState, ProgramCard },

  setup() {
    const router = inject('router')
    const state = reactive({ loading: false, error: null, programs: [] })

    async function load() {
      state.loading = true; state.error = null
      try {
        const data = await api.list()
        state.programs = data.programs
      } catch (err) { state.error = err.message }
      finally { state.loading = false }
    }

    onMounted(load)
    return { state, load, router, t }
  },

  template: `
    <div class="view">
      <PageHeader :title="t('programs.title')">
        <template #actions>
          <button class="btn btn--primary" style="padding:0.4rem 0.75rem;font-size:0.85rem"
            @click="router.push('#/programs/new')">+</button>
        </template>
      </PageHeader>
      <div class="view__content">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner   v-else-if="state.error" :message="state.error" :retry="load" />
        <EmptyState    v-else-if="!state.programs.length"
          icon="📋" :message="t('programs.no_programs')">
          <button class="btn btn--primary" style="margin-top:1rem"
            @click="router.push('#/programs/new')">
            {{ t('programs.new') }}
          </button>
        </EmptyState>
        <div v-else>
          <ProgramCard
            v-for="p in state.programs" :key="p.id"
            :program="p"
            @click="router.push('#/programs/' + p.id)"
          />
        </div>
      </div>
    </div>
  `
}
```

## `ProgramDetailView.js`

```javascript
import { reactive, onMounted, inject } from '/vendor/vue.esm-browser.prod.js'
import { programs as api, sessions as sessionsApi } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader     from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner    from '../common/ErrorBanner.js'
import ConfirmModal   from '../common/ConfirmModal.js'

export default {
  name: 'ProgramDetailView',
  props: { id: { type: String, required: true } },
  components: { PageHeader, LoadingSpinner, ErrorBanner, ConfirmModal },

  setup(props) {
    const router = inject('router')
    const store  = inject('store')
    const state = reactive({
      loading: false, error: null, program: null,
      activating: false, showDeleteConfirm: false,
    })

    async function load() {
      state.loading = true; state.error = null
      try { state.program = await api.get(props.id) }
      catch (err) { state.error = err.message }
      finally { state.loading = false }
    }

    async function activate() {
      state.activating = true
      try {
        await api.activate(props.id)
        store.showToast(t('programs.activated'), 'success')
        await load()
      } catch (err) { store.showToast(err.message, 'error') }
      finally { state.activating = false }
    }

    async function startNextSession() {
      try {
        const next = await api.nextSession(props.id)
        const sessionRes = await sessionsApi.fromTemplate(next.template_id, {
          scheduled_date: next.suggested_date
        })
        await sessionsApi.start(sessionRes.id)
        store.setActiveSession(sessionRes)
        router.push('#/session')
      } catch (err) { store.showToast(err.message, 'error') }
    }

    async function doDelete() {
      try {
        await api.delete(props.id)
        store.showToast('Programme supprimé.', 'success')
        router.push('#/programs')
      } catch (err) {
        store.showToast(err.message, 'error')
        state.showDeleteConfirm = false
      }
    }

    onMounted(load)
    return { state, load, activate, startNextSession, doDelete, router, t }
  },

  template: `
    <div class="view">
      <PageHeader :title="state.program?.name || '…'" :show-back="true">
        <template #actions>
          <button class="btn btn--secondary"
            style="padding:0.4rem 0.75rem;font-size:0.85rem"
            @click="router.push('#/programs/' + id + '/edit')">
            {{ t('common.edit') }}
          </button>
        </template>
      </PageHeader>
      <div class="view__content" style="padding:1rem">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner v-else-if="state.error" :message="state.error" :retry="load" />

        <template v-else-if="state.program">
          <!-- Infos -->
          <div class="detail-section">
            <div style="display:flex;gap:0.75rem;flex-wrap:wrap;margin-bottom:1rem">
              <span class="chip chip--active">{{ state.program.goal }}</span>
              <span class="chip chip--active">{{ state.program.level }}</span>
              <span class="chip chip--active">{{ state.program.days_per_week }}j/sem</span>
              <span v-if="state.program.is_active" class="badge badge--active">
                {{ t('programs.active_badge') }}
              </span>
            </div>
          </div>

          <!-- Actions -->
          <div style="display:flex;flex-direction:column;gap:0.75rem;margin-bottom:1.5rem">
            <button v-if="state.program.is_active" class="btn btn--primary btn--full"
              @click="startNextSession()">
              {{ t('programs.start_session') }}
            </button>
            <button v-else class="btn btn--secondary btn--full"
              :disabled="state.activating" @click="activate()">
              {{ state.activating ? t('common.loading') : t('programs.activate') }}
            </button>
          </div>

          <!-- Templates -->
          <div class="detail-section">
            <h3 class="detail-section__title">{{ t('programs.templates') }}</h3>
            <div v-for="tpl in state.program.templates" :key="tpl.id"
              class="template-block">
              <div class="template-block__name">{{ tpl.name }}</div>
              <div v-for="ex in tpl.exercises" :key="ex.id"
                class="template-block__exercise">
                {{ ex.exercise_id.replace(/_/g,' ') }}
                — {{ ex.target_sets }}×{{ ex.target_reps_min }}
                <template v-if="ex.target_reps_max !== ex.target_reps_min">
                  –{{ ex.target_reps_max }}
                </template>
              </div>
            </div>
          </div>

          <!-- Supprimer -->
          <button class="btn btn--danger btn--full" style="margin-top:2rem"
            @click="state.showDeleteConfirm = true">
            {{ t('common.delete') }}
          </button>
        </template>
      </div>

      <ConfirmModal
        v-if="state.showDeleteConfirm"
        :title="t('programs.delete_confirm')"
        :danger="true"
        @confirm="doDelete()"
        @cancel="state.showDeleteConfirm = false"
      />
    </div>
  `
}
```

## `ProgramGenerator.js`

```javascript
import { reactive, inject } from '/vendor/vue.esm-browser.prod.js'
import { programs as api } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader from '../layout/PageHeader.js'

const GOALS   = ['strength','hypertrophy','endurance','weight_loss','general_fitness']
const LEVELS  = ['beginner','intermediate','advanced']
const MUSCLES = ['chest','back','shoulders','biceps','triceps',
                 'quads','hamstrings','glutes','calves','core']

export default {
  name: 'ProgramGenerator',
  components: { PageHeader },

  setup() {
    const router = inject('router')
    const store  = inject('store')

    const state = reactive({
      generating: false,
      form: {
        goal: 'hypertrophy', level: 'intermediate',
        days_per_week: 3, duration_weeks: null,
        priority_muscles: [], name: '',
      }
    })

    function toggleMuscle(muscle) {
      const idx = state.form.priority_muscles.indexOf(muscle)
      if (idx >= 0) state.form.priority_muscles.splice(idx, 1)
      else state.form.priority_muscles.push(muscle)
    }

    async function generate() {
      state.generating = true
      try {
        const program = await api.generate({
          ...state.form,
          days_per_week: parseInt(state.form.days_per_week),
          duration_weeks: state.form.duration_weeks
            ? parseInt(state.form.duration_weeks) : null,
        })
        store.showToast(t('programs.generated'), 'success')
        router.push('#/programs/' + program.id)
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        state.generating = false
      }
    }

    return { state, generate, toggleMuscle, router, t, GOALS, LEVELS, MUSCLES }
  },

  template: `
    <div class="view">
      <PageHeader :title="t('programs.generator_title')" :show-back="true" />
      <div class="view__content" style="padding:0 1rem">
        <form @submit.prevent="generate">
          <div class="form-group">
            <label class="form-label">Nom (optionnel)</label>
            <input class="form-input" v-model="state.form.name"
              placeholder="Laissez vide pour un nom automatique" />
          </div>
          <div class="form-group">
            <label class="form-label">{{ t('programs.generator_goal') }}</label>
            <select class="form-select" v-model="state.form.goal">
              <option v-for="g in GOALS" :key="g" :value="g">
                {{ t('programs.goal_' + (g === 'general_fitness' ? 'general' : g)) }}
              </option>
            </select>
          </div>
          <div class="form-group">
            <label class="form-label">{{ t('programs.generator_level') }}</label>
            <select class="form-select" v-model="state.form.level">
              <option v-for="l in LEVELS" :key="l" :value="l">
                {{ t('programs.level_' + l) }}
              </option>
            </select>
          </div>
          <div class="form-group">
            <label class="form-label">{{ t('programs.generator_days') }}</label>
            <select class="form-select" v-model="state.form.days_per_week">
              <option v-for="d in [2,3,4,5,6]" :key="d" :value="d">{{ d }}</option>
            </select>
          </div>
          <div class="form-group">
            <label class="form-label">{{ t('programs.generator_muscles') }}</label>
            <div class="muscle-chips">
              <button
                v-for="m in MUSCLES" :key="m" type="button"
                class="chip"
                :class="{ 'chip--active': state.form.priority_muscles.includes(m) }"
                @click="toggleMuscle(m)">
                {{ t('muscles.' + m) }}
              </button>
            </div>
          </div>
          <button type="submit" class="btn btn--primary btn--full"
            :disabled="state.generating" style="margin-top:1rem">
            {{ state.generating ? t('programs.generating') : t('programs.generator_btn') }}
          </button>
        </form>
      </div>
    </div>
  `
}
```

## `ProgramForm.js` (création/édition manuelle simplifiée)

```javascript
import { reactive, onMounted, inject } from '/vendor/vue.esm-browser.prod.js'
import { programs as api } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'

export default {
  name: 'ProgramForm',
  props: { id: { type: String, default: null } },
  components: { PageHeader, LoadingSpinner },

  setup(props) {
    const router = inject('router')
    const store  = inject('store')
    const isEdit = !!props.id

    const state = reactive({
      loading: isEdit, saving: false, error: null,
      form: {
        name: '', goal: 'hypertrophy', level: 'intermediate',
        days_per_week: 3, duration_weeks: '',
        progression_model: 'double_progression',
        deload_auto_enabled: false, deload_frequency_weeks: 6,
      }
    })

    async function load() {
      if (!isEdit) { state.loading = false; return }
      try {
        const p = await api.get(props.id)
        Object.assign(state.form, {
          name: p.name, goal: p.goal, level: p.level,
          days_per_week: p.days_per_week,
          duration_weeks: p.duration_weeks || '',
          progression_model: p.progression_model,
          deload_auto_enabled: p.deload_auto_enabled,
          deload_frequency_weeks: p.deload_frequency_weeks,
        })
      } catch (err) { state.error = err.message }
      finally { state.loading = false }
    }

    async function save() {
      if (!state.form.name.trim()) {
        store.showToast('Le nom est obligatoire.', 'error'); return
      }
      state.saving = true
      try {
        const body = {
          ...state.form,
          days_per_week: parseInt(state.form.days_per_week),
          duration_weeks: state.form.duration_weeks
            ? parseInt(state.form.duration_weeks) : null,
        }
        if (isEdit) {
          await api.update(props.id, body)
          store.showToast('Programme mis à jour.', 'success')
          router.push('#/programs/' + props.id)
        } else {
          const created = await api.create(body)
          store.showToast('Programme créé.', 'success')
          router.push('#/programs/' + created.id)
        }
      } catch (err) { store.showToast(err.message, 'error') }
      finally { state.saving = false }
    }

    onMounted(load)
    return { state, save, isEdit, router, t }
  },

  template: `
    <div class="view">
      <PageHeader
        :title="isEdit ? t('common.edit') : t('programs.new')"
        :show-back="true" />
      <div class="view__content" style="padding:0 1rem">
        <LoadingSpinner v-if="state.loading" />
        <form v-else @submit.prevent="save">
          <div class="form-group">
            <label class="form-label">Nom *</label>
            <input class="form-input" v-model="state.form.name" required />
          </div>
          <div class="form-group">
            <label class="form-label">Objectif</label>
            <select class="form-select" v-model="state.form.goal">
              <option value="strength">Force</option>
              <option value="hypertrophy">Hypertrophie</option>
              <option value="endurance">Endurance</option>
              <option value="weight_loss">Perte de poids</option>
              <option value="general_fitness">Forme générale</option>
            </select>
          </div>
          <div class="form-group">
            <label class="form-label">Niveau</label>
            <select class="form-select" v-model="state.form.level">
              <option value="beginner">Débutant</option>
              <option value="intermediate">Intermédiaire</option>
              <option value="advanced">Avancé</option>
            </select>
          </div>
          <div class="form-group">
            <label class="form-label">Jours par semaine</label>
            <select class="form-select" v-model="state.form.days_per_week">
              <option v-for="d in [2,3,4,5,6]" :key="d" :value="d">{{ d }}</option>
            </select>
          </div>
          <div class="form-group">
            <label class="form-label">Durée (semaines, optionnel)</label>
            <input class="form-input" type="number" min="1" max="52"
              v-model="state.form.duration_weeks" placeholder="Indéfini" />
          </div>
          <div class="form-group">
            <label class="form-label">Modèle de progression</label>
            <select class="form-select" v-model="state.form.progression_model">
              <option value="linear">Linéaire</option>
              <option value="double_progression">Double progression</option>
              <option value="rpe_based">Basé RPE</option>
            </select>
          </div>
          <button type="submit" class="btn btn--primary btn--full"
            :disabled="state.saving" style="margin-top:1rem">
            {{ state.saving ? t('common.loading') : t('common.save') }}
          </button>
        </form>

        <!-- Bouton génération automatique si création -->
        <div v-if="!isEdit" style="margin-top:1.5rem;text-align:center">
          <p style="color:var(--color-text-muted);font-size:0.85rem;margin-bottom:0.5rem">
```

## `ExerciseSelector.js` (picker pour ajouter exercice à un template)

```javascript
import { reactive, onMounted, inject } from '/vendor/vue.esm-browser.prod.js'
import { exercises as api } from '../../api.js'
import { t } from '../../i18n.js'
import LoadingSpinner from '../common/LoadingSpinner.js'

export default {
  name: 'ExerciseSelector',
  emits: ['select', 'cancel'],
  components: { LoadingSpinner },

  setup(_, { emit }) {
    const state = reactive({
      loading: false, exercises: [], q: ''
    })

    async function load() {
      state.loading = true
      try {
        const data = await api.list(state.q ? { q: state.q } : {})
        state.exercises = data.exercises
      } finally { state.loading = false }
    }

    let searchTimer = null
    function onSearch(val) {
      state.q = val
      clearTimeout(searchTimer)
      searchTimer = setTimeout(load, 300)
    }

    onMounted(load)
    return { state, onSearch, t, emit }
  },

  template: `
    <div class="exercise-selector">
      <div style="padding:0.75rem">
        <input class="form-input" type="search"
          :placeholder="t('exercises.search')"
          @input="onSearch($event.target.value)" />
      </div>
```
