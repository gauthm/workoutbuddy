## `SessionTimer.js` (stub pour T2.4 — remplacé en T2.5)

```javascript
import { ref, onMounted, onUnmounted } from '/vendor/vue.esm-browser.prod.js'

export default {
  name: 'SessionTimer',
  props: { startedAt: { type: String, default: null } },

  setup(props) {
    const elapsed = ref(0)
    let interval = null

    function update() {
      if (!props.startedAt) return
      elapsed.value = Math.floor(
        (Date.now() - new Date(props.startedAt + 'Z').getTime()) / 1000
      )
    }

    function format(sec) {
      const h = Math.floor(sec / 3600)
      const m = Math.floor((sec % 3600) / 60)
      const s = sec % 60
      if (h > 0) return `${h}:${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`
      return `${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`
    }

    onMounted(() => { update(); interval = setInterval(update, 1000) })
    onUnmounted(() => clearInterval(interval))

    return { elapsed, format }
  },

  template: `<span class="session-timer">{{ format(elapsed) }}</span>`
}
```

## `RestTimer.js` (stub pour T2.4 — remplacé en T2.5)

```javascript
import { ref, watch, onUnmounted } from '/vendor/vue.esm-browser.prod.js'

export default {
  name: 'RestTimer',
  props: {
    duration: { type: Number, default: 90 },  // secondes
    active:   { type: Boolean, default: false },
  },
  emits: ['complete', 'skip'],

  setup(props, { emit }) {
    const remaining = ref(props.duration)
    let interval = null

    function start() {
      if (interval) clearInterval(interval)
      remaining.value = props.duration
      interval = setInterval(() => {
        remaining.value--
        if (remaining.value <= 0) {
          clearInterval(interval)
          interval = null
          emit('complete')
        }
      }, 1000)
    }

    function stop() { if (interval) { clearInterval(interval); interval = null } }

    watch(() => props.active, (val) => { if (val) start(); else stop() })
    onUnmounted(stop)

    const pct = () => Math.max(0, remaining.value / props.duration)
    const circumference = 2 * Math.PI * 28  // rayon 28

    return { remaining, pct, circumference }
  },

  template: `
    <div v-if="active" class="rest-timer">
      <svg width="70" height="70" viewBox="0 0 70 70">
        <circle cx="35" cy="35" r="28" fill="none" stroke="var(--color-border)" stroke-width="4"/>
        <circle cx="35" cy="35" r="28" fill="none"
          :stroke="remaining > duration * 0.2 ? 'var(--color-success)' : 'var(--color-danger)'"
          stroke-width="4"
          :stroke-dasharray="circumference"
          :stroke-dashoffset="circumference * (1 - pct())"
          stroke-linecap="round"
          transform="rotate(-90 35 35)"
        />
        <text x="35" y="40" text-anchor="middle"
          fill="var(--color-text)" font-size="14" font-weight="600">
          {{ remaining }}s
        </text>
      </svg>
      <button class="btn btn--secondary" style="padding:0.3rem 0.75rem;font-size:0.8rem"
        @click="$emit('skip')">Passer</button>
    </div>
  `
}
```

## `SetRow.js`

```javascript
import { reactive, inject } from '/vendor/vue.esm-browser.prod.js'
import { sets as api } from '../../api.js'
import { t } from '../../i18n.js'

export default {
  name: 'SetRow',
  props: {
    set:              { type: Object, required: true },
    sessionExerciseId:{ type: Number, required: true },
    targetWeight:     { type: Number, default: 0 },
    targetReps:       { type: Number, default: 8 },
    setNumber:        { type: Number, required: true },
    isNew:            { type: Boolean, default: false }, // série non encore créée
    restSeconds:      { type: Number, default: 90 },
  },
  emits: ['validated', 'deleted', 'rest-start'],

  setup(props, { emit }) {
    const store = inject('store')
    const settings = inject('store')

    // Si la série existe déjà (is_new=false), afficher les valeurs existantes
    // Si nouvelle, pré-remplir avec target
    const local = reactive({
      weight_kg: props.isNew ? props.targetWeight : props.set.weight_kg,
      reps:      props.isNew ? props.targetReps   : props.set.reps,
      rpe:       props.set.rpe || '',
      saving:    false,
    })

    const showRpe = () => store.getSetting('show_rpe_field', false)
    const showRir = () => store.getSetting('show_rir_field', false)

    async function validate() {
      if (local.saving) return
      local.saving = true
      try {
        let result
        if (props.isNew) {
          // Créer la série
          result = await api.create(props.sessionExerciseId, {
            set_number:   props.setNumber,
            weight_kg:    parseFloat(local.weight_kg) || 0,
            reps:         parseInt(local.reps) || 0,
            rpe:          local.rpe ? parseFloat(local.rpe) : null,
            is_completed: true,
            is_warmup:    false,
          })
        } else {
          // Mettre à jour la série existante
          result = await api.update(props.set.id, {
            weight_kg:    parseFloat(local.weight_kg) || 0,
            reps:         parseInt(local.reps) || 0,
            rpe:          local.rpe ? parseFloat(local.rpe) : null,
            is_completed: true,
          })
        }

        // PR détecté
        if (result._new_pr) {
          store.showToast(
            t('session.pr_toast', {
              weight: result.weight_kg,
              reps:   result.reps,
              rm:     result._estimated_1rm,
            }),
            'success',
            5000
          )
        }

        emit('validated', result)
        emit('rest-start', props.restSeconds)
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        local.saving = false
      }
    }

    return { local, validate, showRpe, showRir, t }
  },

  template: `
    <div class="set-row" :class="{ 'set-row--done': set.is_completed && !isNew }">
      <span class="set-row__num">{{ t('session.set_number') }}{{ setNumber }}</span>

      <input
        class="set-row__input"
        type="number" step="0.5" min="0"
        v-model="local.weight_kg"
        :disabled="set.is_completed && !isNew"
      />
      <span class="set-row__unit">kg</span>

      <input
        class="set-row__input set-row__input--reps"
        type="number" min="0"
        v-model="local.reps"
        :disabled="set.is_completed && !isNew"
      />
      <span class="set-row__unit">r</span>

      <input v-if="showRpe() && (isNew || !set.is_completed)"
        class="set-row__input set-row__input--rpe"
        type="number" step="0.5" min="6" max="10"
        placeholder="RPE"
        v-model="local.rpe"
      />

      <button
        v-if="isNew || !set.is_completed"
        class="set-row__btn"
        :disabled="local.saving"
        @click="validate()">
        {{ local.saving ? '…' : '✓' }}
      </button>
      <span v-else class="set-row__done">✓</span>
    </div>
  `
}
```

## `ExerciseBlock.js`

```javascript
import { reactive, inject } from '/vendor/vue.esm-browser.prod.js'
import { exercises as exercisesApi } from '../../api.js'
import { t } from '../../i18n.js'
import SetRow       from './SetRow.js'
import RestTimer    from './RestTimer.js'
import ExerciseHelp from './ExerciseHelp.js'

export default {
  name: 'ExerciseBlock',
  props: {
    sessionExercise: { type: Object, required: true },
    // { id, exercise_id, name_fr, sets[], template_exercise_id? }
    targetWeight:    { type: Number, default: 0 },
    targetRepsMin:   { type: Number, default: 8 },
    targetRepsMax:   { type: Number, default: 12 },
    restSeconds:     { type: Number, default: 90 },
    exerciseDetail:  { type: Object, default: null }, // Exercise complet
    history:         { type: Array, default: () => [] },
  },
  emits: ['set-validated'],
  components: { SetRow, RestTimer, ExerciseHelp },

  setup(props, { emit }) {
    const store = inject('store')
    const state = reactive({
      sets:       [...(props.sessionExercise.sets || [])],
      restActive: false,
      restDuration: props.restSeconds,
      showHelp:   false,
      pendingNewSet: false,
    })

    function onValidated(set) {
      // Mettre à jour ou ajouter dans la liste locale
      const idx = state.sets.findIndex(s => s.id === set.id)
      if (idx >= 0) state.sets[idx] = { ...set, is_completed: 1 }
      else state.sets.push({ ...set, is_completed: 1 })
      state.pendingNewSet = false
      emit('set-validated', set)
    }

    function startRest(seconds) {
      state.restDuration = seconds
      state.restActive = true
    }

    function addSet() {
      state.pendingNewSet = true
    }

    const formula = () => store.getSetting('rm_formula', 'epley')
    const lastCompletedSet = () => {
      const done = state.sets.filter(s => s.is_completed)
      return done.length ? done[done.length - 1] : null
    }

    return { state, onValidated, startRest, addSet, formula, lastCompletedSet, t }
  },

  template: `
    <div class="exercise-block">
      <div class="exercise-block__header">
        <span class="exercise-block__name">{{ sessionExercise.name_fr }}</span>
        <button class="icon-btn" @click="state.showHelp = true" :title="t('session.help')">❓</button>
      </div>

      <div class="exercise-block__target" v-if="targetWeight > 0">
        {{ t('session.target') }} : {{ targetWeight }} kg ×
        {{ targetRepsMin }}<template v-if="targetRepsMax > targetRepsMin">–{{ targetRepsMax }}</template>
      </div>

      <!-- Séries existantes -->
      <SetRow
        v-for="(set, i) in state.sets" :key="set.id"
        :set="set"
        :session-exercise-id="sessionExercise.id"
        :target-weight="targetWeight"
        :target-reps="targetRepsMin"
        :set-number="i + 1"
        :is-new="false"
        :rest-seconds="restSeconds"
        @validated="onValidated"
        @rest-start="startRest"
      />

      <!-- Série en cours d'ajout -->
      <SetRow
        v-if="state.pendingNewSet"
        :set="{}"
        :session-exercise-id="sessionExercise.id"
        :target-weight="lastCompletedSet()?.weight_kg || targetWeight"
        :target-reps="lastCompletedSet()?.reps || targetRepsMin"
        :set-number="state.sets.length + 1"
        :is-new="true"
        :rest-seconds="restSeconds"
        @validated="onValidated"
        @rest-start="startRest"
      />

      <!-- Bouton ajouter série -->
      <button class="exercise-block__add-set" @click="addSet()" :disabled="state.pendingNewSet">
        + {{ t('session.add_set') }}
      </button>

      <!-- Timer de repos -->
      <div v-if="state.restActive" style="padding:0.75rem 1rem">
        <RestTimer
          :duration="state.restDuration"
          :active="state.restActive"
          @complete="state.restActive = false"
          @skip="state.restActive = false"
        />
      </div>

      <!-- Modal aide -->
      <ExerciseHelp
        v-if="state.showHelp && exerciseDetail"
        :exercise="exerciseDetail"
        :last-weight="lastCompletedSet()?.weight_kg || targetWeight"
        :last-reps="lastCompletedSet()?.reps || targetRepsMin"
        :formula="formula()"
        :history="history"
        @close="state.showHelp = false"
      />
    </div>
  `
}
```

## `SessionActiveView.js`

```javascript
import { reactive, onMounted, inject } from '/vendor/vue.esm-browser.prod.js'
import {
  sessions as sessionsApi,
  exercises as exercisesApi
} from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader    from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import EmptyState    from '../common/EmptyState.js'
import ErrorBanner   from '../common/ErrorBanner.js'
import ConfirmModal  from '../common/ConfirmModal.js'
import ExerciseBlock from './ExerciseBlock.js'
import SessionTimer  from './SessionTimer.js'

export default {
  name: 'SessionActiveView',
  components: {
    PageHeader, LoadingSpinner, EmptyState, ErrorBanner,
    ConfirmModal, ExerciseBlock, SessionTimer
  },

  setup() {
    const router = inject('router')
    const store  = inject('store')

    const state = reactive({
      loading: true, error: null,
      session: null,
      exercises: [],    // [{ sessionExercise, exerciseDetail, history, target }]
      showCompleteConfirm: false,
      completing: false,
    })

    async function load() {
      state.loading = true; state.error = null
      try {
        // Chercher une séance in_progress
        const data = await sessionsApi.list({ status: 'in_progress', limit: 1 })
        if (!data.sessions.length) {
          state.session = null; state.loading = false; return
        }
        const session = await sessionsApi.get(data.sessions[0].id)
        state.session = session
        store.setActiveSession(session)

        // Pour chaque exercise, charger le détail + l'historique + la cible
        const enriched = []
        for (const se of session.exercises) {
          const [detail, history] = await Promise.all([
            exercisesApi.get(se.exercise_id),
            exercisesApi.history(se.exercise_id, { limit: 5 }),
          ])
          let target = { weight_kg: 0, reps_min: 8, reps_max: 12, explanation: '' }
          if (se.template_exercise_id) {
            try {
              target = await sessionsApi.target(session.id, se.template_exercise_id)
            } catch (_) { /* pas de cible si non rattaché à un template */ }
          }
          enriched.push({
            sessionExercise: se,
            exerciseDetail:  detail,
            history:         history.history,
            target,
          })
        }
        state.exercises = enriched
      } catch (err) {
        state.error = err.message
      } finally {
        state.loading = false
      }
    }

    async function completeSession() {
      if (!state.session) return
      state.completing = true
      try {
        await sessionsApi.complete(state.session.id)
        store.clearActiveSession()
        store.showToast('Séance terminée 💪', 'success')
        router.push('#/session/history')
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        state.completing = false
        state.showCompleteConfirm = false
      }
    }

    onMounted(load)

    return { state, load, completeSession, router, t }
  },

  template: `
    <div class="view" style="padding-bottom:0">
      <!-- Header avec chronomètre -->
      <header class="page-header">
        <h1 class="page-header__title">
          {{ state.session?.name || t('session.title') }}
        </h1>
        <SessionTimer
          v-if="state.session?.started_at"
          :started-at="state.session.started_at"
        />
        <button
          v-if="state.session"
          class="btn btn--primary"
          style="padding:0.35rem 0.75rem;font-size:0.8rem"
          @click="state.showCompleteConfirm = true">
          {{ t('session.complete') }}
        </button>
      </header>

      <div class="view__content">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner   v-else-if="state.error" :message="state.error" :retry="load" />

        <!-- Aucune séance active -->
        <EmptyState
          v-else-if="!state.session"
          icon="🏋️"
          :message="t('session.no_active')">
          <button class="btn btn--primary" style="margin-top:1rem"
            @click="router.push('#/programs')">
            {{ t('session.start_button') }}
          </button>
        </EmptyState>

        <!-- Séance active -->
        <template v-else>
          <ExerciseBlock
            v-for="item in state.exercises"
            :key="item.sessionExercise.id"
            :session-exercise="item.sessionExercise"
            :target-weight="item.target.weight_kg"
            :target-reps-min="item.target.reps_min"
            :target-reps-max="item.target.reps_max"
            :rest-seconds="item.sessionExercise.rest_seconds || 90"
            :exercise-detail="item.exerciseDetail"
            :history="item.history"
            @set-validated="() => {}"
          />
        </template>
      </div>

      <!-- Confirmation terminer -->
      <ConfirmModal
        v-if="state.showCompleteConfirm"
        :title="t('session.complete_confirm')"
        :confirm-text="t('session.complete')"
        @confirm="completeSession()"
        @cancel="state.showCompleteConfirm = false"
      />
    </div>
  `
}
```
