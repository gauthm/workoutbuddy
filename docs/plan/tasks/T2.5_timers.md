## Pourquoi un Web Worker ?

`setInterval` dans un onglet en arrière-plan est throttlé par les navigateurs
(Chrome, Safari) à 1 tick/seconde minimum, mais parfois beaucoup moins souvent.
Un Web Worker n'est pas soumis à cette restriction et continue de tourner
normalement.

## `public/js/timer-worker.js` — implémentation complète

```javascript
// Web Worker — pas d'accès au DOM, pas d'import ES module (syntaxe classic)
// Communique via postMessage / onmessage

let restInterval    = null
let sessionInterval = null
let restRemaining   = 0
let sessionElapsed  = 0

self.onmessage = function(e) {
  const { type, payload } = e.data

  // ── Repos ────────────────────────────────────────────────
  if (type === 'START_REST') {
    if (restInterval) clearInterval(restInterval)
    restRemaining = payload.duration_seconds
    self.postMessage({ type: 'REST_TICK', remaining: restRemaining })
    restInterval = setInterval(() => {
      restRemaining--
      self.postMessage({ type: 'REST_TICK', remaining: restRemaining })
      if (restRemaining <= 0) {
        clearInterval(restInterval)
        restInterval = null
        self.postMessage({ type: 'REST_COMPLETE' })
      }
    }, 1000)
  }

  if (type === 'STOP_REST') {
    if (restInterval) { clearInterval(restInterval); restInterval = null }
    self.postMessage({ type: 'REST_STOPPED' })
  }

  if (type === 'SKIP_REST') {
    if (restInterval) { clearInterval(restInterval); restInterval = null }
    restRemaining = 0
    self.postMessage({ type: 'REST_COMPLETE' })
  }

  // ── Séance ───────────────────────────────────────────────
  if (type === 'START_SESSION') {
    if (sessionInterval) clearInterval(sessionInterval)
    sessionElapsed = payload.elapsed_seconds || 0
    self.postMessage({ type: 'SESSION_TICK', elapsed: sessionElapsed })
    sessionInterval = setInterval(() => {
      sessionElapsed++
      self.postMessage({ type: 'SESSION_TICK', elapsed: sessionElapsed })
    }, 1000)
  }

  if (type === 'STOP_SESSION') {
    if (sessionInterval) { clearInterval(sessionInterval); sessionInterval = null }
  }

  if (type === 'PING') {
    // Test de connectivité
    self.postMessage({ type: 'PONG' })
  }
}
```

export function getTimerWorker() {

## `public/js/components/session/SessionTimer.js` — version complète

```javascript
import { ref, onMounted, onUnmounted } from '/vendor/vue.esm-browser.prod.js'
import { getTimerWorker } from '../../timer-worker-singleton.js'

export default {
  name: 'SessionTimer',
  props: {
    startedAt: { type: String, required: true }
    // Format : "2025-07-10T14:30:00" (sans Z — UTC de SQLite)
  },

  setup(props) {
    const elapsed = ref(0)
    const worker = getTimerWorker()

    function onMessage(e) {
      if (e.data.type === 'SESSION_TICK') {
        elapsed.value = e.data.elapsed
      }
    }

    function format(sec) {
      const h = Math.floor(sec / 3600)
      const m = Math.floor((sec % 3600) / 60)
      const s = sec % 60
      if (h > 0) {
        return `${h}:${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`
      }
      return `${String(m).padStart(2,'0')}:${String(s).padStart(2,'0')}`
    }

    onMounted(() => {
      worker.addEventListener('message', onMessage)
      // Calculer le nombre de secondes écoulées depuis le démarrage
      const elapsedSec = Math.max(
        0,
        Math.floor((Date.now() - new Date(props.startedAt + 'Z').getTime()) / 1000)
      )
      worker.postMessage({ type: 'START_SESSION', payload: { elapsed_seconds: elapsedSec } })
    })

    onUnmounted(() => {
      worker.removeEventListener('message', onMessage)
      // NE PAS stopper le worker ici — il continue en arrière-plan
      // Il sera stoppé par RestTimer ou à la fin de la séance
    })

    return { elapsed, format }
  },

  template: `
    <span class="session-timer">{{ format(elapsed) }}</span>
  `
}
```

## `public/js/components/session/RestTimer.js` — version complète

```javascript
import { ref, onMounted, onUnmounted, watch } from '/vendor/vue.esm-browser.prod.js'
import { getTimerWorker } from '../../timer-worker-singleton.js'

// Son de fin de repos — initialisé une seule fois
let audioCtx = null
function playBeep() {
  try {
    // Essayer d'abord le fichier MP3
    const audio = new Audio('/sounds/timer-end.mp3')
    audio.play().catch(() => {
      // Fallback : bip synthétique via Web Audio API
      if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)()
      const osc = audioCtx.createOscillator()
      const gain = audioCtx.createGain()
      osc.connect(gain)
      gain.connect(audioCtx.destination)
      osc.frequency.value = 880
      gain.gain.setValueAtTime(0.3, audioCtx.currentTime)
      gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.5)
      osc.start(audioCtx.currentTime)
      osc.stop(audioCtx.currentTime + 0.5)
    })
  } catch (_) { /* silencieux si tout échoue */ }
}

function vibrate() {
  try {
    if ('vibrate' in navigator) {
      navigator.vibrate([200, 100, 200, 100, 200])
    }
  } catch (_) { /* silencieux */ }
}

export default {
  name: 'RestTimer',
  props: {
    duration:      { type: Number, required: true },  // secondes
    active:        { type: Boolean, default: false },
    soundEnabled:  { type: Boolean, default: true },
    vibrEnabled:   { type: Boolean, default: true },
  },
  emits: ['complete', 'skip'],

  setup(props, { emit }) {
    const remaining   = ref(props.duration)
    const worker = getTimerWorker()
    let wakeLock = null

    async function acquireWakeLock() {
      try {
        if ('wakeLock' in navigator) {
          wakeLock = await navigator.wakeLock.request('screen')
        }
      } catch (_) { /* Wake Lock non supporté, silencieux */ }
    }

    function releaseWakeLock() {
      if (wakeLock) { wakeLock.release().catch(() => {}); wakeLock = null }
    }

    function onMessage(e) {
      if (e.data.type === 'REST_TICK') {
        remaining.value = e.data.remaining
      }
      if (e.data.type === 'REST_COMPLETE') {
        remaining.value = 0
        releaseWakeLock()
        if (props.soundEnabled) playBeep()
        if (props.vibrEnabled) vibrate()
        emit('complete')
      }
      if (e.data.type === 'REST_STOPPED') {
        releaseWakeLock()
      }
    }

    watch(() => props.active, async (val) => {
      if (val) {
        remaining.value = props.duration
        worker.postMessage({ type: 'START_REST', payload: { duration_seconds: props.duration } })
        await acquireWakeLock()
      } else {
        worker.postMessage({ type: 'STOP_REST' })
        releaseWakeLock()
      }
    }, { immediate: false })

    onMounted(() => {
      worker.addEventListener('message', onMessage)
    })

    onUnmounted(() => {
      worker.removeEventListener('message', onMessage)
      releaseWakeLock()
    })

    function skip() {
      worker.postMessage({ type: 'SKIP_REST' })
      emit('skip')
    }

    // Cercle SVG
    const RADIUS = 28
    const circumference = 2 * Math.PI * RADIUS
    const dashOffset = () => {
      const pct = Math.max(0, remaining.value / props.duration)
      return circumference * (1 - pct)
    }
    const strokeColor = () => {
      const pct = remaining.value / props.duration
      if (pct > 0.5) return 'var(--color-success)'
      if (pct > 0.2) return 'var(--color-warning)'
      return 'var(--color-danger)'
    }

    return { remaining, skip, circumference, RADIUS, dashOffset, strokeColor }
  },

  template: `
    <div v-if="active" class="rest-timer">
      <svg width="80" height="80" viewBox="0 0 80 80">
        <!-- Fond -->
        <circle cx="40" cy="40" :r="RADIUS"
          fill="none" stroke="var(--color-border)" stroke-width="5"/>
        <!-- Progression -->
        <circle cx="40" cy="40" :r="RADIUS"
          fill="none"
          :stroke="strokeColor()"
          stroke-width="5"
          :stroke-dasharray="circumference"
          :stroke-dashoffset="dashOffset()"
          stroke-linecap="round"
          transform="rotate(-90 40 40)"
          style="transition: stroke-dashoffset 0.9s linear, stroke 0.3s"
        />
        <!-- Texte -->
        <text x="40" y="45" text-anchor="middle"
          fill="var(--color-text)" font-size="16" font-weight="700">
          {{ remaining }}s
        </text>
      </svg>
      <button class="btn btn--secondary"
        style="padding:0.3rem 0.9rem;font-size:0.8rem;margin-top:0.3rem"
        @click="skip()">
        Passer
      </button>
    </div>
  `
}
```

## Son `public/sounds/timer-end.mp3`

**Option A — Télécharger un son libre de droits :**

```bash
# Depuis Freesound.org (compte gratuit requis) :
# https://freesound.org/people/zapsplat/sounds/  (CC0)
# Chercher "timer beep" ou "notification"
# Télécharger en MP3, renommer en timer-end.mp3, < 50 KB
```

**Option B — Générer avec ffmpeg (si disponible) :**
```bash
ffmpeg -f lavfi -i "sine=frequency=880:duration=0.3" \
  -af "afade=t=out:st=0.2:d=0.1" \
  public/sounds/timer-end.mp3
```

**Option C — Fichier vide (le fallback Web Audio API prend le relais) :**
```bash
# Créer un MP3 minimal valide (1 octet) :
printf '\xff\xfb\x90\x00' > public/sounds/timer-end.mp3
# Le navigateur échouera à le lire → fallback bip synthétique via AudioContext
```

**Recommandation** : utiliser l'Option B si ffmpeg est installé.
Sinon Option C pour débloquer le développement — le bip synthétique
fonctionne correctement en fallback.

## Notes sur les limitations plateformes

| Plateforme | Web Worker | Wake Lock | Son en arrière-plan |
|---|---|---|---|
| Chrome Android | ✅ | ✅ | ✅ (si audio démarré par geste) |
| Chrome macOS | ✅ | ✅ | ✅ |
| Safari iOS 16+ | ✅ | ⚠️ Partiel | ❌ (bloqué en arrière-plan strict) |
| Firefox | ✅ | ❌ | ✅ |

**Gestion Safari iOS** : le son ne se déclenchera pas si l'onglet passe
en arrière-plan strict (app en background). La vibration fonctionne.
Afficher un texte "⚠️ Son désactivé sur iOS en arrière-plan" dans les
paramètres timers. Ne pas considérer cela comme un bug.

**Wake Lock et visibility** : le Wake Lock est automatiquement libéré
si l'utilisateur verrouille l'écran. Ré-acquérir sur `visibilitychange` :

```javascript
// Ajouter dans RestTimer.js dans onMounted() :
document.addEventListener('visibilitychange', async () => {
  if (document.visibilityState === 'visible' && props.active) {
    await acquireWakeLock()
  }
})
```

```javascript

```

**Points d'attention :**
- Le Worker est créé avec `new Worker('/js/timer-worker.js')` — chemin
  absolu depuis la racine `public/`. Le fichier doit exister à cet endroit.
- `timer-worker.js` utilise la syntaxe **classic** (pas `type: 'module'`) —
  c'est la forme par défaut et la plus compatible.
- Le singleton `getTimerWorker()` retourne toujours le même worker.
  `SessionTimer` et `RestTimer` partagent le même worker et filtrent
  les messages par type — c'est intentionnel.
- Ne pas appeler `worker.terminate()` dans `onUnmounted()` —
  le worker doit survivre au démontage des composants pour continuer
  à tourner en arrière-plan.
- L'`AudioContext` doit être créé APRÈS un geste utilisateur (tap, click).
  Dans notre cas, le timer est démarré par un bouton → geste → OK.

```javascript
// Note : les Web Workers ne sont pas disponibles dans node:test.
// Ce test simule la logique du worker manuellement.

import { describe, it } from 'node:test'
import assert from 'node:assert/strict'

// Logique extraite du worker pour test unitaire
function createWorkerLogic() {
  let restRemaining = 0
  const messages = []
  const logic = {
    postMessage(msg) { messages.pop(); messages.push(msg) },
    getMessages() { return [...messages] },
    handle(e) {
      const { type, payload } = e.data
      if (type === 'START_REST') {
        restRemaining = payload.duration_seconds
        logic.postMessage({ type: 'REST_TICK', remaining: restRemaining })
      }
      if (type === 'SKIP_REST') {
        restRemaining = 0
        logic.postMessage({ type: 'REST_COMPLETE' })
      }
    }
  }
  return logic
}

describe('T2.5 — Timer Worker Logic', () => {
  it('START_REST émet REST_TICK avec la durée initiale', () => {
    const logic = createWorkerLogic()
    logic.handle({ data: { type: 'START_REST', payload: { duration_seconds: 90 } } })
    const msgs = logic.getMessages()
    assert.equal(msgs[0].type, 'REST_TICK')
    assert.equal(msgs[0].remaining, 90)
  })

  it('SKIP_REST émet REST_COMPLETE', () => {
    const logic = createWorkerLogic()
    logic.handle({ data: { type: 'START_REST', payload: { duration_seconds: 90 } } })
    logic.handle({ data: { type: 'SKIP_REST' } })
    const msgs = logic.getMessages()
    assert.equal(msgs[0].type, 'REST_COMPLETE')
  })
})
```
