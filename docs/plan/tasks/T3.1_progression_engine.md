```javascript
// Formules de RM estimé — source de vérité côté serveur
// Le client (utils/rm.js) en est un miroir pour l'affichage uniquement.

export const FORMULAS = {
  epley:    (w, r) => r === 1 ? w : w * (1 + r / 30),
  brzycki:  (w, r) => r >= 37 ? w : w * 36 / (37 - r),
  lander:   (w, r) => 100 * w / (101.3 - 2.67123 * r),
  oconner:  (w, r) => w * (1 + 0.025 * r),
  lombardi: (w, r) => w * Math.pow(r, 0.10),
}

/**
 * Estime le 1RM depuis un poids et un nombre de répétitions.
 * @param {number} weightKg  - poids soulevé
 * @param {number} reps      - répétitions effectuées
 * @param {string} formula   - clé de FORMULAS (défaut: 'epley')
 * @returns {number} 1RM estimé, arrondi à 2 décimales
 */
export function estimate1RM(weightKg, reps, formula = 'epley') {
  if (!weightKg || weightKg <= 0) return 0
  if (!reps || reps <= 0) return 0
  if (reps === 1) return Math.round(weightKg * 100) / 100
  const fn = FORMULAS[formula] || FORMULAS.epley
  return Math.round(fn(weightKg, reps) * 100) / 100
}

/**
 * Génère un tableau RM pour N=1..15 répétitions.
 * Inversion de la formule Epley pour tous les modèles
 * (approximation conservative, suffisante pour l'affichage).
 * @param {number} estimated1rm - 1RM de référence
 * @param {string} formula      - utilisé pour le libellé uniquement
 * @returns {{ reps: number, weight_kg: number }[]}
 */
export function rmTable(estimated1rm, formula = 'epley') {
  if (!estimated1rm || estimated1rm <= 0) return []
  const result = []
  for (let n = 1; n <= 15; n++) {
    const w = n === 1
      ? estimated1rm
      : estimated1rm / (1 + n / 30)  // inversion Epley
    result.push({ reps: n, weight_kg: Math.round(w * 10) / 10 })
  }
  return result
}
```

## `src/services/progressionEngine.js` — implémentation complète

```javascript
import { calculateAchievableLoads } from './equipmentCalc.js'

// ── Helper : arrondir vers le bas à la charge atteignable la plus proche ──
export function roundDownToAchievable(targetWeight, achievableLoads) {
  if (!achievableLoads || achievableLoads.length === 0) return 0
  const valid = achievableLoads.filter(l => l <= targetWeight + 0.001)
  if (valid.length === 0) return achievableLoads[0]
  return valid[valid.length - 1]
}

// ── Résoudre le modèle effectif ──────────────────────────────────────────
export function resolveProgressionModel(templateExercise, program) {
  if (templateExercise.progression_model !== 'inherit') {
    return templateExercise.progression_model
  }
  return program?.progression_model || 'double_progression'
}

/**
 * Calcule le prochain état de progression après une séance.
 *
 * @param {object} state            - ProgressionState courant depuis la DB
 * @param {object[]} lastSessionSets - SetLog[] de la dernière séance
 *                                    (is_warmup=0, is_completed=1 uniquement)
 * @param {object} templateExercise  - TemplateExercise
 * @param {object} program           - Program (pour progression_model hérité)
 * @param {number[]} achievableLoads - Charges atteignables triées ASC
 * @param {object} settings          - { progression_failure_threshold,
 *                                       progression_weight_reduction_pct }
 * @returns {{ current_weight_kg, current_target_reps?, consecutive_failures, explanation }}
 */
export function computeNextState(
  state,
  lastSessionSets,
  templateExercise,
  program,
  achievableLoads,
  settings
) {
  const model = resolveProgressionModel(templateExercise, program)

  switch (model) {
    case 'linear':
      return linearProgression(state, lastSessionSets, templateExercise, achievableLoads, settings)
    case 'double_progression':
      return doubleProgression(state, lastSessionSets, templateExercise, achievableLoads, settings)
    case 'rpe_based':
      return rpeProgression(state, lastSessionSets, templateExercise, achievableLoads, settings)
    default:
      return doubleProgression(state, lastSessionSets, templateExercise, achievableLoads, settings)
  }
}

// ── Progression linéaire ─────────────────────────────────────────────────
function linearProgression(state, sets, te, achievable, settings) {
  const threshold   = settings.progression_failure_threshold  || 2
  const reductionPct = settings.progression_weight_reduction_pct || 10

  // Cible de reps : target_reps si défini, sinon target_reps_min
  const targetReps = te.target_reps || te.target_reps_min

  // Succès = toutes les séries prévues effectuées avec au moins targetReps reps
  const allOk = sets.length >= te.target_sets &&
    sets.every(s => s.reps >= targetReps)

  if (allOk) {
    // Déterminer l'incrément
    const increment = te.progression_weight_increment_kg
      || minIncrement(achievable)
      || 2.5

    const newW = roundDownToAchievable(state.current_weight_kg + increment, achievable)
    return {
      current_weight_kg:   newW,
      consecutive_failures: 0,
      explanation: `Objectif atteint (${sets.length}×${targetReps} reps). +${increment} kg → ${newW} kg.`,
    }
  }

  // Échec
  const fails = (state.consecutive_failures || 0) + 1
  if (fails >= threshold) {
    const newW = roundDownToAchievable(
      state.current_weight_kg * (1 - reductionPct / 100),
      achievable
    )
    return {
      current_weight_kg:   newW,
      consecutive_failures: 0,
      explanation: `${fails} échecs consécutifs. -${reductionPct}% → ${newW} kg.`,
    }
  }

  return {
    current_weight_kg:   state.current_weight_kg,
    consecutive_failures: fails,
    explanation: `Objectif non atteint (${fails}/${threshold}). Maintien ${state.current_weight_kg} kg.`,
  }
}

// ── Double progression ───────────────────────────────────────────────────
function doubleProgression(state, sets, te, achievable, settings) {
  const threshold    = settings.progression_failure_threshold  || 2
  const reductionPct = settings.progression_weight_reduction_pct || 10
  const minR = te.target_reps_min
  const maxR = te.target_reps_max
  const curTargetReps = state.current_target_reps || minR

  const completedSets = sets.length >= te.target_sets
  const minRepsDone   = completedSets
    ? Math.min(...sets.map(s => s.reps))
    : 0

  // Toutes les séries ≥ max_reps → augmenter le poids
  if (completedSets && minRepsDone >= maxR) {
    const increment = te.progression_weight_increment_kg
      || minIncrement(achievable)
      || 2.5
    const newW = roundDownToAchievable(state.current_weight_kg + increment, achievable)
    return {
      current_weight_kg:    newW,
      current_target_reps:  minR,
      consecutive_failures: 0,
      explanation: `Toutes séries ≥ ${maxR} reps. +${increment} kg → ${newW} kg. Retour à ${minR} reps.`,
    }
  }

  // Toutes les séries ≥ min_reps → augmenter les reps cibles
  if (completedSets && minRepsDone >= minR) {
    const newReps = Math.min(curTargetReps + 1, maxR)
    return {
      current_weight_kg:    state.current_weight_kg,
      current_target_reps:  newReps,
      consecutive_failures: 0,
      explanation: `Objectif atteint. Prochaine cible : ${newReps} reps à ${state.current_weight_kg} kg.`,
    }
  }

  // Échec
  const fails = (state.consecutive_failures || 0) + 1
  if (fails >= threshold) {
    const newW = roundDownToAchievable(
      state.current_weight_kg * (1 - reductionPct / 100),
      achievable
    )
    return {
      current_weight_kg:    newW,
      current_target_reps:  minR,
      consecutive_failures: 0,
      explanation: `${fails} échecs. Réduction à ${newW} kg, retour à ${minR} reps.`,
    }
  }

  return {
    current_weight_kg:    state.current_weight_kg,
    current_target_reps:  curTargetReps,
    consecutive_failures: fails,
    explanation: `Objectif non atteint (${fails}/${threshold}). Maintien ${state.current_weight_kg} kg.`,
  }
}

// ── Progression RPE ──────────────────────────────────────────────────────
function rpeProgression(state, sets, te, achievable, settings) {
  // Prendre le RPE de la dernière série complétée
  const completedSets = sets.filter(s => s.rpe !== null && s.rpe !== undefined)
  if (completedSets.length === 0) {
    return {
      current_weight_kg:    state.current_weight_kg,
      consecutive_failures: state.consecutive_failures,
      explanation: 'RPE non renseigné. Pas de changement.',
    }
  }

  const lastRpe    = completedSets[completedSets.length - 1].rpe
  const targetRpe  = te.rpe_target || 8.0
  const diff       = lastRpe - targetRpe

  // RPE très inférieur à la cible → augmenter le poids
  if (diff <= -1.0) {
    const pct  = 0.025 * Math.abs(diff)
    const newW = roundDownToAchievable(state.current_weight_kg * (1 + pct), achievable)
    return {
      current_weight_kg:    newW,
      consecutive_failures: 0,
      explanation: `RPE ${lastRpe} < cible ${targetRpe}. +${(pct * 100).toFixed(1)}% → ${newW} kg.`,
    }
  }

  // RPE très supérieur → réduire le poids
  if (diff >= 1.0) {
    const pct  = 0.025 * Math.abs(diff)
    const newW = roundDownToAchievable(state.current_weight_kg * (1 - pct), achievable)
    return {
      current_weight_kg:    newW,
      consecutive_failures: 0,
      explanation: `RPE ${lastRpe} > cible ${targetRpe}. -${(pct * 100).toFixed(1)}% → ${newW} kg.`,
    }
  }

  // RPE dans la cible ±1
  return {
    current_weight_kg:    state.current_weight_kg,
    consecutive_failures: 0,
    explanation: `RPE ${lastRpe} ≈ cible ${targetRpe}. Maintien ${state.current_weight_kg} kg.`,
  }
}

// ── Helper : incrément minimal d'une liste de charges ────────────────────
function minIncrement(loads) {
  if (!loads || loads.length <= 1) return 0
  let min = Infinity
  for (let i = 1; i < loads.length; i++) {
    const d = loads[i] - loads[i - 1]
    if (d > 0.001) min = Math.min(min, d)
  }
  return min === Infinity ? 0 : Math.round(min * 100) / 100
}
```

## Exemples chiffrés de vérification

### Linéaire
```
État initial : current_weight_kg=80, consecutive_failures=0
te : target_sets=4, target_reps=5, progression_weight_increment_kg=2.5
sets : [{reps:5},{reps:5},{reps:5},{reps:5}]  (succès)
achievable : [77.5, 80.0, 82.5, 85.0, ...]
→ 80 + 2.5 = 82.5 → roundDown(82.5, achievable) = 82.5
Résultat : current_weight_kg=82.5, consecutive_failures=0
```

### Double progression
```
État initial : current_weight_kg=60, current_target_reps=8
te : target_sets=3, target_reps_min=8, target_reps_max=12
sets : [{reps:12},{reps:12},{reps:12}]  (toutes ≥ max)
achievable : [57.5, 60.0, 62.5, ...]
→ +2.5 kg → 62.5, retour à 8 reps
Résultat : current_weight_kg=62.5, current_target_reps=8
```

### Double progression — progression de reps
```
État initial : current_weight_kg=60, current_target_reps=9
sets : [{reps:10},{reps:10},{reps:10}]  (toutes ≥ min=8, < max=12)
→ new_reps = min(9+1, 12) = 10, poids inchangé
Résultat : current_weight_kg=60, current_target_reps=10
```

### Échec linéaire (seuil=2, réduction=10%)
```
État : current_weight_kg=100, consecutive_failures=1
sets : [{reps:3}]  (< target 5) → fails=2 ≥ threshold=2
→ 100 * 0.90 = 90 → roundDown(90, achievable)
Résultat : current_weight_kg=90, consecutive_failures=0
```

```javascript
    target_sets: 3, target_reps_min: 8, target_reps_max: 12,
    progression_weight_increment_kg: 2.5,

  const state = { current_weight_kg: 60, current_target_reps: 8, consecutive_failures: 0 }

  it('toutes séries ≥ max_reps → +poids, retour min_reps', () => {
    const sets = Array(3).fill({ reps: 12, rpe: null })
    const r = computeNextState(state, sets, te, { progression_model: 'double_progression' }, LOADS, SETTINGS)
    assert.equal(r.current_weight_kg, 62.5)
    assert.equal(r.current_target_reps, 8)
  })

  it('toutes séries dans la fourchette → +1 rep', () => {
    const sets = Array(3).fill({ reps: 10, rpe: null })
    const r = computeNextState(state, sets, te, { progression_model: 'double_progression' }, LOADS, SETTINGS)
    assert.equal(r.current_weight_kg, 60)
    assert.equal(r.current_target_reps, 9)
  })

  it('current_target_reps = max-1, reps atteints → passe à max', () => {
    const stateNear = { ...state, current_target_reps: 11 }
    const sets = Array(3).fill({ reps: 11, rpe: null })
    const r = computeNextState(stateNear, sets, te, { progression_model: 'double_progression' }, LOADS, SETTINGS)
    assert.equal(r.current_target_reps, 12)
  })

  it('échec threshold → réduction poids', () => {
    const stateF = { current_weight_kg: 60, current_target_reps: 8, consecutive_failures: 1 }
    const sets = [{ reps: 5, rpe: null }]  // < min_reps
    const r = computeNextState(stateF, sets, te, { progression_model: 'double_progression' }, LOADS, SETTINGS)
    // 60 * 0.9 = 54 → roundDown(54, LOADS) = 50
    assert.equal(r.current_weight_kg, 50)
    assert.equal(r.current_target_reps, 8)
    assert.equal(r.consecutive_failures, 0)
  })
})

describe('rpe_based', () => {

    const sets = [{ reps: 8, rpe: 10.0 }]  // diff = +2 → pct = 5%
    const r = computeNextState(state, sets, te, { progression_model: 'rpe_based' }, LOADS, SETTINGS)

describe('resolveProgressionModel', () => {
```
