/**
 * Détecte un plateau de progression sur un exercice.
 *
 * Algorithme : comparer le RM estimé (Epley) de la session la plus récente
 * au RM de la session la plus ancienne dans la fenêtre de N sessions.
 * Si l'amélioration est ≤ 1%, c'est un plateau.
 *
 * @param {string}  exerciseId  - ID de l'exercice
 * @param {number}  programId   - ID du programme
 * @param {number}  n           - nombre de sessions à analyser (défaut 4)
 * @param {object}  db          - instance better-sqlite3
 * @returns {{ is_plateau, sessions_checked, first_1rm?, current_1rm?, delta_pct?, reason }}
 */
export function detectPlateau(exerciseId, programId, n = 4, db) {
  const rows = db.prepare(`
    SELECT
      s.id      AS session_id,
      s.completed_at,
      MAX(sl.weight_kg * (1.0 + sl.reps / 30.0)) AS session_1rm
    FROM set_log sl
    JOIN session_exercise se ON sl.session_exercise_id = se.id
    JOIN session s            ON se.session_id = s.id
    WHERE se.exercise_id  = ?
      AND s.program_id    = ?
      AND s.status        = 'completed'
      AND sl.is_warmup    = 0
      AND sl.is_completed = 1
      AND sl.reps         > 0
      AND sl.weight_kg    > 0
    GROUP BY s.id
    ORDER BY s.completed_at DESC
    LIMIT ?
  `).all(exerciseId, programId, n)

  if (rows.length < n) {
    return {
      is_plateau: false,
      sessions_checked: rows.length,
      reason: `Pas assez de données (${rows.length}/${n} séances).`,
    }
  }

  // rows[0] = plus récent, rows[n-1] = plus ancien
  const currentRm = rows[0].session_1rm
  const firstRm   = rows[n - 1].session_1rm

  if (firstRm <= 0) {
    return { is_plateau: false, sessions_checked: n, reason: 'RM de référence nul.' }
  }

  const deltaPct = ((currentRm - firstRm) / firstRm) * 100

  if (deltaPct <= 1.0) {  // tolérance 1% pour les arrondis de charge
    return {
      is_plateau:       true,
      sessions_checked: n,
      first_1rm:        Math.round(firstRm * 100) / 100,
      current_1rm:      Math.round(currentRm * 100) / 100,
      delta_pct:        Math.round(deltaPct * 10) / 10,
    }
  }

  return {
    is_plateau:       false,
    sessions_checked: n,
    first_1rm:        Math.round(firstRm * 100) / 100,
    current_1rm:      Math.round(currentRm * 100) / 100,
    delta_pct:        Math.round(deltaPct * 10) / 10,
  }
}

/**
 * Vérifie si au moins `minFraction` des exercices actifs d'un programme
 * sont en plateau.
 *
 * @param {number}  programId    - ID du programme
 * @param {number}  n            - fenêtre de sessions
 * @param {number}  minFraction  - fraction minimale (ex: 0.5 = 50%)
 * @param {object}  db
 * @returns {{ is_program_plateau, plateau_count, total_exercises }}
 */
export function detectProgramPlateau(programId, n, minFraction = 0.5, db) {
  const templateExercises = db.prepare(`
    SELECT te.id, te.exercise_id
    FROM template_exercise te
    JOIN workout_template wt ON te.workout_template_id = wt.id
    WHERE wt.program_id = ?
  `).all(programId)

  if (templateExercises.length === 0) {
    return { is_program_plateau: false, plateau_count: 0, total_exercises: 0 }
  }

  let plateauCount = 0
  for (const te of templateExercises) {
    const result = detectPlateau(te.exercise_id, programId, n, db)
    if (result.is_plateau) plateauCount++
  }

  return {
    is_program_plateau: plateauCount >= Math.ceil(templateExercises.length * minFraction),
    plateau_count:      plateauCount,
    total_exercises:    templateExercises.length,
  }
}

## `src/services/deloadEngine.js` — implémentation complète

```javascript
import { detectProgramPlateau } from './plateauDetector.js'

/**
 * Détermine si un programme doit passer en deload.
 *
 * Deux déclencheurs :
 * 1. Deload planifié : N semaines écoulées depuis le dernier deload
 * 2. Plateau : ≥ 50% des exercices en plateau sur `plateau_sessions` séances
 *
 * @param {object}  program   - ligne de la table `program`
 * @param {object}  settings  - settings parsés (objet clé→valeur)
 * @param {object}  db
 * @returns {{ should: bool, reason?: 'scheduled'|'plateau', detail?: string }}
 */
export function shouldDeload(program, settings, db) {
  if (program.is_in_deload) {
    return { should: false }
  }

  // ── Déclencheur 1 : deload planifié ──────────────────────────────
  if (program.deload_auto_enabled) {
    const freqWeeks = program.deload_frequency_weeks || 6
    const refDate   = program.last_deload_date || program.created_at
    const weeksSince = weeksBetween(refDate, new Date().toISOString())

    if (weeksSince >= freqWeeks) {
      return {
        should: true,
        reason: 'scheduled',
        detail: `${weeksSince} semaines depuis le dernier deload (fréquence : ${freqWeeks}).`,
      }
    }
  }

  // ── Déclencheur 2 : plateau persistant ───────────────────────────
  const plateauSessions = settings.plateau_detection_sessions || 4
  const result = detectProgramPlateau(program.id, plateauSessions, 0.5, db)

  if (result.is_program_plateau) {
    return {
      should: true,
      reason: 'plateau',
      detail: `${result.plateau_count}/${result.total_exercises} exercices en plateau sur ${plateauSessions} séances.`,
    }
  }

  return { should: false }
}

/**
 * Calcule les objectifs d'une séance de deload pour un template_exercise.
 *
 * Règles :
 * - Poids     : current_weight_kg × (1 - load_reduction_pct / 100)
 * - Séries    : max(deload_min_sets, round(target_sets × (1 - volume_reduction_pct / 100)))
 * - Reps      : target_reps_min (fourchette basse)
 *
 * @param {object}  templateExercise  - ligne template_exercise
 * @param {object}  progressionState  - ligne progression_state
 * @param {object}  program           - ligne program
 * @param {object}  settings          - settings parsés
 * @param {number[]} achievableLoads  - charges atteignables
 * @returns {{ weight_kg, sets, reps_min, reps_max }}
 */
export function computeDeloadTargets(
  templateExercise,
  progressionState,
  program,
  settings,
  achievableLoads
) {
  const loadPct   = program.deload_load_reduction_pct   || 20
  const volPct    = program.deload_volume_reduction_pct || 40
  const minSets   = settings.deload_min_sets            || 2

  const targetW = progressionState.current_weight_kg * (1 - loadPct / 100)

  // Arrondir vers le bas à la charge atteignable la plus proche
  let weight
  if (achievableLoads && achievableLoads.length > 0) {
    const valid = achievableLoads.filter(l => l <= targetW + 0.001)
    weight = valid.length > 0 ? valid[valid.length - 1] : achievableLoads[0]
  } else {
    weight = Math.round(targetW * 100) / 100
  }

  const targetSets = Math.max(
    minSets,
    Math.round(templateExercise.target_sets * (1 - volPct / 100))
  )

  return {
    weight_kg: weight,
    sets:      targetSets,
    reps_min:  templateExercise.target_reps_min,
    reps_max:  templateExercise.target_reps_min,  // fourchette resserrée au minimum
  }
}

// ── Helper : nombre de semaines entre deux dates ISO ────────────────────
function weeksBetween(dateA, dateB) {
  const a = new Date(dateA.slice(0, 10))
  const b = new Date(dateB.slice(0, 10))
  const diffMs   = Math.abs(b - a)
  const diffDays = diffMs / (1000 * 60 * 60 * 24)
  return Math.floor(diffDays / 7)
}
```

## Exemples chiffrés de vérification

### Plateau
```
5 séances de bench press (program_id=1) :
  Séance 1 : 80kg×5  → RM Epley = 80*(1+5/30) = 93.3 kg
  Séance 2 : 80kg×5  → RM = 93.3 kg
  Séance 3 : 80kg×5  → RM = 93.3 kg
  Séance 4 : 80kg×5  → RM = 93.3 kg  (n=4 fenêtre)

  first_1rm   = 93.3 (séance 1)
  current_1rm = 93.3 (séance 4)
  delta_pct   = 0% ≤ 1% → is_plateau = true ✅
```

### Pas de plateau
```
  Séance 1 : 80kg×5  → RM = 93.3
  Séance 4 : 85kg×5  → RM = 85*(1+5/30) = 99.2
  delta_pct = (99.2-93.3)/93.3 * 100 = 6.3% > 1% → is_plateau = false ✅
```

### Deload planifié
```
program.deload_auto_enabled = 1
program.deload_frequency_weeks = 6
program.last_deload_date = "2025-05-20"
today = "2025-07-10"
weeksBetween("2025-05-20","2025-07-10") = floor(51/7) = 7 ≥ 6 → deload ✅
```

### Calcul deload targets
```
current_weight_kg = 100
load_reduction_pct = 20  → target = 80 kg
achievable = [77.5, 80, 82.5] → weight = 80 kg ✅

target_sets = 4, vol_reduction = 40% → 4 * 0.6 = 2.4 → round = 2
deload_min_sets = 2 → sets = max(2, 2) = 2 ✅
```
