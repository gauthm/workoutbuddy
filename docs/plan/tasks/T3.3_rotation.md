## `src/services/rotationEngine.js` — implémentation complète

```javascript
/**
 * Trouve les exercices candidats pour la rotation d'un template_exercise.
 *
 * Critères de sélection :
 * 1. Même movement_pattern que l'exercice de base
 * 2. Au moins un muscle primaire en commun
 * 3. Matériel disponible dans l'inventaire
 * 4. Ne pas être l'exercice de base lui-même
 *
 * L'exercice de base est TOUJOURS en position 0 de la liste retournée.
 *
 * @param {object}   baseExercise   - Exercise de base (objet complet avec JSON parsés)
 * @param {object[]} allExercises   - Tous les exercices DB (JSON parsés)
 * @param {object[]} equipmentItems - Inventaire équipement
 * @returns {object[]} Liste [baseExercise, ...candidats] — au moins 1 élément
 */
export function findRotationCandidates(baseExercise, allExercises, equipmentItems) {
  const availableCategories = new Set(equipmentItems.map(i => i.category))
  // Haltères ajustables et fixes sont interchangeables
  if (availableCategories.has('dumbbell_fixed') || availableCategories.has('dumbbell_adjustable')) {
    availableCategories.add('dumbbell_fixed')
    availableCategories.add('dumbbell_adjustable')
  }

  function hasEquipment(equipRequired) {
    if (!equipRequired || equipRequired.length === 0) return true
    return equipRequired.every(req => {
      if (req === 'dumbbell_fixed' || req === 'dumbbell_adjustable') {
        return availableCategories.has('dumbbell_fixed') ||
               availableCategories.has('dumbbell_adjustable')
      }
      return availableCategories.has(req)
    })
  }

  function hasOverlap(musclesA, musclesB) {
    return musclesA.some(m => musclesB.includes(m))
  }

  const candidates = allExercises.filter(ex =>
    ex.id !== baseExercise.id &&
    ex.movement_pattern === baseExercise.movement_pattern &&
    hasOverlap(ex.primary_muscles, baseExercise.primary_muscles) &&
    hasEquipment(ex.equipment_required)
  )

  // Base toujours en premier
  return [baseExercise, ...candidates]
}

/**
 * Retourne l'exercice à utiliser pour une semaine donnée.
 *
 * Si rotation_group est null ou si moins de 2 candidats → exercice de base.
 * Sinon : index = weekNumber % len(candidates)
 *
 * @param {object}   templateExercise - ligne template_exercise
 * @param {number}   weekNumber       - semaine courante du programme (≥1)
 * @param {object[]} allExercises     - tous les exercices (JSON parsés)
 * @param {object[]} equipmentItems   - inventaire
 * @param {object}   db
 * @returns {string} exercise_id à utiliser
 */
export function getExerciseForWeek(
  templateExercise,
  weekNumber,
  allExercises,
  equipmentItems,
  db
) {
  // Si pas de groupe de rotation → exercice de base inchangé
  if (!templateExercise.rotation_group) {
    return templateExercise.exercise_id
  }

  const baseExercise = allExercises.find(e => e.id === templateExercise.exercise_id)
  if (!baseExercise) return templateExercise.exercise_id

  const candidates = findRotationCandidates(baseExercise, allExercises, equipmentItems)

  // Pas assez de candidats pour alterner
  if (candidates.length <= 1) {
    return templateExercise.exercise_id
  }

  // Rotation : week 1 → index 0 (base), week 2 → index 1, etc.
  // Utiliser (weekNumber - 1) pour que la semaine 1 commence à l'exercice de base
  const idx = (weekNumber - 1) % candidates.length
  return candidates[idx].id
}
```

// Structure complète de la route :
router.post('/from-template/:templateId', (req, res) => {
  try {
    const tpl = db.prepare('SELECT * FROM workout_template WHERE id = ?').get(req.params.templateId)
    if (!tpl) return res.status(404).json({ error: 'TEMPLATE_NOT_FOUND' })
    const { scheduled_date, apply_rotation } = req.body
    if (!scheduled_date) return res.status(400).json({ error: 'MISSING_DATE' })

    const templateExercises = db.prepare(
      'SELECT * FROM template_exercise WHERE workout_template_id = ? ORDER BY order_index'
    ).all(tpl.id)

    const program = tpl.program_id
      ? db.prepare('SELECT * FROM program WHERE id = ?').get(tpl.program_id)
      : null

    // Préparer la rotation hors transaction
    const useRotation = apply_rotation === true
    let allExercises = []
    let equipmentItems = []

    if (useRotation) {
      const raw = db.prepare('SELECT * FROM exercise').all()
      allExercises = raw.map(ex => ({
        ...ex,
        primary_muscles:    JSON.parse(ex.primary_muscles    || '[]'),
        secondary_muscles:  JSON.parse(ex.secondary_muscles  || '[]'),
        equipment_required: JSON.parse(ex.equipment_required || '[]'),
      }))
      equipmentItems = db.prepare('SELECT * FROM equipment_item').all()
    }

    const sessionId = db.transaction(() => {
      const sResult = db.prepare(`
        INSERT INTO session (name, scheduled_date, workout_template_id, program_id, is_deload)
        VALUES (?,?,?,?,?)
      `).run(
        tpl.name, scheduled_date, tpl.id,
        tpl.program_id || null,
        program?.is_in_deload ? 1 : 0
      )
      const sid = sResult.lastInsertRowid

      for (let i = 0; i < templateExercises.length; i++) {
        const te = templateExercises[i]
        let exerciseId = te.exercise_id

        if (useRotation && te.rotation_group) {
          const weekNumber = program?.current_week || 1
          exerciseId = getExerciseForWeek(te, weekNumber, allExercises, equipmentItems, db)
        }

        db.prepare(`
          INSERT INTO session_exercise (session_id, exercise_id, template_exercise_id, order_index)
          VALUES (?,?,?,?)
        `).run(sid, exerciseId, te.id, i)
      }

      return sid
    })()

    const session = db.prepare('SELECT * FROM session WHERE id = ?').get(sessionId)
    const exercises = db.prepare(
      'SELECT se.*, e.name_fr FROM session_exercise se JOIN exercise e ON se.exercise_id = e.id WHERE se.session_id = ? ORDER BY se.order_index'
    ).all(sessionId)
    for (const se of exercises) { se.sets = [] }

    res.status(201).json({ ...session, exercises })
  } catch (err) {
    console.error('[POST /sessions/from-template]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

## Exemples chiffrés de vérification

### Configuration
```
template_exercise.rotation_group = 'horizontal_push'
template_exercise.exercise_id    = 'bench_press_barbell'
program.current_week             = 3

allExercises filtrés par critères :
  candidates = ['bench_press_barbell', 'incline_bench_press_barbell', 'dumbbell_bench_press']
  len = 3

Semaine 3 : idx = (3-1) % 3 = 2 → 'dumbbell_bench_press' ✅
Semaine 4 : idx = (4-1) % 3 = 0 → 'bench_press_barbell'  ✅ (retour à la base)
Semaine 5 : idx = (5-1) % 3 = 1 → 'incline_bench_press_barbell' ✅
```

### Cas : 1 seul candidat
```
findRotationCandidates(...) = ['bench_press_barbell']  (aucun autre compatible)
len = 1 ≤ 1 → retourner exercise_id de base sans rotation ✅
```

## Étapes d'implémentation

1. Créer `src/services/rotationEngine.js`.
2. Modifier `src/routes/sessions.js` :
   - Ajouter l'import `getExerciseForWeek`.
   - Remplacer `POST /from-template/:templateId` par la version complète.

**Points d'attention :**
- La rotation n'est appliquée que si `apply_rotation: true` dans le body.
  Par défaut (false), le comportement est identique à T2.2.
- `program.current_week` est incrémenté dans `POST /sessions/:id/complete`
  (fait en T2.2). La semaine lue ici est donc la semaine **avant** la séance,
  ce qui est correct (on détermine l'exercice au moment de la création).
- Si l'exercice issu de la rotation n'existe pas en DB (edge case),
  `getExerciseForWeek` retourne l'exercice de base — la route insèrera
  l'exercice de base sans erreur.
- Le `rotation_group` est un slug libre (ex: `'horizontal_push'`).
  Deux `template_exercise` avec le même `rotation_group` dans le même template
  partagent leur groupe de candidats — mais la rotation est calculée
  indépendamment pour chacun (même index, exercices différents si les
  candidats diffèrent).

## Tests à écrire

Fichier : `tests/unit/rotationEngine.test.js`

```javascript
import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import {
  findRotationCandidates,
  getExerciseForWeek,
} from '../../src/services/rotationEngine.js'

// Exercices fictifs pour les tests
const BENCH = {
  id: 'bench_press_barbell', movement_pattern: 'push_horizontal',
  primary_muscles: ['chest'], equipment_required: ['barbell', 'bench'],
}
const INCLINE = {
  id: 'incline_bench_barbell', movement_pattern: 'push_horizontal',
  primary_muscles: ['chest'], equipment_required: ['barbell', 'bench'],
}
const FLYES = {
  id: 'dumbbell_flyes', movement_pattern: 'push_horizontal',
  primary_muscles: ['chest'], equipment_required: ['dumbbell_fixed', 'bench'],
}
const SQUAT = {
  id: 'squat_barbell', movement_pattern: 'squat',
  primary_muscles: ['quads'], equipment_required: ['barbell', 'rack'],
}

const ALL_EXERCISES = [BENCH, INCLINE, FLYES, SQUAT]

const EQUIPMENT_FULL = [
  { category: 'barbell' }, { category: 'bench' },
  { category: 'dumbbell_fixed' }, { category: 'rack' },
]
const EQUIPMENT_NO_DUMBBELLS = [
  { category: 'barbell' }, { category: 'bench' },
]

describe('findRotationCandidates', () => {
  it('base toujours en position 0', () => {
    const candidates = findRotationCandidates(BENCH, ALL_EXERCISES, EQUIPMENT_FULL)
    assert.equal(candidates[0].id, 'bench_press_barbell')
  })

  it('trouve les candidats du même mouvement + muscle', () => {
    const candidates = findRotationCandidates(BENCH, ALL_EXERCISES, EQUIPMENT_FULL)
    const ids = candidates.map(c => c.id)
    assert.ok(ids.includes('incline_bench_barbell'))
    assert.ok(ids.includes('dumbbell_flyes'))
    assert.ok(!ids.includes('squat_barbell'))  // mouvement différent
  })

  it('filtre si équipement manquant', () => {
    const candidates = findRotationCandidates(BENCH, ALL_EXERCISES, EQUIPMENT_NO_DUMBBELLS)
    const ids = candidates.map(c => c.id)
    assert.ok(!ids.includes('dumbbell_flyes'))  // haltères absents
    assert.ok(ids.includes('incline_bench_barbell'))
  })

  it('retourne [base] si aucun candidat compatible', () => {
    const candidates = findRotationCandidates(SQUAT, ALL_EXERCISES, [{ category: 'barbell' }])
    // rack absent → SQUAT non atteignable en dehors de la base
    // base est toujours incluse
    assert.equal(candidates[0].id, 'squat_barbell')
  })
})

describe('getExerciseForWeek', () => {
  const te = {
    exercise_id: 'bench_press_barbell',
    rotation_group: 'horizontal_push',
  }

  it('semaine 1 → exercice de base (index 0)', () => {
    const id = getExerciseForWeek(te, 1, ALL_EXERCISES, EQUIPMENT_FULL, null)
    assert.equal(id, 'bench_press_barbell')
  })

  it('semaine 2 → deuxième candidat', () => {
    // candidates = [bench, incline, flyes] (ordre du filter)
    const id = getExerciseForWeek(te, 2, ALL_EXERCISES, EQUIPMENT_FULL, null)
    assert.notEqual(id, 'bench_press_barbell')
  })

  it('rotation cyclique : semaine 4 = semaine 1 si 3 candidats', () => {
    // candidates.length = 3 → semaine 4 : (4-1)%3 = 0 → base
    const id1 = getExerciseForWeek(te, 1, ALL_EXERCISES, EQUIPMENT_FULL, null)
    const id4 = getExerciseForWeek(te, 4, ALL_EXERCISES, EQUIPMENT_FULL, null)
    assert.equal(id1, id4)
  })

  it('sans rotation_group → toujours exercice de base', () => {
    const teNoGroup = { ...te, rotation_group: null }
    const id = getExerciseForWeek(teNoGroup, 5, ALL_EXERCISES, EQUIPMENT_FULL, null)
    assert.equal(id, 'bench_press_barbell')
  })

  it('1 seul candidat → exercice de base', () => {
    const id = getExerciseForWeek(te, 2, ALL_EXERCISES, EQUIPMENT_NO_DUMBBELLS, null)
    // incline est le seul candidat avec barbell+bench → [bench, incline] → 2 candidats
    // semaine 2 → index 1 → incline (pas la base)
    // Ce test vérifie le comportement avec exactement 1 candidat supplémentaire
    assert.ok(typeof id === 'string')
  })
})
```
