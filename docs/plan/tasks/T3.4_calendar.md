## Objectif

Implémenter le composant `CalendarView` : grille mensuelle avec points
colorés par statut de séance, navigation mois précédent/suivant,
interactions tap (détail / replanifier / marquer sautée).

`POST /sessions/:id/skip`.

## Endpoint à ajouter dans `src/routes/sessions.js`

```javascript
// ── POST /sessions/:id/skip ───────────────────────────────────
router.post('/:id/skip', (req, res) => {
  try {
    const session = db.prepare('SELECT * FROM session WHERE id = ?').get(req.params.id)
    if (!session) return res.status(404).json({ error: 'SESSION_NOT_FOUND' })
    if (session.status === 'completed') {
      return res.status(409).json({ error: 'SESSION_ALREADY_COMPLETED' })
    }
    if (session.status === 'in_progress') {
      return res.status(409).json({ error: 'CANNOT_SKIP_IN_PROGRESS_SESSION' })
    }
    db.prepare(`
      UPDATE session
      SET status = 'skipped', updated_at = datetime('now')
      WHERE id = ?
    `).run(req.params.id)
    res.json(db.prepare('SELECT * FROM session WHERE id = ?').get(req.params.id))
  } catch (err) {
    console.error('[POST /sessions/:id/skip]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})
```

## `public/js/components/calendar/CalendarView.js` — implémentation complète

```javascript
import { reactive, onMounted, computed, inject } from '/vendor/vue.esm-browser.prod.js'
import { sessions as api } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader    from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner   from '../common/ErrorBanner.js'
import ConfirmModal  from '../common/ConfirmModal.js'

// ── Helpers date ──────────────────────────────────────────────────────────

function isoDate(d) {
  return d.toISOString().slice(0, 10)
}

// Retourne le lundi de la semaine contenant la date
function mondayOf(date) {
  const d = new Date(date)
  const day = d.getDay()  // 0=dimanche
  const diff = day === 0 ? -6 : 1 - day
  d.setDate(d.getDate() + diff)
  return d
}

// Retourne toutes les cellules de la grille pour un mois donné
// (peut inclure des jours du mois précédent/suivant pour compléter les semaines)
function buildGrid(year, month) {
  // month : 0-indexé (0=janvier)
  const firstDay  = new Date(year, month, 1)
  const lastDay   = new Date(year, month + 1, 0)
  const startDate = mondayOf(firstDay)

  const cells = []
  const current = new Date(startDate)

  // Générer jusqu'à 6 semaines × 7 jours = 42 cellules max
  while (cells.length < 42) {
    cells.push({
      date:         isoDate(current),
      day:          current.getDate(),
      isCurrentMonth: current.getMonth() === month,
      isToday:      isoDate(current) === isoDate(new Date()),
    })
    current.setDate(current.getDate() + 1)
    // Arrêter si on a passé le dernier jour du mois ET qu'on est à un lundi
    if (current.getDate() === 1 && current.getMonth() !== month) {
      // Compléter la semaine en cours
      while (current.getDay() !== 1) {
        cells.push({
          date:           isoDate(current),
          day:            current.getDate(),
          isCurrentMonth: false,
          isToday:        false,
        })
        current.setDate(current.getDate() + 1)
      }
      break
    }
  }

  return cells
}

const MONTH_NAMES_FR = [
  'Janvier','Février','Mars','Avril','Mai','Juin',
  'Juillet','Août','Septembre','Octobre','Novembre','Décembre'
]
const MONTH_NAMES_EN = [
  'January','February','March','April','May','June',
  'July','August','September','October','November','December'
]
const DAY_HEADERS_FR = ['Lun','Mar','Mer','Jeu','Ven','Sam','Dim']
const DAY_HEADERS_EN = ['Mon','Tue','Wed','Thu','Fri','Sat','Sun']

// Couleur des points par statut
const STATUS_COLORS = {
  completed:   'var(--color-success)',
  in_progress: 'var(--color-primary)',
  planned:     'var(--color-warning)',
  skipped:     'var(--color-border)',
}

export default {
  name: 'CalendarView',
  components: { PageHeader, LoadingSpinner, ErrorBanner, ConfirmModal },

  setup() {
    const router  = inject('router')
    const store   = inject('store')
    const today   = new Date()

    const state = reactive({
      year:    today.getFullYear(),
      month:   today.getMonth(),  // 0-indexé
      loading: false,
      error:   null,
      sessions: [],               // Session[] du mois courant

      // Popup jour sélectionné
      selectedDate:     null,
      selectedSessions: [],

      // Modale replanification
      showReschedule:   false,
      rescheduleSession: null,
      newDate:          '',

      // Modale confirmation skip
      showSkip:         false,
      skipSession:      null,
    })

    // ── Grille ────────────────────────────────────────────────────
    const grid = computed(() => buildGrid(state.year, state.month))

    const monthLabel = computed(() => {
      const lang = store.getSetting('language', 'fr')
      const names = lang === 'fr' ? MONTH_NAMES_FR : MONTH_NAMES_EN
      return `${names[state.month]} ${state.year}`
    })

    const dayHeaders = computed(() => {
      const lang = store.getSetting('language', 'fr')
      return lang === 'fr' ? DAY_HEADERS_FR : DAY_HEADERS_EN
    })

    // Map date → sessions pour un accès O(1)
    const sessionsByDate = computed(() => {
      const map = {}
      for (const s of state.sessions) {
        if (!map[s.scheduled_date]) map[s.scheduled_date] = []
        map[s.scheduled_date].push(s)
      }
      return map
    })

    // ── Chargement ────────────────────────────────────────────────
    async function load() {
      state.loading = true
      state.error   = null
      state.sessions = []

      // Charger le mois courant + un peu de marge (jours du mois adjacent visibles)
      const from = isoDate(new Date(state.year, state.month - 1, 20))
      const to   = isoDate(new Date(state.year, state.month + 1, 10))

      try {
        const data = await api.list({ from, to, limit: 200 })
        state.sessions = data.sessions
      } catch (err) {
        state.error = err.message
      } finally {
        state.loading = false
      }
    }

    // ── Navigation ────────────────────────────────────────────────
    function prevMonth() {
      if (state.month === 0) { state.year--; state.month = 11 }
      else state.month--
      state.selectedDate = null
      load()
    }

    function nextMonth() {
      if (state.month === 11) { state.year++; state.month = 0 }
      else state.month++
      state.selectedDate = null
      load()
    }

    // ── Sélection d'un jour ───────────────────────────────────────
    function selectDay(cell) {
      if (state.selectedDate === cell.date) {
        state.selectedDate = null
        return
      }
      state.selectedDate     = cell.date
      state.selectedSessions = sessionsByDate.value[cell.date] || []
    }

    // ── Actions séance ────────────────────────────────────────────
    async function startSession(session) {
      try {
        await api.start(session.id)
        store.setActiveSession(session)
        router.push('#/session')
      } catch (err) {
        store.showToast(err.message, 'error')
      }
    }

    function openReschedule(session) {
      state.rescheduleSession = session
      state.newDate           = session.scheduled_date
      state.showReschedule    = true
    }

    async function doReschedule() {
      if (!state.newDate || !state.rescheduleSession) return
      try {
        await api.update(state.rescheduleSession.id, { scheduled_date: state.newDate })
        store.showToast('Séance replanifiée.', 'success')
        state.showReschedule    = false
        state.rescheduleSession = null
        state.selectedDate      = null
        await load()
      } catch (err) {
        store.showToast(err.message, 'error')
      }
    }

    function openSkip(session) {
      state.skipSession = session
      state.showSkip    = true
    }

    async function doSkip() {
      if (!state.skipSession) return
      try {
        await api.skip(state.skipSession.id)
        store.showToast('Séance marquée comme sautée.', 'info')
        state.showSkip    = false
        state.skipSession = null
        state.selectedDate = null
        await load()
      } catch (err) {
        store.showToast(err.message, 'error')
      }
    }

    function viewSession(session) {
      router.push('#/session/' + session.id)
    }

    onMounted(load)

    return {
      state, grid, monthLabel, dayHeaders,
      sessionsByDate, STATUS_COLORS,
      prevMonth, nextMonth, selectDay,
      startSession, openReschedule, doReschedule,
      openSkip, doSkip, viewSession, t,
    }
  },

  template: `
    <div class="view">
      <PageHeader :title="t('calendar.title')" />

      <div class="view__content">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner   v-else-if="state.error" :message="state.error" :retry="load" />

        <template v-else>
          <!-- Navigation mois -->
          <div class="calendar-nav">
            <button class="calendar-nav__btn" @click="prevMonth()">‹</button>
            <span class="calendar-nav__label">{{ monthLabel }}</span>
            <button class="calendar-nav__btn" @click="nextMonth()">›</button>
          </div>

          <!-- En-têtes jours -->
          <div class="calendar-grid">
            <div
              v-for="h in dayHeaders" :key="h"
              class="calendar-grid__header">
              {{ h }}
            </div>

            <!-- Cellules -->
            <div
              v-for="cell in grid" :key="cell.date"
              class="calendar-cell"
              :class="{
                'calendar-cell--other-month': !cell.isCurrentMonth,
                'calendar-cell--today':        cell.isToday,
                'calendar-cell--selected':     state.selectedDate === cell.date,
                'calendar-cell--has-session':  !!sessionsByDate[cell.date],
              }"
              @click="selectDay(cell)">

              <span class="calendar-cell__day">{{ cell.day }}</span>

              <!-- Points de statut -->
              <div class="calendar-cell__dots">
                <span
                  v-for="s in (sessionsByDate[cell.date] || [])"
                  :key="s.id"
                  class="calendar-cell__dot"
                  :style="{ background: STATUS_COLORS[s.status] || STATUS_COLORS.planned }"
                />
              </div>
            </div>
          </div>

          <!-- Légende -->
          <div class="calendar-legend">
            <span v-for="(color, status) in STATUS_COLORS" :key="status"
              class="calendar-legend__item">
              <span class="calendar-cell__dot" :style="{ background: color }" />
              {{ t('calendar.' + status) }}
            </span>
          </div>

          <!-- Panneau jour sélectionné -->
          <div v-if="state.selectedDate" class="calendar-day-panel">
            <div class="calendar-day-panel__date">{{ state.selectedDate }}</div>

            <div v-if="!state.selectedSessions.length"
              style="color:var(--color-text-muted);font-size:0.9rem;padding:0.5rem 0">
              {{ t('calendar.no_session') }}
            </div>

            <div
              v-for="session in state.selectedSessions"
              :key="session.id"
              class="calendar-day-session">
              <div class="calendar-day-session__name">{{ session.name }}</div>
              <div class="calendar-day-session__status"
                :style="{ color: STATUS_COLORS[session.status] }">
                {{ t('calendar.' + session.status) }}
              </div>

              <!-- Actions selon statut -->
              <div class="calendar-day-session__actions">
                <template v-if="session.status === 'completed'">
                  <button class="btn btn--secondary"
                    style="padding:0.3rem 0.75rem;font-size:0.8rem"
                    @click="viewSession(session)">
                    Voir
                  </button>
                </template>

                <template v-if="session.status === 'in_progress'">
                  <button class="btn btn--primary"
                    style="padding:0.3rem 0.75rem;font-size:0.8rem"
                    @click="router.push('#/session')">
                    Reprendre
                  </button>
                </template>

                <template v-if="session.status === 'planned'">
                  <button class="btn btn--primary"
                    style="padding:0.3rem 0.75rem;font-size:0.8rem"
                    @click="startSession(session)">
                    {{ t('calendar.start') }}
                  </button>
                  <button class="btn btn--secondary"
                    style="padding:0.3rem 0.75rem;font-size:0.8rem"
                    @click="openReschedule(session)">
                    {{ t('calendar.reschedule') }}
                  </button>
                  <button class="btn btn--secondary"
                    style="padding:0.3rem 0.75rem;font-size:0.8rem;color:var(--color-text-muted)"
                    @click="openSkip(session)">
                    {{ t('calendar.mark_skipped') }}
                  </button>
                </template>
              </div>
            </div>
          </div>
        </template>
      </div>

      <!-- Modale replanification -->
      <div v-if="state.showReschedule" class="modal-overlay" @click.self="state.showReschedule=false">
        <div class="modal">
          <h2 class="modal__title">{{ t('calendar.reschedule_title') }}</h2>
          <div class="form-group">
            <label class="form-label">{{ t('calendar.reschedule_date') }}</label>
            <input class="form-input" type="date" v-model="state.newDate" />
          </div>
          <div class="modal__actions">
            <button class="btn btn--secondary" @click="state.showReschedule=false">
              {{ t('common.cancel') }}
            </button>
            <button class="btn btn--primary" @click="doReschedule()">
              {{ t('common.confirm') }}
            </button>
          </div>
        </div>
      </div>

      <!-- Modale skip -->
      <ConfirmModal
        v-if="state.showSkip"
        :title="t('calendar.mark_skipped')"
        :body="state.skipSession?.name"
        @confirm="doSkip()"
        @cancel="state.showSkip = false"
      />
    </div>
  `
}
```

## Étapes d'implémentation

1. Ajouter `POST /:id/skip` dans `src/routes/sessions.js` (avant `export default`).
2. Ajouter `skip` dans l'objet `sessions` de `public/js/api.js`.
3. Créer `public/js/components/calendar/CalendarView.js`.
4. Ajouter le CSS dans `public/css/components.css`.
5. Ajouter les clés i18n dans `fr.json` et `en.json`.
6. Modifier `AppShell.js`.

**Points d'attention :**
- `buildGrid()` retourne des cellules qui peuvent appartenir au mois
  précédent ou suivant (pour compléter les semaines). Ces cellules
  ont `isCurrentMonth: false` et sont affichées avec `opacity: 0.35`.
- Le calendrier charge toujours `from = mois-1 jour 20` et
  `to = mois+1 jour 10` pour avoir les jours des semaines adjacentes.
  L'API `GET /sessions` supporte ces dates hors du mois affiché.
- `sessionsByDate` est un `computed` — il se recalcule automatiquement
  quand `state.sessions` change après `load()`.
- Tap sur un jour déjà sélectionné → déselectionner (toggle).
- Si plusieurs séances un même jour → plusieurs points (autant que de séances).
- `api.skip` doit être ajouté manuellement dans `api.js` (non présent en T1.6) :
  ```javascript
  // Dans l'objet sessions :
  skip: (id) => request('POST', `/sessions/${id}/skip`),
  ```

## Tests à écrire

Pas de tests unitaires automatisés pour le composant Vue.
Vérification manuelle via les critères ci-dessous.

Fichier : `tests/integration/calendar.test.js`

```javascript
import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import request from 'supertest'
import { app } from '../../server.js'

describe('T3.4 — Calendrier API', () => {
  let sessionId

  it('POST /sessions crée une séance planifiée', async () => {
    const res = await request(app).post('/api/v1/sessions').send({
      name: 'Push test', scheduled_date: '2025-07-15'
    })
    assert.equal(res.status, 201)
    assert.equal(res.body.status, 'planned')
    sessionId = res.body.id
  })

  it('GET /sessions?from=&to= retourne les séances du mois', async () => {
    const res = await request(app)
      .get('/api/v1/sessions?from=2025-07-01&to=2025-07-31')
    assert.equal(res.status, 200)
    assert.ok(Array.isArray(res.body.sessions))
    const found = res.body.sessions.find(s => s.id === sessionId)
    assert.ok(found)
  })

  it('POST /sessions/:id/skip passe status à skipped', async () => {
    const res = await request(app).post(`/api/v1/sessions/${sessionId}/skip`)
    assert.equal(res.status, 200)
    assert.equal(res.body.status, 'skipped')
  })

  it('PUT /sessions/:id replanifie une séance planned', async () => {
    const res2 = await request(app).post('/api/v1/sessions').send({
      name: 'Push test 2', scheduled_date: '2025-07-20'
    })
    const res = await request(app)
      .put(`/api/v1/sessions/${res2.body.id}`)
      .send({ scheduled_date: '2025-07-22' })
    assert.equal(res.status, 200)
    assert.equal(res.body.scheduled_date, '2025-07-22')
  })

  it('POST /sessions/:id/skip sur séance complétée → 409', async () => {
    // Créer et compléter une séance
    const s = await request(app).post('/api/v1/sessions')
      .send({ name: 'Done', scheduled_date: '2025-07-10' })
    await request(app).post(`/api/v1/sessions/${s.body.id}/start`)
    await request(app).post(`/api/v1/sessions/${s.body.id}/complete`)
    const res = await request(app).post(`/api/v1/sessions/${s.body.id}/skip`)
    assert.equal(res.status, 409)
  })
})
```
