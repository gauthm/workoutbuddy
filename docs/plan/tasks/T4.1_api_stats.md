## `src/services/streakCalculator.js` — implémentation complète

```javascript
/**
 * Calcule les streaks de l'utilisateur à partir des séances complétées.
 *
 * Définitions :
 * - streak semaine : nombre de semaines ISO consécutives avec ≥1 séance complétée,
 *   en remontant à partir de la semaine courante.
 * - streak séance : non utilisé dans le MVP (remplacé par total_sessions).
 * - total_weeks   : nombre de semaines distinctes avec au moins une séance.
 *
 * @param {object} db - instance better-sqlite3
 * @returns {{
 *   current_week_streak: number,
 *   longest_week_streak: number,
 *   total_sessions: number,
 *   total_weeks: number
 * }}
 */
export function calculateStreaks(db) {
  // 1. Dates de toutes les séances complétées
  const rows = db.prepare(`
    SELECT DISTINCT date(scheduled_date) AS d
    FROM session
    WHERE status = 'completed'
    ORDER BY d DESC
  `).all()

  const totalSessions = db.prepare(
    "SELECT COUNT(*) AS n FROM session WHERE status = 'completed'"
  ).get().n

  if (rows.length === 0) {
    return {
      current_week_streak: 0,
      longest_week_streak: 0,
      total_sessions:      0,
      total_weeks:         0,
    }
  }

  // 2. Ensemble des semaines ISO avec au moins une séance
  const weeksWithSession = new Set(rows.map(r => isoWeek(r.d)))
  const totalWeeks = weeksWithSession.size

  // 3. Streak courant : remonter depuis la semaine courante
  const currentWeekStr = isoWeek(todayIso())
  let streakWeeks = 0
  let w = currentWeekStr
  while (weeksWithSession.has(w)) {
    streakWeeks++
    w = previousIsoWeek(w)
  }

  // 4. Streak le plus long
  const sortedWeeks = Array.from(weeksWithSession).sort()
  let longest = sortedWeeks.length > 0 ? 1 : 0
  let current = 1
  for (let i = 1; i < sortedWeeks.length; i++) {
    if (sortedWeeks[i] === nextIsoWeek(sortedWeeks[i - 1])) {
      current++
      if (current > longest) longest = current
    } else {
      current = 1
    }
  }

  return {
    current_week_streak: streakWeeks,
    longest_week_streak: longest,
    total_sessions:      totalSessions,
    total_weeks:         totalWeeks,
  }
}

// ── Helpers date ─────────────────────────────────────────────────────────

function todayIso() {
  return new Date().toISOString().slice(0, 10)
}

/**
 * Retourne la semaine ISO au format "YYYY-Www" (ex: "2025-W27").
 * @param {string} dateStr - "YYYY-MM-DD"
 */
export function isoWeek(dateStr) {
  const d = new Date(dateStr + 'T12:00:00Z')  // midi UTC pour éviter les décalages
  d.setUTCDate(d.getUTCDate() + 3 - ((d.getUTCDay() + 6) % 7))
  const week1 = new Date(Date.UTC(d.getUTCFullYear(), 0, 4))
  const weekNum = 1 + Math.round(
    ((d - week1) / 86400000 - 3 + (week1.getUTCDay() + 6) % 7) / 7
  )
  return `${d.getUTCFullYear()}-W${String(weekNum).padStart(2, '0')}`
}

/**
 * Retourne la semaine ISO précédente.
 * @param {string} weekStr - "YYYY-Www"
 */
export function previousIsoWeek(weekStr) {
  const [year, w] = weekStr.split('-W').map(Number)
  if (w === 1) {
    // Dernière semaine de l'année précédente : trouver le 28 décembre
    const dec28 = new Date(Date.UTC(year - 1, 11, 28))
    return isoWeek(dec28.toISOString().slice(0, 10))
  }
  return `${year}-W${String(w - 1).padStart(2, '0')}`
}

/**
 * Retourne la semaine ISO suivante.
 * @param {string} weekStr - "YYYY-Www"
 */
export function nextIsoWeek(weekStr) {
  const [year, w] = weekStr.split('-W').map(Number)
  // Trouver le nombre de semaines dans cette année
  const dec28 = new Date(Date.UTC(year, 11, 28))
  const weeksInYear = isoWeek(dec28.toISOString().slice(0, 10)).split('-W')[1]
  if (w >= parseInt(weeksInYear)) {
    return `${year + 1}-W01`
  }
  return `${year}-W${String(w + 1).padStart(2, '0')}`
}
```

```javascript
  try {

      SELECT COALESCE(SUM(total_volume_kg), 0) AS vol
      FROM session WHERE status = 'completed'

      "SELECT scheduled_date FROM session WHERE status = 'completed' ORDER BY scheduled_date DESC LIMIT 1"
    ).get()

      "SELECT name FROM program WHERE is_active = 1 LIMIT 1"

      'SELECT COUNT(*) AS n FROM personal_record'

// Query: weeks=12 (défaut 12), group_by='week'|'muscle' (défaut 'week')
router.get('/volume', (req, res) => {
  try {
    const weeks   = Math.min(parseInt(req.query.weeks) || 12, 52)
    const groupBy = req.query.group_by === 'muscle' ? 'muscle' : 'week'

    if (groupBy === 'week') {
      // Volume total par semaine ISO
      const rows = db.prepare(`
        SELECT
          strftime('%Y', s.scheduled_date)                      AS year,
          CAST(strftime('%W', s.scheduled_date) AS INTEGER) + 1 AS week_num,
          COALESCE(SUM(sl.weight_kg * sl.reps), 0)              AS volume_kg,
          COUNT(DISTINCT sl.id)                                  AS sets
        FROM session s
        JOIN session_exercise se ON se.session_id = s.id
        JOIN set_log sl          ON sl.session_exercise_id = se.id
        WHERE s.status        = 'completed'
          AND sl.is_warmup    = 0
          AND sl.is_completed = 1
          AND s.scheduled_date >= date('now', ? || ' days')
        GROUP BY year, week_num
        ORDER BY year ASC, week_num ASC
      `).all(`-${weeks * 7}`)

      // Formater en "YYYY-Www"
      const data = rows.map(r => ({
        period:    `${r.year}-W${String(r.week_num).padStart(2, '0')}`,
        volume_kg: Math.round(r.volume_kg * 100) / 100,
        sets:      r.sets,
      }))
      return res.json({ data })
    }

    // group_by = 'muscle'
    // Récupérer les séries récentes avec l'exercice associé
    const rows = db.prepare(`
      SELECT
        e.primary_muscles,
        strftime('%Y', s.scheduled_date)                      AS year,
        CAST(strftime('%W', s.scheduled_date) AS INTEGER) + 1 AS week_num,
        SUM(sl.weight_kg * sl.reps)                            AS volume_kg,
        COUNT(DISTINCT sl.id)                                  AS sets
      FROM session s
      JOIN session_exercise se ON se.session_id = s.id
      JOIN set_log sl          ON sl.session_exercise_id = se.id
      JOIN exercise e          ON se.exercise_id = e.id
      WHERE s.status        = 'completed'
        AND sl.is_warmup    = 0
        AND sl.is_completed = 1
        AND s.scheduled_date >= date('now', ? || ' days')
      GROUP BY e.id, year, week_num
      ORDER BY year ASC, week_num ASC
    `).all(`-${weeks * 7}`)

    // Éclater par muscle primaire
    const data = []
    for (const row of rows) {
      const muscles = JSON.parse(row.primary_muscles || '[]')
      const primary = muscles[0] || 'other'
      data.push({
        period:    `${row.year}-W${String(row.week_num).padStart(2, '0')}`,
        muscle:    primary,
        volume_kg: Math.round((row.volume_kg || 0) * 100) / 100,
        sets:      row.sets,
      })
    }
    res.json({ data })
  } catch (err) {
    console.error('[GET /stats/volume]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
```

```javascript
import express from 'express'
import { getDb } from '../db/database.js'

const router = express.Router()
const db     = getDb()

// ── GET /bodyweight ───────────────────────────────────────────
// Query: days=90 (défaut : setting bodyweight_display_days)
router.get('/', (req, res) => {
  try {
    const settingRow = db.prepare(
      "SELECT value FROM settings WHERE key = 'bodyweight_display_days'"
    ).get()
    const defaultDays = settingRow ? JSON.parse(settingRow.value) : 90
    const days = Math.min(parseInt(req.query.days) || defaultDays, 365 * 3)

    const rows = db.prepare(`
      SELECT * FROM body_weight
      WHERE date >= date('now', ? || ' days')
      ORDER BY date ASC
    `).all(`-${days}`)

    res.json({ entries: rows })
  } catch (err) {
    console.error('[GET /bodyweight]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── POST /bodyweight ──────────────────────────────────────────
router.post('/', (req, res) => {
  try {
    const { date, weight_kg, notes } = req.body
    if (!date)       return res.status(400).json({ error: 'MISSING_DATE' })
    if (!weight_kg || weight_kg <= 0) {
      return res.status(400).json({ error: 'INVALID_WEIGHT' })
    }

    // Vérifier la date (format YYYY-MM-DD)
    if (!/^\d{4}-\d{2}-\d{2}$/.test(date)) {
      return res.status(400).json({ error: 'INVALID_DATE_FORMAT' })
    }

    // Vérifier doublon
    const existing = db.prepare('SELECT id FROM body_weight WHERE date = ?').get(date)
    if (existing) {
      return res.status(409).json({ error: 'BODYWEIGHT_DATE_EXISTS', detail: `Une entrée existe déjà pour le ${date}.` })
    }

    const result = db.prepare(
      'INSERT INTO body_weight (date, weight_kg, notes) VALUES (?,?,?)'
    ).run(date, parseFloat(weight_kg), notes || null)

    res.status(201).json(
      db.prepare('SELECT * FROM body_weight WHERE id = ?').get(result.lastInsertRowid)
    )
  } catch (err) {
    console.error('[POST /bodyweight]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── PUT /bodyweight/:id ───────────────────────────────────────
router.put('/:id', (req, res) => {
  try {
    const entry = db.prepare('SELECT id FROM body_weight WHERE id = ?').get(req.params.id)
    if (!entry) return res.status(404).json({ error: 'BODYWEIGHT_NOT_FOUND' })

    const updates = []; const values = []
    if (req.body.weight_kg !== undefined) {
      if (req.body.weight_kg <= 0) return res.status(400).json({ error: 'INVALID_WEIGHT' })
      updates.push('weight_kg = ?')
      values.push(parseFloat(req.body.weight_kg))
    }
    if (req.body.notes !== undefined) {
      updates.push('notes = ?')
      values.push(req.body.notes)
    }
    if (updates.length === 0) return res.status(400).json({ error: 'NO_FIELDS_TO_UPDATE' })

    values.push(req.params.id)
    db.prepare(`UPDATE body_weight SET ${updates.join(', ')} WHERE id = ?`).run(...values)
    res.json(db.prepare('SELECT * FROM body_weight WHERE id = ?').get(req.params.id))
  } catch (err) {
    console.error('[PUT /bodyweight/:id]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── DELETE /bodyweight/:id ────────────────────────────────────
router.delete('/:id', (req, res) => {
  try {
    const entry = db.prepare('SELECT id FROM body_weight WHERE id = ?').get(req.params.id)
    if (!entry) return res.status(404).json({ error: 'BODYWEIGHT_NOT_FOUND' })
    db.prepare('DELETE FROM body_weight WHERE id = ?').run(req.params.id)
    res.status(204).end()
  } catch (err) {
    console.error('[DELETE /bodyweight/:id]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

export default router
```

## Exemples chiffrés de vérification

### Streaks

```
Séances complétées (dates) :
  2025-06-02 (semaine 2025-W23)
  2025-06-09 (semaine 2025-W24)
  2025-06-16 (semaine 2025-W25)
  2025-06-30 (semaine 2025-W27)  ← gap : W26 manquante
  2025-07-07 (semaine 2025-W28)

Aujourd'hui = 2025-07-09 (semaine 2025-W28)

Streak courant : W28 ✅, W27 ✅, W26 ❌ → current_week_streak = 2
Longest : W23-W25 = 3 semaines consécutives → longest_week_streak = 3
total_weeks = 5
```

### isoWeek

```
isoWeek('2025-01-01') = '2025-W01'  (1er janvier 2025 = mercredi semaine 1)
isoWeek('2025-12-29') = '2026-W01'  (29 déc 2025 appartient à la semaine 1 de 2026)
isoWeek('2025-07-07') = '2025-W28'
```

## Étapes d'implémentation

1. Créer `src/services/streakCalculator.js`.
2. Créer `src/routes/stats.js` (importe `streakCalculator` et `rmCalculator`).
3. Créer `src/routes/bodyweight.js`.
4. Modifier `server.js` : 2 imports + 2 `app.use()`.

**Points d'attention :**
- `isoWeek()` utilise `T12:00:00Z` pour éviter les décalages de timezone.
  Ne pas utiliser `new Date(dateStr)` sans heure — il peut interpréter
  la date en local time et décaler d'un jour.
- `strftime('%W', ...)` dans SQLite retourne 0 pour la première semaine
  partielle de l'année (semaine 0). Ajouter `+ 1` dans la requête SQL
  pour approximer. Ce n'est pas la semaine ISO exacte (qui peut être W52/W53
  de l'année précédente), mais c'est suffisant pour les graphiques de volume.
- `GET /stats/volume?group_by=muscle` peut retourner plusieurs entrées pour
  une même semaine si l'exercice a plusieurs muscles primaires — on prend
  uniquement `muscles[0]` (premier muscle primaire) pour éviter de compter
  le volume plusieurs fois.
- `GET /stats/personal-records` sans `exercise_id` retourne TOUS les records
  de TOUS les exercices — potentiellement long mais acceptable (< 1000 lignes).

```javascript
    assert.equal(res.status, 200)
    assert.ok(typeof res.body.current_week_streak === 'number')
    assert.ok(typeof res.body.longest_week_streak === 'number')

  })
```
