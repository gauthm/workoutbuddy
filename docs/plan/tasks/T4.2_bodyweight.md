## Note sur Chart.js

Chart.js est chargé en UMD via `<script src="/vendor/chart.umd.min.js">`.
Il est disponible comme **variable globale `Chart`** (pas d'import ES module).

Dans les composants Vue, accéder à Chart.js ainsi :
```javascript
// PAS d'import — utiliser la variable globale
const chart = new window.Chart(canvasElement, config)
```

Si `window.Chart` est undefined, Chart.js n'a pas été chargé.
Vérifier que `public/index.html` a bien ce script **avant** `app.js` :
```html
<script src="/vendor/chart.umd.min.js"></script>
<script type="module" src="/js/app.js"></script>
```

## `public/js/components/stats/BodyWeightView.js` — implémentation complète

```javascript
import { reactive, onMounted, onUnmounted, inject, computed, nextTick }
  from '/vendor/vue.esm-browser.prod.js'
import { bodyweight as api } from '../../api.js'
import { kgToLb, formatWeight } from '../../utils/weight.js'
import { t } from '../../i18n.js'
import PageHeader     from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner    from '../common/ErrorBanner.js'
import EmptyState     from '../common/EmptyState.js'
import ConfirmModal   from '../common/ConfirmModal.js'

// ── Calcul de la moyenne mobile 7 jours ──────────────────────────────────
function movingAverage(entries, windowDays = 7) {
  return entries.map((entry, idx) => {
    const windowStart = idx - windowDays + 1
    const slice = entries.slice(Math.max(0, windowStart), idx + 1)
    const avg   = slice.reduce((sum, e) => sum + e.weight_kg, 0) / slice.length
    return { date: entry.date, avg: Math.round(avg * 100) / 100 }
  })
}

// ── Tendance (comparaison 7 derniers jours vs 7 jours précédents) ────────
function computeTrend(entries) {
  if (entries.length < 2) return 'stable'
  const last7  = entries.slice(-7)
  const prev7  = entries.slice(-14, -7)
  if (prev7.length === 0) return 'stable'
  const avgLast = last7.reduce((s, e) => s + e.weight_kg, 0) / last7.length
  const avgPrev = prev7.reduce((s, e) => s + e.weight_kg, 0) / prev7.length
  const diff = avgLast - avgPrev
  if (diff > 0.3) return 'up'
  if (diff < -0.3) return 'down'
  return 'stable'
}

export default {
  name: 'BodyWeightView',
  components: {
    PageHeader, LoadingSpinner, ErrorBanner, EmptyState, ConfirmModal
  },

  setup() {
    const store = inject('store')

    const state = reactive({
      loading:    false,
      error:      null,
      entries:    [],           // BodyWeight[] triés ASC par date
      showForm:   false,
      form:       { date: new Date().toISOString().slice(0, 10), weight_kg: '', notes: '' },
      saving:     false,
      deleteTarget: null,
    })

    let chartInstance = null

    const unit = computed(() => store.getSetting('unit', 'kg'))

    const latestEntry = computed(() => {
      if (!state.entries.length) return null
      return state.entries[state.entries.length - 1]
    })

    const trend = computed(() => computeTrend(state.entries))

    // ── Chargement ──────────────────────────────────────────────
    async function load() {
      state.loading = true; state.error = null
      try {
        const data = await api.list({ days: 365 })
        // Trier ASC par date pour la courbe
        state.entries = [...data.entries].sort((a, b) => a.date.localeCompare(b.date))
        await nextTick()
        renderChart()
      } catch (err) {
        state.error = err.message
      } finally {
        state.loading = false
      }
    }

    // ── Rendu du graphique ──────────────────────────────────────
    function renderChart() {
      if (!window.Chart) {
        console.warn('[BodyWeightView] Chart.js non disponible')
        return
      }
      const canvas = document.getElementById('bodyweight-chart')
      if (!canvas || state.entries.length === 0) return

      // Détruire l'instance précédente
      if (chartInstance) {
        chartInstance.destroy()
        chartInstance = null
      }

      const movAvg = movingAverage(state.entries)

      // Convertir selon l'unité
      const rawData = state.entries.map(e => ({
        x: e.date,
        y: unit.value === 'lb'
          ? Math.round(kgToLb(e.weight_kg) * 10) / 10
          : e.weight_kg,
      }))
      const avgData = movAvg.map(e => ({
        x: e.date,
        y: unit.value === 'lb'
          ? Math.round(kgToLb(e.avg) * 10) / 10
          : e.avg,
      }))

      chartInstance = new window.Chart(canvas, {
        type: 'line',
        data: {
          datasets: [
            {
              label:           t('bodyweight.weight'),
              data:            rawData,
              borderColor:     'rgba(230,126,34,0.5)',
              backgroundColor: 'rgba(230,126,34,0.1)',
              pointRadius:     3,
              pointHoverRadius: 5,
              tension:         0.2,
              fill:            false,
            },
            {
              label:           t('bodyweight.moving_avg'),
              data:            avgData,
              borderColor:     '#e67e22',
              backgroundColor: 'transparent',
              pointRadius:     0,
              borderWidth:     2,
              tension:         0.3,
              fill:            false,
            },
          ],
        },
        options: {
          responsive:         true,
          maintainAspectRatio: false,
          interaction:        { mode: 'index', intersect: false },
          plugins: {
            legend: {
              labels: { color: '#e8e8e8', font: { size: 12 } }
            },
            tooltip: {
              callbacks: {
                label: (ctx) => `${ctx.dataset.label}: ${ctx.parsed.y} ${unit.value}`
              }
            },
          },
          scales: {
            x: {
              type:    'time',
              time:    { unit: 'week', tooltipFormat: 'dd/MM/yyyy' },
              ticks:   { color: '#888', maxTicksLimit: 8 },
              grid:    { color: 'rgba(255,255,255,0.05)' },
            },
            y: {
              ticks: {
                color:    '#888',
                callback: (v) => `${v} ${unit.value}`,
              },
              grid: { color: 'rgba(255,255,255,0.05)' },
            },
          },
        },
      })
    }

    // ── Ajout ───────────────────────────────────────────────────
    async function saveEntry() {
      if (!state.form.weight_kg || parseFloat(state.form.weight_kg) <= 0) {
        store.showToast('Poids invalide.', 'error'); return
      }
      state.saving = true
      try {
        // Convertir lb → kg si nécessaire
        const weightKg = unit.value === 'lb'
          ? Math.round(parseFloat(state.form.weight_kg) / 2.20462 * 100) / 100
          : parseFloat(state.form.weight_kg)

        await api.create({
          date:      state.form.date,
          weight_kg: weightKg,
          notes:     state.form.notes || null,
        })
        state.form   = { date: new Date().toISOString().slice(0, 10), weight_kg: '', notes: '' }
        state.showForm = false
        store.showToast('Poids enregistré.', 'success')
        await load()
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        state.saving = false
      }
    }

    // ── Suppression ─────────────────────────────────────────────
    async function doDelete() {
      if (!state.deleteTarget) return
      try {
        await api.delete(state.deleteTarget.id)
        store.showToast('Entrée supprimée.', 'success')
        state.deleteTarget = null
        await load()
      } catch (err) {
        store.showToast(err.message, 'error')
      }
    }

    function displayWeight(kg) {
      return formatWeight(kg, unit.value)
    }

    onMounted(load)
    onUnmounted(() => {
      if (chartInstance) { chartInstance.destroy(); chartInstance = null }
    })

    return {
      state, unit, latestEntry, trend,
      saveEntry, doDelete, displayWeight, load, t,
    }
  },

  template: `
    <div class="view">
      <PageHeader :title="t('bodyweight.title')">
        <template #actions>
          <button class="btn btn--primary"
            style="padding:0.4rem 0.75rem;font-size:0.85rem"
            @click="state.showForm = !state.showForm">
            {{ state.showForm ? t('common.close') : t('bodyweight.add') }}
          </button>
        </template>
      </PageHeader>

      <div class="view__content" style="padding:0 1rem">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner v-else-if="state.error" :message="state.error" :retry="load" />

        <template v-else>
          <!-- Formulaire d'ajout -->
          <div v-if="state.showForm" class="bw-form">
            <form @submit.prevent="saveEntry">
              <div class="form-group">
                <label class="form-label">{{ t('bodyweight.date') }}</label>
                <input class="form-input" type="date" v-model="state.form.date" required />
              </div>
              <div class="form-group">
                <label class="form-label">
                  {{ t('bodyweight.weight') }} ({{ unit }})
                </label>
                <input class="form-input" type="number" step="0.1" min="20" max="500"
                  v-model="state.form.weight_kg" required
                  :placeholder="unit === 'kg' ? '75.0' : '165.0'" />
              </div>
              <div class="form-group">
                <label class="form-label">{{ t('bodyweight.notes') }}</label>
                <input class="form-input" type="text" v-model="state.form.notes" />
              </div>
              <button type="submit" class="btn btn--primary btn--full"
                :disabled="state.saving">
                {{ state.saving ? t('common.loading') : t('common.save') }}
              </button>
            </form>
          </div>

          <EmptyState v-if="!state.entries.length"
            icon="⚖️" :message="t('bodyweight.no_entries')">
            <button class="btn btn--primary" style="margin-top:1rem"
              @click="state.showForm = true">
              {{ t('bodyweight.add') }}
            </button>
          </EmptyState>

          <template v-else>
            <!-- Résumé -->
            <div class="bw-summary">
              <div class="bw-summary__item">
                <div class="detail-section__title">{{ t('bodyweight.latest') }}</div>
                <div class="bw-summary__value">
                  {{ displayWeight(latestEntry.weight_kg) }}
                </div>
                <div style="font-size:0.75rem;color:var(--color-text-muted)">
                  {{ latestEntry.date }}
                </div>
              </div>
              <div class="bw-summary__item">
                <div class="detail-section__title">{{ t('bodyweight.trend') }}</div>
                <div class="bw-summary__value"
                  :style="{
                    color: trend==='up' ? 'var(--color-warning)'
                         : trend==='down' ? 'var(--color-success)'
                         : 'var(--color-text-muted)'
                  }">
                  {{ t('bodyweight.trend_' + trend) }}
                </div>
              </div>
            </div>

            <!-- Graphique -->
            <div class="bw-chart-container">
              <canvas id="bodyweight-chart"></canvas>
            </div>

            <!-- Liste des entrées (30 dernières) -->
            <div class="bw-list">
              <div
                v-for="entry in [...state.entries].reverse().slice(0, 30)"
                :key="entry.id"
                class="bw-list-row">
                <div class="bw-list-row__info">
                  <span class="bw-list-row__date">{{ entry.date }}</span>
                  <span class="bw-list-row__weight">
                    {{ displayWeight(entry.weight_kg) }}
                  </span>
                </div>
                <button class="icon-btn icon-btn--danger"
                  @click="state.deleteTarget = entry">🗑</button>
              </div>
            </div>
          </template>
        </template>
      </div>

      <ConfirmModal
        v-if="state.deleteTarget"
        :title="t('bodyweight.delete_confirm')"
        :body="state.deleteTarget ? state.deleteTarget.date + ' — ' + displayWeight(state.deleteTarget.weight_kg) : ''"
        :danger="true"
        @confirm="doDelete()"
        @cancel="state.deleteTarget = null"
      />
    </div>
  `
}
```

## Étapes d'implémentation

1. Modifier `public/index.html` : ajouter le script Chart.js.
2. Ajouter les clés i18n dans `fr.json` et `en.json`.
3. Créer `public/js/components/stats/BodyWeightView.js`.
4. Ajouter le CSS dans `components.css`.
5. Modifier `AppShell.js`.

**Points d'attention :**
- `renderChart()` est appelée après `nextTick()` pour s'assurer que
  le `<canvas>` est bien dans le DOM avant d'instancier Chart.js.
- `onUnmounted` détruit l'instance Chart pour éviter les fuites mémoire
  quand l'utilisateur navigue vers une autre vue.
- Chart.js `type: 'time'` requiert le module `chartjs-adapter-date-fns`
  OU l'adaptateur inclus dans le bundle UMD de Chart.js 4. Si l'axe
  de temps ne fonctionne pas, remplacer par `type: 'category'` :
  ```javascript
  // Fallback si 'time' ne fonctionne pas :
  scales: { x: { type: 'category', ... } }
  ```
- `movingAverage()` avec `windowDays=7` : les 6 premières entrées ont
  une fenêtre plus petite — c'est correct (comportement standard de la
  moyenne mobile "expanding window").
- La saisie en livres est convertie en kg avant le `api.create()`.
  La DB stocke toujours en kg.
