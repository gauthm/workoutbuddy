## `VolumeChart.js`

```javascript
import { onMounted, onUnmounted, watch } from '/vendor/vue.esm-browser.prod.js'
import { t } from '../../i18n.js'

// Palette de couleurs par groupe musculaire
const MUSCLE_COLORS = {
  chest:      '#e74c3c',
  back:       '#3498db',
  quads:      '#2ecc71',
  hamstrings: '#27ae60',
  glutes:     '#1abc9c',
  shoulders:  '#9b59b6',
  biceps:     '#e67e22',
  triceps:    '#f39c12',
  calves:     '#16a085',
  core:       '#2980b9',
  traps:      '#8e44ad',
  forearms:   '#d35400',
  other:      '#95a5a6',
}

export default {
  name: 'VolumeChart',
  props: {
    data:    { type: Array, default: () => [] },
    // data: [{ period: "2025-W27", volume_kg: float, sets?: int, muscle?: string }]
    groupBy: { type: String, default: 'week' }, // 'week' | 'muscle'
    canvasId:{ type: String, default: 'volume-chart' },
  },

  setup(props) {
    let chartInstance = null

    function render() {
      if (!window.Chart) return
      if (!props.data || props.data.length === 0) return

      const canvas = document.getElementById(props.canvasId)
      if (!canvas) return

      if (chartInstance) { chartInstance.destroy(); chartInstance = null }

      if (props.groupBy === 'week') {
        // Histogramme simple : volume total par semaine
        chartInstance = new window.Chart(canvas, {
          type: 'bar',
          data: {
            labels: props.data.map(d => d.period),
            datasets: [{
              label:           t('stats.volume_chart'),
              data:            props.data.map(d => d.volume_kg),
              backgroundColor: 'rgba(230,126,34,0.7)',
              borderColor:     '#e67e22',
              borderWidth:     1,
              borderRadius:    4,
            }],
          },
          options: {
            responsive: true, maintainAspectRatio: false,
            plugins: {
              legend: { display: false },
              tooltip: {
                callbacks: {
                  label: (ctx) => `${Math.round(ctx.parsed.y)} kg`,
                }
              },
            },
            scales: {
              x: {
                ticks: { color: '#888', maxTicksLimit: 8, maxRotation: 45 },
                grid:  { display: false },
              },
              y: {
                ticks: { color: '#888', callback: (v) => `${v} kg` },
                grid:  { color: 'rgba(255,255,255,0.05)' },
                beginAtZero: true,
              },
            },
          },
        })
        return
      }

      // group_by = muscle : histogramme empilé par muscle
      // Collecter toutes les périodes et tous les muscles uniques
      const periodsSet = new Set(props.data.map(d => d.period))
      const musclesSet = new Set(props.data.map(d => d.muscle || 'other'))
      const periods    = Array.from(periodsSet).sort()
      const muscles    = Array.from(musclesSet)

      // Construire un objet { muscle → Map<period, volume> }
      const volumeMap = {}
      for (const m of muscles) {
        volumeMap[m] = new Map()
        for (const p of periods) volumeMap[m].set(p, 0)
      }
      for (const d of props.data) {
        const m = d.muscle || 'other'
        volumeMap[m].set(d.period, (volumeMap[m].get(d.period) || 0) + d.volume_kg)
      }

      const datasets = muscles.map(m => ({
        label:           m,
        data:            periods.map(p => Math.round(volumeMap[m].get(p) * 100) / 100),
        backgroundColor: MUSCLE_COLORS[m] || MUSCLE_COLORS.other,
        stack:           'volume',
        borderRadius:    2,
      }))

      chartInstance = new window.Chart(canvas, {
        type: 'bar',
        data: { labels: periods, datasets },
        options: {
          responsive: true, maintainAspectRatio: false,
          plugins: {
            legend: {
              labels: { color: '#e8e8e8', font: { size: 10 }, boxWidth: 12 },
            },
          },
          scales: {
            x: {
              stacked: true,
              ticks:   { color: '#888', maxTicksLimit: 8 },
              grid:    { display: false },
            },
            y: {
              stacked: true,
              ticks:   { color: '#888', callback: (v) => `${v} kg` },
              grid:    { color: 'rgba(255,255,255,0.05)' },
              beginAtZero: true,
            },
          },
        },
      })
    }

    watch(() => [props.data, props.groupBy], render, { deep: true })
    onMounted(render)
    onUnmounted(() => { if (chartInstance) { chartInstance.destroy(); chartInstance = null } })
  },

  template: `
    <div class="chart-container">
      <canvas :id="canvasId" />
    </div>
  `
}
```

## `StatsDashboardView.js`

```javascript
import { reactive, onMounted, inject } from '/vendor/vue.esm-browser.prod.js'
import { stats as statsApi } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader     from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner    from '../common/ErrorBanner.js'
import StreakCard      from './StreakCard.js'
import VolumeChart     from './VolumeChart.js'
import PRList          from './PRList.js'

export default {
  name: 'StatsDashboardView',
  components: { PageHeader, LoadingSpinner, ErrorBanner, StreakCard, VolumeChart, PRList },

  setup() {
    const router = inject('router')
    const state = reactive({
      loading:  false,
      error:    null,
      overview: null,
      volume:   [],
      records:  [],
      volumeGroupBy: 'week',
    })

    async function load() {
      state.loading = true; state.error = null
      try {
        const [overview, volumeData, recordsData] = await Promise.all([
          statsApi.overview(),
          statsApi.volume({ weeks: 12, group_by: state.volumeGroupBy }),
          statsApi.records(),
        ])
        state.overview = overview
        state.volume   = volumeData.data
        state.records  = recordsData.records
      } catch (err) {
        state.error = err.message
      } finally {
        state.loading = false
      }
    }

    async function toggleGroupBy() {
      state.volumeGroupBy = state.volumeGroupBy === 'week' ? 'muscle' : 'week'
      try {
        const data = await statsApi.volume({ weeks: 12, group_by: state.volumeGroupBy })
        state.volume = data.data
      } catch (_) {}
    }

    onMounted(load)
    return { state, load, toggleGroupBy, router, t }
  },

  template: `
    <div class="view">
      <PageHeader :title="t('stats.title')" />

      <div class="view__content">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner v-else-if="state.error" :message="state.error" :retry="load" />

        <template v-else-if="state.overview">
          <!-- Streak -->
          <StreakCard
            :current-streak="state.overview.current_week_streak"
            :longest-streak="state.overview.longest_week_streak"
            :total-sessions="state.overview.total_sessions"
            :total-weeks="state.overview.total_weeks_trained"
          />

          <!-- Volume -->
          <section style="padding:0 1rem">
            <div style="display:flex;align-items:center;justify-content:space-between;margin-bottom:0.5rem">
              <h3 class="detail-section__title" style="margin:0">{{ t('stats.volume_chart') }}</h3>
              <button class="btn btn--secondary"
                style="padding:0.25rem 0.6rem;font-size:0.75rem"
                @click="toggleGroupBy()">
                {{ state.volumeGroupBy === 'week' ? 'Par muscle' : 'Par semaine' }}
              </button>
            </div>
            <div v-if="!state.volume.length"
              style="color:var(--color-text-muted);font-size:0.85rem;padding:1rem 0">
              {{ t('stats.no_data') }}
            </div>
            <VolumeChart
              v-else
              :data="state.volume"
              :group-by="state.volumeGroupBy"
              canvas-id="volume-chart-main"
            />
          </section>

          <!-- Records -->
          <section style="padding:0 1rem;margin-top:1.5rem">
            <div style="display:flex;align-items:center;justify-content:space-between;margin-bottom:0.5rem">
              <h3 class="detail-section__title" style="margin:0">{{ t('stats.personal_records') }}</h3>
            </div>
            <PRList :records="state.records" :limit="5" />
            <button v-if="state.records.length > 5"
              class="btn btn--secondary btn--full" style="margin-top:0.75rem"
              @click="router.push('#/stats/records')">
              {{ t('stats.see_all_records') }}
            </button>
          </section>

          <!-- Liens rapides -->
          <section style="padding:1rem;margin-top:1rem;display:flex;flex-direction:column;gap:0.5rem">
            <button class="btn btn--secondary btn--full"
              @click="router.push('#/stats/bodyweight')">
              ⚖️ {{ t('stats.bodyweight_link') }}
            </button>
          </section>
        </template>
      </div>
    </div>
  `
}
```

## `ExerciseProgressView.js`

```javascript
import { reactive, onMounted, onUnmounted, inject, nextTick }
  from '/vendor/vue.esm-browser.prod.js'
import { stats as statsApi, exercises as exercisesApi } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader     from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner    from '../common/ErrorBanner.js'
import EmptyState     from '../common/EmptyState.js'

const PERIODS = [
  { key: 30,  labelKey: 'stats.days_30' },
  { key: 90,  labelKey: 'stats.days_90' },
  { key: 180, labelKey: 'stats.days_180' },
  { key: 365, labelKey: 'stats.days_365' },
]

const FORMULAS = ['epley', 'brzycki', 'lander', 'oconner', 'lombardi']

export default {
  name: 'ExerciseProgressView',
  props: { id: { type: String, required: true } },
  components: { PageHeader, LoadingSpinner, ErrorBanner, EmptyState },

  setup(props) {
    const store = inject('store')
    let chartInstance = null

    const state = reactive({
      loading:    false,
      error:      null,
      exercise:   null,
      data:       [],
      days:       180,
      formula:    store.getSetting('rm_formula', 'epley'),
    })

    async function load() {
      state.loading = true; state.error = null
      try {
        const [exercise, progress] = await Promise.all([
          exercisesApi.get(props.id),
          statsApi.exercise(props.id, { days: state.days }),
        ])
        state.exercise = exercise
        state.data     = progress.data
        state.formula  = progress.formula
        await nextTick()
        renderChart()
      } catch (err) {
        state.error = err.message
      } finally {
        state.loading = false
      }
    }

    async function changePeriod(days) {
      state.days = days
      try {
        const progress = await statsApi.exercise(props.id, { days })
        state.data    = progress.data
        state.formula = progress.formula
        await nextTick()
        renderChart()
      } catch (_) {}
    }

    async function changeFormula(formula) {
      state.formula = formula
      // Recalculer côté client (même données, formule différente)
      // Pour une vraie recalculation, recharger depuis l'API
      await load()
    }

    function renderChart() {
      if (!window.Chart) return
      const canvas = document.getElementById('exercise-progress-chart')
      if (!canvas || !state.data.length) return
      if (chartInstance) { chartInstance.destroy(); chartInstance = null }

      chartInstance = new window.Chart(canvas, {
        type: 'line',
        data: {
          datasets: [{
            label:           t('stats.rm_estimated'),
            data:            state.data.map(d => ({ x: d.date, y: d.estimated_1rm_kg })),
            borderColor:     '#e67e22',
            backgroundColor: 'rgba(230,126,34,0.1)',
            pointRadius:     4,
            pointHoverRadius: 6,
            tension:         0.2,
            fill:            true,
          }],
        },
        options: {
          responsive: true, maintainAspectRatio: false,
          plugins: {
            legend: { display: false },
            tooltip: {
              callbacks: {
                title: (items) => items[0]?.label || '',
                label: (ctx) =>
                  `${t('stats.rm_estimated')} : ${ctx.parsed.y} kg`,
              }
            },
          },
          scales: {
            x: {
              type:  'time',
              time:  { unit: 'week', tooltipFormat: 'dd/MM/yyyy' },
              ticks: { color: '#888', maxTicksLimit: 8 },
              grid:  { color: 'rgba(255,255,255,0.05)' },
            },
            y: {
              ticks: { color: '#888', callback: (v) => `${v} kg` },
              grid:  { color: 'rgba(255,255,255,0.05)' },
            },
          },
        },
      })
    }

    onMounted(load)
    onUnmounted(() => {
      if (chartInstance) { chartInstance.destroy(); chartInstance = null }
    })

    return { state, load, changePeriod, changeFormula, PERIODS, FORMULAS, t }
  },

  template: `
    <div class="view">
      <PageHeader
        :title="state.exercise?.name_fr || t('stats.exercise_progress')"
        :show-back="true"
      />
      <div class="view__content" style="padding:0 1rem">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner v-else-if="state.error" :message="state.error" :retry="load" />

        <template v-else>
          <!-- Sélecteur de période -->
          <div class="period-selector">
            <button
              v-for="p in PERIODS" :key="p.key"
              class="period-btn"
              :class="{ 'period-btn--active': state.days === p.key }"
              @click="changePeriod(p.key)">
              {{ t(p.labelKey) }}
            </button>
          </div>

          <!-- Sélecteur de formule -->
          <div class="formula-selector">
            <span class="form-label" style="margin-right:0.5rem">
              {{ t('stats.formula_label') }} :
            </span>
            <select class="form-select" style="width:auto;display:inline;font-size:0.85rem"
              :value="state.formula"
              @change="changeFormula($event.target.value)">
              <option v-for="f in FORMULAS" :key="f" :value="f">{{ f }}</option>
            </select>
          </div>

          <!-- Graphique -->
          <EmptyState v-if="!state.data.length"
            icon="📈" :message="t('stats.no_data')" />

          <div v-else class="progress-chart-container">
            <canvas id="exercise-progress-chart" />
          </div>

          <!-- Tableau des données récentes -->
          <div v-if="state.data.length" class="progress-table">
            <div class="detail-section__title" style="margin-bottom:0.5rem">
              {{ t('stats.rm_estimated') }} — dernières séances
            </div>
            <div v-for="d in [...state.data].reverse().slice(0, 10)" :key="d.date"
              class="progress-row">
              <span class="progress-row__date">{{ d.date }}</span>
              <span class="progress-row__weight">{{ d.weight_kg }} kg × {{ d.reps }}</span>
              <span class="progress-row__rm">{{ d.estimated_1rm_kg }} kg</span>
            </div>
          </div>
        </template>
      </div>
    </div>
  `
}
```

  `{ deep: true }` pour se re-rendre automatiquement quand les données changent.

  `type: 'time'` par `type: 'category'` dans les deux cas (supprimer le

```javascript
import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import request from 'supertest'
import { app } from '../../server.js'

describe('T4.3 — Stats endpoints supplémentaires', () => {
  it('GET /stats/overview inclut personal_records_count', async () => {
    const res = await request(app).get('/api/v1/stats/overview')
    assert.equal(res.status, 200)
    assert.ok(typeof res.body.personal_records_count === 'number')
    assert.ok(typeof res.body.total_volume_kg === 'number')
    assert.ok(typeof res.body.active_program_name !== 'undefined')
  })

  it('GET /stats/volume?weeks=12 retourne au max 12 semaines de données', async () => {
    const res = await request(app).get('/api/v1/stats/volume?weeks=12')
    assert.equal(res.status, 200)
    assert.ok(Array.isArray(res.body.data))
    // Chaque entrée a period et volume_kg
    if (res.body.data.length > 0) {
      assert.ok('period'    in res.body.data[0])
      assert.ok('volume_kg' in res.body.data[0])
    }
  })

  it('GET /stats/exercises/:id/progress retourne data avec estimated_1rm_kg', async () => {
    const res = await request(app)
      .get('/api/v1/stats/exercises/bench_press_barbell/progress?days=365')
    assert.equal(res.status, 200)
    if (res.body.data.length > 0) {
      assert.ok('estimated_1rm_kg' in res.body.data[0])
      assert.ok('date'             in res.body.data[0])
      assert.ok('weight_kg'        in res.body.data[0])
    }
  })

  it('GET /stats/personal-records?exercise_id filtre correctement', async () => {
    const res = await request(app)
      .get('/api/v1/stats/personal-records?exercise_id=bench_press_barbell')
    assert.equal(res.status, 200)
    assert.ok(Array.isArray(res.body.records))
    // Si des records existent, ils appartiennent tous à bench_press_barbell
    for (const r of res.body.records) {
      assert.equal(r.exercise_id, 'bench_press_barbell')
    }
  })
})
```
