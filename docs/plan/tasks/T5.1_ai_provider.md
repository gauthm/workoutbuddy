// ── Constantes ────────────────────────────────────────────────
const AI_TIMEOUT_MS = 30_000  // 30 secondes

// ── Normalisation de l'URL de base ───────────────────────────
/**
 * Normalise une base URL pour l'API OpenAI-compatible.
 * Garantit que l'URL se termine par "/v1" sans slash final.
 *
 * Cas gérés :
 *   "https://api.openai.com"        → "https://api.openai.com/v1"
 *   "https://api.openai.com/"       → "https://api.openai.com/v1"
 *   "https://api.openai.com/v1"     → "https://api.openai.com/v1"
 *   "https://api.openai.com/v1/"    → "https://api.openai.com/v1"
 *   "http://localhost:11434"        → "http://localhost:11434/v1"
 *   "https://api.groq.com/openai"   → "https://api.groq.com/openai/v1"
 *   "https://api.groq.com/openai/v1" → "https://api.groq.com/openai/v1"
 */
export function normalizeBaseUrl(url) {
  if (!url || typeof url !== 'string') return ''
  // Supprimer les slashes finaux
  let u = url.trim().replace(/\/+$/, '')
  // Ajouter /v1 si absent
  if (!u.endsWith('/v1')) {
    u = u + '/v1'
  }
  return u
}

// ── Lecture asynchrone d'un ReadableStream ligne par ligne ────
async function* readLines(body) {
  const reader  = body.getReader()
  const decoder = new TextDecoder()
  let   buffer  = ''

  while (true) {
    const { done, value } = await reader.read()
    if (done) {
      if (buffer.trim()) yield buffer
      break
    }
    buffer += decoder.decode(value, { stream: true })
    const lines = buffer.split('\n')
    buffer = lines.pop()  // garder la ligne incomplète
    for (const line of lines) yield line
  }
}

// ── Streaming Chat Completions ────────────────────────────────
/**
 * Générateur asynchrone : émet des événements SSE parsés.
 *
 * Types d'événements émis :
 *   { type: 'token',  content: string }
 *   { type: 'done',   usage: object|null }
 *   { type: 'error',  message: string }
 *
 * @param {object[]} messages - [{ role: 'user'|'assistant'|'system', content: string }]
 * @param {object}   aiSettings - { base_url, api_key, model }
 */
export async function* streamChat(messages, aiSettings) {
  const { base_url, api_key, model } = aiSettings

  if (!base_url || !model) {
    yield { type: 'error', message: 'Provider IA non configuré. Allez dans Paramètres > Coach IA.' }
    return
  }

  const url        = normalizeBaseUrl(base_url) + '/chat/completions'
  const controller = new AbortController()
  const timeout    = setTimeout(() => controller.abort(), AI_TIMEOUT_MS)

  try {
    const res = await fetch(url, {
      method:  'POST',
      headers: {
        'Content-Type':  'application/json',
        'Authorization': `Bearer ${api_key || 'no-key'}`,
      },
      body: JSON.stringify({
        model,
        messages,
        stream:     true,
        max_tokens: 2000,
      }),
      signal: controller.signal,
    })

    if (!res.ok) {
      const body = await res.text().catch(() => '')
      const msg  = errorMessageForStatus(res.status, body)
      yield { type: 'error', message: msg }
      return
    }

    let usage = null

    for await (const line of readLines(res.body)) {
      if (!line.startsWith('data: ')) continue
      const raw = line.slice(6).trim()
      if (raw === '[DONE]') {
        yield { type: 'done', usage }
        return
      }
      if (!raw) continue

      let parsed
      try {
        parsed = JSON.parse(raw)
      } catch {
        // Ligne non-JSON — ignorer silencieusement
        continue
      }

      // Extraire le token (OpenAI delta OU Ollama message)
      const token =
        parsed.choices?.[0]?.delta?.content ??
        parsed.choices?.[0]?.message?.content ??
        ''

      if (token) yield { type: 'token', content: token }

      // Usage (présent dans le dernier chunk de certains providers)
      if (parsed.usage) usage = parsed.usage
    }

    yield { type: 'done', usage }

  } catch (err) {
    if (err.name === 'AbortError') {
      yield { type: 'error', message: 'Délai dépassé (30 s). Vérifiez la connexion au provider IA.' }
    } else {
      yield { type: 'error', message: `Erreur réseau : ${err.message}` }
    }
  } finally {
    clearTimeout(timeout)
  }
}

// ── Lister les modèles disponibles ────────────────────────────
/**
 * @param {object} aiSettings - { base_url, api_key }
 * @returns {{ models: { id: string, name: string }[] }}
 * @throws Error si la requête échoue
 */
export async function listModels(aiSettings) {
  const { base_url, api_key } = aiSettings
  if (!base_url) throw new Error('URL de base non configurée.')

  const url        = normalizeBaseUrl(base_url) + '/models'
  const controller = new AbortController()
  const timeout    = setTimeout(() => controller.abort(), AI_TIMEOUT_MS)

  try {
    const res = await fetch(url, {
      headers: { 'Authorization': `Bearer ${api_key || 'no-key'}` },
      signal:  controller.signal,
    })

    if (!res.ok) {
      const body = await res.text().catch(() => '')
      throw new Error(errorMessageForStatus(res.status, body))
    }

    const data = await res.json()
    // Format OpenAI : { data: [{ id, ... }] }
    const items = data.data || data.models || []
    return {
      models: items.map(m => ({
        id:   m.id   || m.name || String(m),
        name: m.name || m.id   || String(m),
      }))
    }
  } catch (err) {
    if (err.name === 'AbortError') throw new Error('Délai dépassé.')
    throw err
  } finally {
    clearTimeout(timeout)
  }
}

// ── Test de connexion ─────────────────────────────────────────
/**
 * @param {object} aiSettings - { base_url, api_key, model }
 * @returns {{ success: true, latency_ms: number, model: string }}
 * @throws Error si la connexion échoue
 */
export async function testConnection(aiSettings) {
  const start = Date.now()

  const messages = [
    { role: 'user', content: 'Reply with exactly the word: OK' }
  ]

  let collected = ''
  for await (const event of streamChat(messages, aiSettings)) {
    if (event.type === 'token') collected += event.content
    if (event.type === 'error') throw new Error(event.message)
    if (event.type === 'done')  break
  }

  return {
    success:    true,
    latency_ms: Date.now() - start,
    model:      aiSettings.model,
  }
}

// ── Helper messages d'erreur HTTP ─────────────────────────────
function errorMessageForStatus(status, body) {
  const preview = body ? body.slice(0, 200) : ''
  switch (status) {
    case 401: return 'Clé API invalide ou expirée. Vérifiez dans Paramètres > Coach IA.'
    case 403: return 'Accès refusé. Vérifiez les droits de la clé API.'
    case 404: return `Endpoint introuvable (${status}). Vérifiez l'URL de base (/v1 manquant ?).`
    case 429: return 'Limite de requêtes atteinte. Attendez quelques secondes.'
    case 500: return `Erreur serveur du provider IA (${status}).`
    case 503: return `Provider IA indisponible (${status}).`
    default:  return `Erreur HTTP ${status}${preview ? ' : ' + preview : ''}.`
  }
}

import express from 'express'
import { getDb } from '../db/database.js'
import { testConnection, listModels } from '../services/aiProvider.js'

const router = express.Router()
const db     = getDb()

// ── Clés autorisées et leurs types/plages ─────────────────────
// Format : { type, min?, max?, enum? }
const SETTINGS_SCHEMA = {
  unit:                               { type: 'enum',    enum: ['kg', 'lb'] },
  rm_formula:                         { type: 'enum',    enum: ['epley','brzycki','lander','oconner','lombardi'] },
  default_rest_seconds:               { type: 'number',  min: 10,  max: 600 },

  ai_model:                           { type: 'string' },

  deload_load_reduction_pct:          { type: 'number',  min: 10, max: 40 },
  deload_volume_reduction_pct:        { type: 'number',  min: 20, max: 60 },
  deload_min_sets:                    { type: 'number',  min: 1, max: 4 },
  equipment_advice_cooldown_days:     { type: 'number',  min: 1, max: 90 },

  show_rpe_field:                     { type: 'boolean' },
  show_rir_field:                     { type: 'boolean' },
  equipment_advice_max_load_sessions: { type: 'number',  min: 1, max: 5 },
  equipment_advice_min_increment_pct: { type: 'number',  min: 1, max: 20 },
  equipment_advice_plateau_sessions:  { type: 'number',  min: 3, max: 10 },
  warmup_sets_count:                  { type: 'number',  min: 0, max: 3 },
  auto_next_exercise:                 { type: 'boolean' },
}

// Clés à exclure de la réponse GET (sécurité)
const HIDDEN_KEYS = new Set(['ai_api_key'])

function getAllSettings(db) {
  const rows = db.prepare('SELECT key, value FROM settings').all()
  const result = {}
  for (const row of rows) {
    if (HIDDEN_KEYS.has(row.key)) continue
    result[row.key] = JSON.parse(row.value)
  }
  // Indicateur clé API configurée
  const apiKeyRow = db.prepare("SELECT value FROM settings WHERE key = 'ai_api_key'").get()
  result.ai_api_key_set = apiKeyRow
    ? JSON.parse(apiKeyRow.value) !== ''
    : false
  return result
}

function validateSettingValue(key, value) {
  const schema = SETTINGS_SCHEMA[key]
  if (!schema) return { valid: false, error: `Clé inconnue : ${key}` }

  if (schema.type === 'boolean') {
    if (typeof value !== 'boolean') {
      return { valid: false, error: `${key} doit être un booléen.` }
    }
  } else if (schema.type === 'number') {
    const n = Number(value)
    if (isNaN(n)) return { valid: false, error: `${key} doit être un nombre.` }
    if (schema.min !== undefined && n < schema.min) {
      return { valid: false, error: `${key} doit être ≥ ${schema.min}.` }
    }
    if (schema.max !== undefined && n > schema.max) {
      return { valid: false, error: `${key} doit être ≤ ${schema.max}.` }
    }
  } else if (schema.type === 'enum') {
    if (!schema.enum.includes(value)) {
      return { valid: false, error: `${key} doit être l'une de : ${schema.enum.join(', ')}.` }
    }
  } else if (schema.type === 'string') {
    if (typeof value !== 'string') {
      return { valid: false, error: `${key} doit être une chaîne.` }
    }
  }
  return { valid: true }
}

// ── GET /settings ─────────────────────────────────────────────
router.get('/', (req, res) => {
  try {
    res.json(getAllSettings(db))
  } catch (err) {
    console.error('[GET /settings]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── PUT /settings ─────────────────────────────────────────────
router.put('/', (req, res) => {
  try {
    const updates = req.body
    if (!updates || typeof updates !== 'object') {
      return res.status(400).json({ error: 'INVALID_BODY' })
    }

    const errors = []
    const valid  = []

    for (const [key, value] of Object.entries(updates)) {
      const result = validateSettingValue(key, value)
      if (!result.valid) { errors.push(result.error); continue }
      valid.push({ key, value })
    }

    if (errors.length > 0) {
      return res.status(400).json({ error: 'VALIDATION_ERRORS', details: errors })
    }

    const stmt = db.prepare(
      "INSERT INTO settings (key, value, updated_at) VALUES (?, ?, datetime('now')) " +
      "ON CONFLICT(key) DO UPDATE SET value = excluded.value, updated_at = excluded.updated_at"
    )
    db.transaction(() => {
      for (const { key, value } of valid) {
        stmt.run(key, JSON.stringify(value))
      }
    })()

    // Vérifier si réseau a changé → indiquer redémarrage requis
    const networkKeys = new Set(['network_host', 'network_port'])
    const restartRequired = valid.some(v => networkKeys.has(v.key))

    const response = getAllSettings(db)
    if (restartRequired) response.restart_required = true

    res.json(response)
  } catch (err) {
    console.error('[PUT /settings]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── POST /settings/test-ai ────────────────────────────────────
router.post('/test-ai', async (req, res) => {
  try {
    const aiSettings = getAiSettings(db)
    if (!aiSettings.base_url || !aiSettings.model) {
      return res.status(400).json({
        success: false,
        error: 'Provider IA non configuré (URL ou modèle manquant).',
      })
    }

    const result = await testConnection(aiSettings)
    res.json(result)
  } catch (err) {
    res.status(400).json({ success: false, error: err.message })
  }
})

// ── GET /settings/ai-models ───────────────────────────────────
router.get('/ai-models', async (req, res) => {
  try {
    const aiSettings = getAiSettings(db)
    if (!aiSettings.base_url) {
      return res.status(400).json({ error: 'URL de base non configurée.' })
    }

    const result = await listModels(aiSettings)
    res.json(result)
  } catch (err) {
    res.status(400).json({ error: err.message })
  }
})

// ── Helper : lire les settings IA depuis la DB ────────────────
export function getAiSettings(db) {
  const keys = ['ai_base_url', 'ai_model', 'ai_api_key']
  const result = {}
  for (const key of keys) {
    const row = db.prepare('SELECT value FROM settings WHERE key = ?').get(key)
    result[key.replace('ai_', '')] = row ? JSON.parse(row.value) : ''
  }
  return result  // { base_url, model, api_key }
}

export default router

## Étapes d'implémentation

1. Créer `src/services/aiProvider.js`.
2. Créer `src/routes/settings.js`.
3. Modifier `server.js` : import + `app.use`.

**Points d'attention :**
- `GET /settings` ne retourne **jamais** `ai_api_key`. Il retourne
  `ai_api_key_set: true/false`.
- `PUT /settings` avec `ai_api_key: ""` efface la clé (comportement voulu).
- `readLines()` utilise `body.getReader()` — disponible sur les réponses
  `fetch()` en Node.js 20. Ne pas utiliser `response.text()` pour le
  streaming.
- `streamChat` est un générateur asynchrone (`async function*`). Il doit
  être consommé avec `for await...of` dans la route SSE.
- Le `timeout` `AbortController` doit être dans un `try/finally` pour
  garantir que `clearTimeout` est appelé même si le générateur est
  interrompu.
- `SETTINGS_SCHEMA` couvre exactement les 31 clés définies en migration
  002. Tout autre clé retourne `INVALID_BODY`.

import { describe, it } from 'node:test'
import assert from 'node:assert/strict'

describe('normalizeBaseUrl', () => {
  it('URL sans /v1 → ajoute /v1', () => {
    assert.equal(normalizeBaseUrl('https://api.openai.com'), 'https://api.openai.com/v1')

    assert.equal(normalizeBaseUrl('https://api.openai.com/v1'), 'https://api.openai.com/v1')
  })
  it('URL avec /v1/ → supprime slash final', () => {
    assert.equal(normalizeBaseUrl('https://api.openai.com/v1/'), 'https://api.openai.com/v1')
  })
  it('localhost Ollama → ajoute /v1', () => {
    assert.equal(normalizeBaseUrl('http://localhost:11434'), 'http://localhost:11434/v1')
  })
  it('Groq avec chemin intermédiaire → ajoute /v1 à la fin', () => {
    assert.equal(normalizeBaseUrl('https://api.groq.com/openai'), 'https://api.groq.com/openai/v1')
  })
  it('chaîne vide → chaîne vide', () => {
    assert.equal(normalizeBaseUrl(''), '')
