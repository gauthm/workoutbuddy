## `src/services/equipmentAdvisor.js` — implémentation complète

```javascript
import { calculateAchievableLoads } from './equipmentCalc.js'
import { detectPlateau }            from './plateauDetector.js'

// ── Types de signaux ──────────────────────────────────────────
// 'max_load_reached'       — charge max atteinte, reps dépassées
// 'min_increment_too_large'— incrément trop grand vs poids courant (+ plateau)
// 'no_exercise_available'  — aucun exercice réalisable pour un muscle
// 'no_rotation_possible'   — rotation impossible (matériel insuffisant)
// 'persistent_plateau'     — plateau long malgré incrément correct

// Tous les groupes musculaires suivis
const ALL_MUSCLES = [
  'chest','back','shoulders','biceps','triceps',
  'quads','hamstrings','glutes','calves','core','traps','forearms',
]

/**
 * Lance tous les contrôles de signal pour le programme actif.
 * Insère les nouveaux signaux dans equipment_advice (status='active').
 * Respecte le cooldown par signal/exercice.
 *
 * @param {object} db       - instance better-sqlite3
 * @param {object} settings - settings parsés { clé: valeur }
 * @returns {number} nombre de nouveaux conseils créés
 */
export function runAllSignalChecks(db, settings) {
  const activeProgram = db.prepare(
    "SELECT * FROM program WHERE is_active = 1 LIMIT 1"
  ).get()
  if (!activeProgram) return 0

  const equipmentItems    = db.prepare('SELECT * FROM equipment_item').all()
  const activeAdvice      = db.prepare(
    "SELECT * FROM equipment_advice WHERE status = 'active'"
  ).all()

  const cooldownDays      = settings.equipment_advice_cooldown_days      || 14
  const maxLoadSessions   = settings.equipment_advice_max_load_sessions  || 3
  const minIncrPct        = settings.equipment_advice_min_increment_pct  || 5
  const plateauSessions   = settings.equipment_advice_plateau_sessions   || 6
  const detectionSessions = settings.plateau_detection_sessions          || 4

  // Template exercises actifs
  const templateExercises = db.prepare(`
    SELECT te.*, e.primary_muscles, e.equipment_required, e.is_unilateral, e.name_fr
    FROM template_exercise te
    JOIN workout_template  wt ON te.workout_template_id = wt.id
    JOIN exercise           e ON te.exercise_id = e.id
    WHERE wt.program_id = ?
  `).all(activeProgram.id).map(te => ({
    ...te,
    primary_muscles:    JSON.parse(te.primary_muscles    || '[]'),
    equipment_required: JSON.parse(te.equipment_required || '[]'),
  }))

  let created = 0

  // ── Signal 1 : Charge maximale atteinte ──────────────────────
  for (const te of templateExercises) {
    if (isOnCooldown('max_load_reached', te.exercise_id, null, activeAdvice, cooldownDays)) continue

    const loadsResult = calculateAchievableLoads(equipmentItems, {
      equipment_required: te.equipment_required,
      is_unilateral:      te.is_unilateral,
    })
    if (loadsResult.loads_kg.length === 0) continue
    const maxLoad = loadsResult.max_load_kg

    // Récupérer les N dernières séances pour cet exercice
    const recentSets = db.prepare(`
      SELECT sl.weight_kg, sl.reps, te2.target_reps_max
      FROM set_log sl
      JOIN session_exercise se ON sl.session_exercise_id = se.id
      JOIN session s           ON se.session_id = s.id
      JOIN template_exercise te2 ON se.template_exercise_id = te2.id
      WHERE se.exercise_id    = ?
        AND s.program_id      = ?
        AND s.status          = 'completed'
        AND sl.is_warmup      = 0
        AND sl.is_completed   = 1
        AND te2.id            = ?
      ORDER BY s.completed_at DESC
      LIMIT ?
    `).all(te.exercise_id, activeProgram.id, te.id, maxLoadSessions)

    if (recentSets.length < maxLoadSessions) continue

    const allAtMax = recentSets.every(s => s.weight_kg >= maxLoad * 0.99)
    const allExceeded = recentSets.every(s => s.reps >= (s.target_reps_max || te.target_reps_max))

    if (allAtMax && allExceeded) {
      const avgReps = recentSets.reduce((sum, s) => sum + s.reps, 0) / recentSets.length
      const signal = {
        type:           'max_load_reached',
        exercise_id:    te.exercise_id,
        max_load_kg:    maxLoad,
        sessions_checked: maxLoadSessions,
        target_reps_max: te.target_reps_max,
        actual_reps_avg: Math.round(avgReps * 10) / 10,
        exercise_name_fr: te.name_fr,
      }
      insertAdvice(db, 'max_load_reached', te.exercise_id, null, null, signal)
      created++
    }
  }

  // ── Signal 2 : Incrément minimal trop grand ───────────────────
  for (const te of templateExercises) {
    if (isOnCooldown('min_increment_too_large', te.exercise_id, null, activeAdvice, cooldownDays)) continue

    const state = db.prepare(
      'SELECT * FROM progression_state WHERE template_exercise_id = ?'
    ).get(te.id)
    if (!state || state.current_weight_kg <= 0) continue

    const loadsResult = calculateAchievableLoads(equipmentItems, {
      equipment_required: te.equipment_required,
      is_unilateral:      te.is_unilateral,
    })
    if (loadsResult.min_increment_kg <= 0) continue

    const pct = (loadsResult.min_increment_kg / state.current_weight_kg) * 100

    if (pct > minIncrPct) {
      // Vérifier aussi qu'il y a un plateau
      const plateau = detectPlateau(te.exercise_id, activeProgram.id, detectionSessions, db)
      if (!plateau.is_plateau) continue

      if (isOnCooldown('min_increment_too_large', te.exercise_id, null, activeAdvice, cooldownDays)) continue

      const signal = {
        type:                'min_increment_too_large',
        exercise_id:         te.exercise_id,
        min_increment_kg:    loadsResult.min_increment_kg,
        current_weight_kg:   state.current_weight_kg,
        pct:                 Math.round(pct * 10) / 10,
        exercise_name_fr:    te.name_fr,
      }
      insertAdvice(db, 'min_increment_too_large', te.exercise_id, null, null, signal)
      created++
    }
  }

  // ── Signal 3 : Aucun exercice disponible pour un muscle ───────
  for (const muscle of ALL_MUSCLES) {
    if (isOnCooldown('no_exercise_available', null, muscle, activeAdvice, cooldownDays)) continue

    const candidates = db.prepare(
      "SELECT * FROM exercise WHERE primary_muscles LIKE ? AND category = 'strength'"
    ).all(`%"${muscle}"%`)

    const feasible = candidates.filter(ex => {
      const req = JSON.parse(ex.equipment_required || '[]')
      if (req.length === 0) return true
      const res = calculateAchievableLoads(equipmentItems, {
        equipment_required: req,
        is_unilateral: ex.is_unilateral,
      })
      return res.loads_kg.length > 0
    })

    if (feasible.length === 0) {
      const signal = {
        type:         'no_exercise_available',
        muscle_group: muscle,
      }
      insertAdvice(db, 'no_exercise_available', null, muscle, null, signal)
      created++
    }
  }

  // ── Signal 4 : Rotation impossible ───────────────────────────
  if (activeProgram.use_exercise_rotation) {
    for (const te of templateExercises) {
      if (!te.rotation_group) continue
      if (isOnCooldown('no_rotation_possible', te.exercise_id, null, activeAdvice, cooldownDays)) continue

      // Trouver les candidats de rotation
      const baseEx = db.prepare('SELECT * FROM exercise WHERE id = ?').get(te.exercise_id)
      if (!baseEx) continue
      const basePrimary = JSON.parse(baseEx.primary_muscles || '[]')

      const candidates = db.prepare(
        'SELECT * FROM exercise WHERE movement_pattern = ? AND id != ?'
      ).all(baseEx.movement_pattern, te.exercise_id).filter(ex => {
        const muscles = JSON.parse(ex.primary_muscles || '[]')
        const hasOverlap = muscles.some(m => basePrimary.includes(m))
        const req = JSON.parse(ex.equipment_required || '[]')
        const res = calculateAchievableLoads(equipmentItems, {
          equipment_required: req,
          is_unilateral: ex.is_unilateral,
        })
        return hasOverlap && res.loads_kg.length > 0
      })

      if (candidates.length === 0) {
        const signal = {
          type:             'no_rotation_possible',
          exercise_id:      te.exercise_id,
          rotation_group:   te.rotation_group,
          exercise_name_fr: te.name_fr,
        }
        insertAdvice(db, 'no_rotation_possible', te.exercise_id, null, null, signal)
        created++
      }
    }
  }

  // ── Signal 5 : Plateau persistant ────────────────────────────
  for (const te of templateExercises) {
    if (isOnCooldown('persistent_plateau', te.exercise_id, null, activeAdvice, cooldownDays)) continue

    const plateau = detectPlateau(te.exercise_id, activeProgram.id, plateauSessions, db)
    if (!plateau.is_plateau) continue

    // Ne signaler que si l'incrément est aussi problématique
    const state = db.prepare(
      'SELECT * FROM progression_state WHERE template_exercise_id = ?'
    ).get(te.id)
    if (!state || state.current_weight_kg <= 0) continue

    const loadsResult = calculateAchievableLoads(equipmentItems, {
      equipment_required: te.equipment_required,
      is_unilateral:      te.is_unilateral,
    })
    const pct = loadsResult.min_increment_kg > 0
      ? (loadsResult.min_increment_kg / state.current_weight_kg) * 100
      : 0

    if (pct <= minIncrPct) continue  // incrément acceptable → pas un pb de matériel

    const signal = {
      type:             'persistent_plateau',
      exercise_id:      te.exercise_id,
      sessions_checked: plateauSessions,
      min_increment_kg: loadsResult.min_increment_kg,
      current_weight_kg: state.current_weight_kg,
      delta_pct:        plateau.delta_pct,
      exercise_name_fr: te.name_fr,
    }
    insertAdvice(db, 'persistent_plateau', te.exercise_id, null, null, signal)
    created++
  }

  return created
}

// ── Helpers ───────────────────────────────────────────────────

/**
 * Vérifie si un signal est en cooldown.
 * Un signal est en cooldown si un advice du même type + exercise_id + muscle_group
 * existe avec status='active' ET last_generated_at < maintenant - cooldownDays.
 */
function isOnCooldown(signalType, exerciseId, muscleGroup, activeAdvice, cooldownDays) {
  for (const a of activeAdvice) {
    if (a.signal_type  !== signalType)  continue
    if (exerciseId     !== null && a.exercise_id  !== exerciseId)  continue
    if (muscleGroup    !== null && a.muscle_group !== muscleGroup) continue

    const generated = new Date(a.last_generated_at)
    const now       = new Date()
    const diffDays  = (now - generated) / (1000 * 60 * 60 * 24)
    if (diffDays < cooldownDays) return true
  }
  return false
}

/**
 * Insère un nouveau conseil en DB.
 * Utilise INSERT OR IGNORE sur l'index unique (signal_type, exercise_id, muscle_group)
 * pour éviter les doublons si le signal existe déjà.
 */
function insertAdvice(db, signalType, exerciseId, muscleGroup, movementPattern, signalData) {
  try {
    db.prepare(`
      INSERT OR IGNORE INTO equipment_advice
        (signal_type, exercise_id, muscle_group, movement_pattern, signal_data,
         status, last_generated_at)
      VALUES (?, ?, ?, ?, ?, 'active', datetime('now'))
    `).run(
      signalType,
      exerciseId    || null,
      muscleGroup   || null,
      movementPattern || null,
      JSON.stringify(signalData)
    )
  } catch (err) {
    console.warn('[equipmentAdvisor] insertAdvice error:', err.message)
  }
}
```

## Exemples chiffrés de vérification

### Signal 1 — max_load_reached
```
Inventaire : barre 1 kg + plaques 1/2/5 kg → max_load = 23 kg
template_exercise : target_reps_max = 12
3 dernières séances : weight_kg=23, reps=13 (> 12) ✓

allAtMax    = 23 ≥ 23 * 0.99 = 22.77 → true ✓
allExceeded = 13 ≥ 12 → true ✓
→ Signal créé : max_load_reached ✅
```

### Signal 2 — min_increment_too_large
```
min_increment_kg = 2.5 kg
current_weight_kg = 30 kg
pct = 2.5/30 * 100 = 8.3% > seuil 5% ✓
detectPlateau → is_plateau = true ✓
→ Signal créé : min_increment_too_large ✅
```

### Signal 3 — no_exercise_available
```
muscle = 'chest'
Tous les exercices chest nécessitent barbell/bench
Inventaire : aucune barre ni haltères
feasible = [] → Signal créé : no_exercise_available ✅
```

### Cooldown
```
Signal max_load_reached pour bench_press_barbell créé il y a 5 jours.
cooldown_days = 14 → 5 < 14 → isOnCooldown = true → signal ignoré ✅
```

## Étapes d'implémentation

1. Créer `src/services/equipmentAdvisor.js`.
2. Modifier `src/routes/sessions.js` :
   - Ajouter l'import.
   - Insérer le bloc signaux dans `POST /:id/complete`.

**Points d'attention :**
- Le bloc signaux est dans un `try/catch` non bloquant — une erreur
  dans `runAllSignalChecks` ne doit pas faire échouer la complétion.
- `insertAdvice` utilise `INSERT OR IGNORE` : si l'advice existe déjà
  (même signal_type + exercise_id + muscle_group actif), l'insertion
  est silencieusement ignorée. C'est le comportement voulu.
- Signal 5 (persistent_plateau) ne se déclenche que si l'incrément
  est aussi problématique (`pct > minIncrPct`). Un plateau sur un
  exercice avec un incrément de 1.25 kg n'est pas un problème de
  matériel.
- Le signal 3 (no_exercise_available) est vérifié pour **tous** les
  muscles, pas seulement ceux du programme actif. C'est intentionnel :
  si l'utilisateur veut ajouter un exercice dos mais n'a pas de traction
  bar, le signal s'affiche même si son programme actuel ne cible pas le dos.

### `tests/unit/equipmentAdvisor.test.js`

```javascript
import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import Database from 'better-sqlite3'
import { migrate } from '../../src/db/migrate.js'
import { seedExercises } from '../../src/seed/seed.js'
import { runAllSignalChecks } from '../../src/services/equipmentAdvisor.js'

function createDb() {
  const db = new Database(':memory:')
  db.pragma('foreign_keys = ON')
  migrate(db)
  seedExercises(db)
  return db
}

const DEFAULT_SETTINGS = {
  equipment_advice_cooldown_days:      1,
  equipment_advice_max_load_sessions:  3,
  equipment_advice_min_increment_pct:  5,
  equipment_advice_plateau_sessions:   4,
  plateau_detection_sessions:          4,
}

function createActiveProgram(db) {
  const pRes = db.prepare(`
    INSERT INTO program (name, goal, level, days_per_week, progression_model, is_active)
    VALUES ('Test', 'strength', 'beginner', 3, 'linear', 1)
  `).run()
  return pRes.lastInsertRowid
}

function createTemplate(db, programId) {
  return db.prepare(
    'INSERT INTO workout_template (program_id, name) VALUES (?, ?)'
  ).run(programId, 'A').lastInsertRowid
}

function createTE(db, templateId, exerciseId, repsMax = 12) {
  const teRes = db.prepare(`
    INSERT INTO template_exercise
      (workout_template_id, exercise_id, target_sets, target_reps_min, target_reps_max)
    VALUES (?, ?, 3, 8, ?)
  `).run(templateId, exerciseId, repsMax)
  db.prepare(
    'INSERT INTO progression_state (template_exercise_id, current_weight_kg) VALUES (?, 0)'
  ).run(teRes.lastInsertRowid)
  return teRes.lastInsertRowid
}

function addBarbell(db, weightKg = 1) {
  return db.prepare(
    "INSERT INTO equipment_item (category, name, weight_kg, count) VALUES ('barbell', 'Barre', ?, 1)"
  ).run(weightKg).lastInsertRowid
}

function addPlates(db, weightKg, count) {
  db.prepare(
    "INSERT INTO equipment_item (category, name, weight_kg, count) VALUES ('plate', ?, ?, ?)"
  ).run(`Disque ${weightKg}kg`, weightKg, count)
}

describe('runAllSignalChecks', () => {
  it('aucun programme actif → retourne 0', () => {
    const db = createDb()
    const n  = runAllSignalChecks(db, DEFAULT_SETTINGS)
    assert.equal(n, 0)
  })

  it('signal max_load_reached créé si toutes séances à charge max + reps dépassées', () => {
    const db  = createDb()
    const pid = createActiveProgram(db)
    const tid = createTemplate(db, pid)
    const teId = createTE(db, tid, 'bench_press_barbell', 12)

    // Inventaire minimal : barre 1kg + plaques 1kg×4 → max = 1+2×2=5 kg
    addBarbell(db, 1)
    addPlates(db, 1, 4)

    // Mettre à jour progression_state pour simuler charge max atteinte
    db.prepare('UPDATE progression_state SET current_weight_kg = 5 WHERE template_exercise_id = ?')
      .run(teId)

    // Créer 3 séances avec weight=5kg et reps=13 (> max 12)
    for (let i = 0; i < 3; i++) {
      const date = `2025-0${6 + i}-0${1 + i}`
      const sRes = db.prepare(`
        INSERT INTO session (name, scheduled_date, program_id, status, completed_at)
        VALUES ('S', ?, ?, 'completed', ?)
      `).run(date, pid, date + 'T10:00:00')

      const seRes = db.prepare(`
        INSERT INTO session_exercise (session_id, exercise_id, template_exercise_id)
        VALUES (?, 'bench_press_barbell', ?)
      `).run(sRes.lastInsertRowid, teId)

      db.prepare(`
        INSERT INTO set_log (session_exercise_id, set_number, weight_kg, reps, is_completed, is_warmup)
        VALUES (?, 1, 5, 13, 1, 0)
      `).run(seRes.lastInsertRowid)
    }

    const n = runAllSignalChecks(db, DEFAULT_SETTINGS)
    assert.ok(n >= 1, `Attendu ≥ 1 signal, obtenu ${n}`)

    const advice = db.prepare(
      "SELECT * FROM equipment_advice WHERE signal_type = 'max_load_reached'"
    ).all()
    assert.ok(advice.length >= 1)
    assert.equal(advice[0].exercise_id, 'bench_press_barbell')
  })

  it('cooldown respecté : signal non dupliqué', () => {
    const db  = createDb()
    const pid = createActiveProgram(db)
    const tid = createTemplate(db, pid)
    createTE(db, tid, 'bench_press_barbell', 12)
    addBarbell(db, 1)
    addPlates(db, 1, 4)

    // Insérer un advice actif récent
    db.prepare(`
      INSERT INTO equipment_advice
        (signal_type, exercise_id, signal_data, status, last_generated_at)
      VALUES ('max_load_reached', 'bench_press_barbell', '{}', 'active', datetime('now'))
    `).run()

    const n = runAllSignalChecks(db, { ...DEFAULT_SETTINGS, equipment_advice_cooldown_days: 14 })
    // L'advice existant est récent → cooldown → pas de doublon
    const total = db.prepare("SELECT COUNT(*) as n FROM equipment_advice").get().n
    assert.equal(total, 1, 'Ne doit pas créer de doublon pendant le cooldown')
  })

  it('signal no_exercise_available si aucun exercice réalisable pour un muscle', () => {
    const db = createDb()
    createActiveProgram(db)
    // Pas d'équipement → tous les exercices qui nécessitent du matériel sont infaisables
    // Les exercices bodyweight (equipment_required=[]) restent faisables
    const n = runAllSignalChecks(db, DEFAULT_SETTINGS)
    const noExAdv = db.prepare(
      "SELECT * FROM equipment_advice WHERE signal_type = 'no_exercise_available'"
    ).all()
    // Avec certains muscles sans exercice bodyweight, des signaux sont créés
    assert.ok(typeof n === 'number')
    assert.ok(Array.isArray(noExAdv))
  })
})
```
