```sql
-- Profil utilisateur singleton (id=1 toujours)
CREATE TABLE IF NOT EXISTS user_profile (
  id                        INTEGER PRIMARY KEY CHECK(id = 1),
  -- Données personnelles
  display_name              TEXT,                        -- prénom ou pseudo optionnel
  age                       INTEGER,
  sex                       TEXT CHECK(sex IN ('male','female','other','prefer_not_to_say')),
  weight_kg                 REAL,                        -- poids actuel
  height_cm                 REAL,                        -- taille en cm
  -- Niveau et objectifs
  fitness_level             TEXT CHECK(fitness_level IN (
                              'beginner','intermediate','advanced','athlete'
                            )),
  primary_goal              TEXT CHECK(primary_goal IN (
                              'strength','hypertrophy','endurance',
                              'weight_loss','general_fitness','sport_performance'
                            )),
  secondary_goals           TEXT NOT NULL DEFAULT '[]',  -- JSON array de primary_goal values
  target_weight_kg          REAL,                        -- objectif poids (optionnel)
  sport_practiced           TEXT,                        -- sport pratiqué (optionnel)
  -- Contraintes d'entraînement
  sessions_per_week         INTEGER,
  session_duration_min      INTEGER,                     -- durée cible par séance en minutes
  prefers_exercise_variation INTEGER NOT NULL DEFAULT 1, -- 1=oui, varier les exos
  -- Santé et limitations
  injury_notes              TEXT,                        -- blessures/douleurs à noter
  medical_notes             TEXT,                        -- contre-indications médicales
  -- Onboarding
  onboarding_completed      INTEGER NOT NULL DEFAULT 0  CHECK(onboarding_completed IN (0,1)),
  onboarding_step           TEXT    NOT NULL DEFAULT 'not_started',
  -- Timestamps
  created_at                TEXT    NOT NULL DEFAULT (datetime('now')),
  updated_at                TEXT    NOT NULL DEFAULT (datetime('now'))
);

-- Insérer la ligne singleton si absente
```

## `src/routes/userProfile.js` — implémentation complète

```javascript
import express from 'express'
import { getDb } from '../db/database.js'

const router = express.Router()
const db     = getDb()

// Champs modifiables avec leurs types et contraintes
const PROFILE_SCHEMA = {
  display_name:              { type: 'string_nullable' },
  age:                       { type: 'number', min: 10, max: 100 },
  sex:                       { type: 'enum', enum: ['male','female','other','prefer_not_to_say'] },
  weight_kg:                 { type: 'number', min: 20, max: 500 },
  height_cm:                 { type: 'number', min: 50, max: 280 },
  fitness_level:             { type: 'enum', enum: ['beginner','intermediate','advanced','athlete'] },
  primary_goal:              { type: 'enum', enum: ['strength','hypertrophy','endurance','weight_loss','general_fitness','sport_performance'] },
  secondary_goals:           { type: 'json_array' },
  target_weight_kg:          { type: 'number_nullable', min: 20, max: 500 },
  sport_practiced:           { type: 'string_nullable' },
  sessions_per_week:         { type: 'number', min: 1, max: 7 },
  session_duration_min:      { type: 'number', min: 15, max: 240 },
  prefers_exercise_variation:{ type: 'boolean_int' },
  injury_notes:              { type: 'string_nullable' },
  medical_notes:             { type: 'string_nullable' },
  onboarding_completed:      { type: 'boolean_int' },
  onboarding_step:           { type: 'string' },
}

function parseProfile(row) {
  if (!row) return null
  return {
    ...row,
    secondary_goals:           JSON.parse(row.secondary_goals || '[]'),
    prefers_exercise_variation: row.prefers_exercise_variation === 1,
    onboarding_completed:       row.onboarding_completed === 1,
  }
}

function validateField(key, value) {
  const schema = PROFILE_SCHEMA[key]
  if (!schema) return { valid: false, error: `Champ inconnu : ${key}` }

  if (value === null || value === undefined) {
    if (['string_nullable','number_nullable'].includes(schema.type)) return { valid: true }
    if (schema.type !== 'json_array') return { valid: false, error: `${key} ne peut pas être null.` }
  }

  switch (schema.type) {
    case 'string':
    case 'string_nullable':
      if (value !== null && typeof value !== 'string') return { valid: false, error: `${key} doit être une chaîne.` }
      break
    case 'number':
    case 'number_nullable': {
      if (value === null) break
      const n = Number(value)
      if (isNaN(n)) return { valid: false, error: `${key} doit être un nombre.` }
      if (schema.min !== undefined && n < schema.min) return { valid: false, error: `${key} ≥ ${schema.min}.` }
      if (schema.max !== undefined && n > schema.max) return { valid: false, error: `${key} ≤ ${schema.max}.` }
      break
    }
    case 'enum':
      if (!schema.enum.includes(value)) return { valid: false, error: `${key} : valeurs valides = ${schema.enum.join(', ')}.` }
      break
    case 'boolean_int':
      if (![0, 1, true, false].includes(value)) return { valid: false, error: `${key} doit être 0, 1, true ou false.` }
      break
    case 'json_array':
      if (!Array.isArray(value)) return { valid: false, error: `${key} doit être un tableau.` }
      break
  }
  return { valid: true }
}

// ── GET /user-profile ─────────────────────────────────────────
router.get('/', (req, res) => {
  try {
    const row = db.prepare('SELECT * FROM user_profile WHERE id = 1').get()
    res.json(parseProfile(row))
  } catch (err) {
    console.error('[GET /user-profile]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── PUT /user-profile ─────────────────────────────────────────
// Met à jour les champs fournis (patch partiel)
router.put('/', (req, res) => {
  try {
    const body = req.body
    if (!body || typeof body !== 'object') {
      return res.status(400).json({ error: 'INVALID_BODY' })
    }

    const errors = []; const updates = []; const values = []

    for (const [key, value] of Object.entries(body)) {
      const r = validateField(key, value)
      if (!r.valid) { errors.push(r.error); continue }

      // Sérialiser selon le type
      let dbValue = value
      if (PROFILE_SCHEMA[key]?.type === 'json_array') dbValue = JSON.stringify(value)
      if (PROFILE_SCHEMA[key]?.type === 'boolean_int') dbValue = (value === true || value === 1) ? 1 : 0

      updates.push(`${key} = ?`)
      values.push(dbValue)
    }

    if (errors.length > 0) return res.status(400).json({ error: 'VALIDATION_ERRORS', details: errors })
    if (updates.length === 0) return res.status(400).json({ error: 'NO_FIELDS_TO_UPDATE' })

    updates.push("updated_at = datetime('now')")
    values.push(1)  // id=1 singleton

    db.prepare(`UPDATE user_profile SET ${updates.join(', ')} WHERE id = ?`).run(...values)

    const updated = db.prepare('SELECT * FROM user_profile WHERE id = 1').get()
    res.json(parseProfile(updated))
  } catch (err) {
    console.error('[PUT /user-profile]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

export default router
```

## Étapes d'implémentation

1. Créer `src/db/migrations/003_user_profile.sql`.
2. Vérifier que `src/db/migrate.js` lit bien les nouveaux fichiers
   (il lit tous les `.sql` du dossier triés alphabétiquement — la migration
   `003_` sera détectée automatiquement).
3. Créer `src/routes/userProfile.js`.
4. Modifier `server.js`.
5. Ajouter `userProfile` dans `public/js/api.js`.

**Points d'attention :**
- La table a `CHECK(id = 1)` : impossible d'insérer une deuxième ligne.
- `INSERT OR IGNORE INTO user_profile (id) VALUES (1)` : dans la migration,
  insère la ligne vide sans écraser une ligne existante (idempotent).
- `prefers_exercise_variation` stocké en INTEGER (0/1) mais `parseProfile`
  le retourne en boolean — cohérent avec le reste de l'API.
- `secondary_goals` : tableau JSON. Exemple : `["hypertrophy","endurance"]`.
- `onboarding_step` est un slug libre géré par le coach IA :
  `'not_started'` → `'asking_basics'` → `'asking_goals'` →
  `'asking_schedule'` → `'asking_equipment'` → `'completed'`

```javascript
import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import request from 'supertest'
import { app } from '../../server.js'

describe('T5.2 — User Profile', () => {
  it('GET /user-profile retourne le profil singleton', async () => {
    const res = await request(app).get('/api/v1/user-profile')
    assert.equal(res.status, 200)
    assert.equal(res.body.id, 1)
    assert.ok('onboarding_completed' in res.body)
    assert.equal(res.body.onboarding_completed, false)
  })

  it('PUT /user-profile met à jour les champs de base', async () => {
    const res = await request(app).put('/api/v1/user-profile').send({
      age: 28, sex: 'male', weight_kg: 80, height_cm: 178,
      fitness_level: 'intermediate', primary_goal: 'hypertrophy',
      sessions_per_week: 4, session_duration_min: 60,
    })
    assert.equal(res.status, 200)
    assert.equal(res.body.age, 28)
    assert.equal(res.body.fitness_level, 'intermediate')
    assert.equal(res.body.primary_goal, 'hypertrophy')
  })

  it('PUT /user-profile avec secondary_goals tableau', async () => {
    const res = await request(app).put('/api/v1/user-profile')
      .send({ secondary_goals: ['strength','endurance'] })
    assert.equal(res.status, 200)
    assert.ok(Array.isArray(res.body.secondary_goals))
    assert.ok(res.body.secondary_goals.includes('strength'))
  })

  it('PUT /user-profile avec champ inconnu → 400', async () => {
    const res = await request(app).put('/api/v1/user-profile')
      .send({ unknown_field: 'test' })
    assert.equal(res.status, 400)
  })

  it('PUT /user-profile avec age hors plage → 400', async () => {
    const res = await request(app).put('/api/v1/user-profile')
      .send({ age: 150 })
    assert.equal(res.status, 400)
  })

  it('PUT /user-profile onboarding_completed → true', async () => {
    const res = await request(app).put('/api/v1/user-profile')
      .send({ onboarding_completed: true, onboarding_step: 'completed' })
    assert.equal(res.status, 200)
    assert.equal(res.body.onboarding_completed, true)
  })
})
```
