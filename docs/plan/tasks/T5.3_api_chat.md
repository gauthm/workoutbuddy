```javascript
function buildSystemPrompt(db) {
  // Équipement
  const items = db.prepare('SELECT * FROM equipment_item').all()
  const equipSummary = items.length === 0
    ? 'Aucun équipement configuré.'
    : items.map(i => {
        if (i.weight_kg) return `${i.name} (${i.weight_kg} kg${i.count > 1 ? ' ×' + i.count : ''})`
        if (i.weight_min_kg) return `${i.name} (${i.weight_min_kg}–${i.weight_max_kg} kg)`
        return i.name
      }).join(', ')

  // Programme actif
  const prog = db.prepare("SELECT * FROM program WHERE is_active = 1 LIMIT 1").get()
  const progSummary = prog
    ? `${prog.name} (${prog.goal}, ${prog.level}, ${prog.days_per_week}j/sem, semaine ${prog.current_week})`
    : 'Aucun programme actif.'

  // 3 dernières séances
  const lastSessions = db.prepare(`
    SELECT s.name, s.scheduled_date, s.duration_seconds, s.total_volume_kg
    FROM session s
    WHERE s.status = 'completed'
    ORDER BY s.completed_at DESC
    LIMIT 3
  `).all()

  const sessionsSummary = lastSessions.length === 0
    ? 'Aucune séance enregistrée.'
    : lastSessions.map(s => {
        const dur = s.duration_seconds ? Math.round(s.duration_seconds / 60) + 'min' : '?'
        const vol = s.total_volume_kg   ? s.total_volume_kg + 'kg'                   : '?'
        return `${s.scheduled_date} ${s.name} ${dur} vol.${vol}`
      }).join(' | ')

  // État progression (5 premiers exercices)
  const progStates = db.prepare(`
    SELECT e.name_fr, ps.current_weight_kg, ps.consecutive_failures
    FROM progression_state ps
    JOIN template_exercise te ON ps.template_exercise_id = te.id
    JOIN exercise e           ON te.exercise_id = e.id
    LIMIT 5
  `).all()
  const progressSummary = progStates.length === 0
    ? 'Aucune donnée de progression.'
    : progStates.map(p =>
        `${p.name_fr}: ${p.current_weight_kg}kg${p.consecutive_failures > 0 ? ` (${p.consecutive_failures} échec(s))` : ''}`
      ).join(', ')

  // Conseils actifs
  const adviceCount = db.prepare(
    "SELECT COUNT(*) AS n FROM equipment_advice WHERE status = 'active'"
  ).get().n

  return `Tu es WorkoutBuddy, un assistant coach fitness expert. Tu travailles exclusivement avec les données d'entraînement réelles de l'utilisateur.

## Contexte utilisateur

**Équipement disponible** : ${equipSummary}

**Programme actif** : ${progSummary}

**Dernières séances** : ${sessionsSummary}

**État progression** : ${progressSummary}

**Conseils matériel actifs** : ${adviceCount} conseil(s) en attente

## Tes capacités
1. Conseils personnalisés basés sur l'historique réel.
2. Créer ou modifier des programmes/séances via des **actions structurées JSON**.
3. Répondre aux questions sur la technique, la science, la nutrition de base.

## Règles IMPÉRATIVES
- Réponds en français sauf si l'utilisateur écrit en anglais.
- Pour toute création ou modification → inclure le bloc \`\`\`action...  \`\`\` (voir format).
- N'invente JAMAIS de données hors du contexte fourni.
- Sois concis (2–4 phrases pour conseils simples, liste pour instructions).
- Pas de marques commerciales, pas de liens d'affiliation (sauf demande explicite).
- Si contexte insuffisant pour répondre → demande UNE précision.

## Format d'action structurée
Pour CRÉER ou MODIFIER, inclure EXACTEMENT un bloc :
\`\`\`action
{ "action": "...", "explanation": "...", "data": { ... } }
\`\`\``
}
```

## Validation des actions

```javascript
// Schéma de validation minimal — vérifier les champs critiques
function validateAction(action, db) {
  const errors = []

  if (!action || typeof action !== 'object') {
    return ['Corps d\'action invalide ou manquant.']
  }

  const VALID_ACTIONS = ['create_program', 'update_program', 'create_session']
  if (!VALID_ACTIONS.includes(action.action)) {
    errors.push(`Action "${action.action}" non reconnue. Valeurs valides : ${VALID_ACTIONS.join(', ')}.`)
  }

  if (!action.explanation || typeof action.explanation !== 'string' || !action.explanation.trim()) {
    errors.push('Le champ "explanation" est obligatoire et non vide.')
  }

  if (!action.data || typeof action.data !== 'object') {
    errors.push('Le champ "data" est obligatoire.')
    return errors
  }

  if (action.action === 'create_program') {
    const { name, goal, level, days_per_week, templates } = action.data
    if (!name?.trim())         errors.push('"data.name" obligatoire.')
    const GOALS = ['strength','hypertrophy','endurance','weight_loss','general_fitness']
    if (!GOALS.includes(goal)) errors.push(`"data.goal" invalide (${goal}).`)
    const LEVELS = ['beginner','intermediate','advanced']
    if (!LEVELS.includes(level)) errors.push(`"data.level" invalide (${level}).`)
    const dpw = parseInt(days_per_week)
    if (!dpw || dpw < 2 || dpw > 6) errors.push('"data.days_per_week" doit être entre 2 et 6.')
    if (templates) {
      for (const tpl of templates) {
        for (const ex of (tpl.exercises || [])) {
          const found = db.prepare('SELECT id FROM exercise WHERE id = ?').get(ex.exercise_id)
          if (!found) errors.push(`exercise_id "${ex.exercise_id}" introuvable en DB.`)
          if (ex.target_sets && (ex.target_sets < 1 || ex.target_sets > 10)) {
            errors.push(`target_sets doit être entre 1 et 10 pour "${ex.exercise_id}".`)
          }
          if (ex.target_reps_min && ex.target_reps_max && ex.target_reps_min > ex.target_reps_max) {
            errors.push(`target_reps_min > target_reps_max pour "${ex.exercise_id}".`)
          }
        }
      }
    }
  }

  if (action.action === 'update_program') {
    if (!action.data.program_id) errors.push('"data.program_id" obligatoire.')
    else {
      const prog = db.prepare('SELECT id FROM program WHERE id = ?').get(action.data.program_id)
      if (!prog) errors.push(`Programme ${action.data.program_id} introuvable.`)
    }
  }

  if (action.action === 'create_session') {
    if (!action.data.name?.trim())       errors.push('"data.name" obligatoire.')
    if (!action.data.scheduled_date)     errors.push('"data.scheduled_date" obligatoire.')
    if (!/^\d{4}-\d{2}-\d{2}$/.test(action.data.scheduled_date || '')) {
      errors.push('"data.scheduled_date" format YYYY-MM-DD requis.')
    }
    for (const ex of (action.data.exercises || [])) {
      const found = db.prepare('SELECT id FROM exercise WHERE id = ?').get(ex.exercise_id)
      if (!found) errors.push(`exercise_id "${ex.exercise_id}" introuvable.`)
    }
  }

  return errors
}
```

## Application des actions en DB

```javascript
function applyAction(action, db) {
  const { action: type, data } = action

  if (type === 'create_program') {
    return db.transaction(() => {
      const pRes = db.prepare(`
        INSERT INTO program
          (name, goal, level, days_per_week, duration_weeks, progression_model,
           use_exercise_rotation, deload_auto_enabled)
        VALUES (?,?,?,?,?,?,0,0)
      `).run(
        data.name.trim(),
        data.goal, data.level,
        parseInt(data.days_per_week),
        data.duration_weeks || null,
        data.progression_model || 'double_progression',
      )
      const pid = pRes.lastInsertRowid

      for (let ti = 0; ti < (data.templates || []).length; ti++) {
        const tpl = data.templates[ti]
        const tRes = db.prepare(
          'INSERT INTO workout_template (program_id, name, order_index) VALUES (?,?,?)'
        ).run(pid, tpl.name, ti)
        const tid = tRes.lastInsertRowid

        for (let ei = 0; ei < (tpl.exercises || []).length; ei++) {
          const ex = tpl.exercises[ei]
          const teRes = db.prepare(`
            INSERT INTO template_exercise
              (workout_template_id, exercise_id, order_index, target_sets,
               target_reps_min, target_reps_max, rest_seconds, progression_model)
            VALUES (?,?,?,?,?,?,?,?)
          `).run(
            tid, ex.exercise_id, ei,
            ex.target_sets    || 3,
            ex.target_reps_min || 8,
            ex.target_reps_max || 12,
            ex.rest_seconds    || 90,
            'inherit',
          )
          db.prepare(
            'INSERT INTO progression_state (template_exercise_id, current_weight_kg) VALUES (?,0)'
          ).run(teRes.lastInsertRowid)
        }
      }

      return { type: 'program', id: pid, name: data.name.trim() }
    })()
  }

  if (type === 'create_session') {
    return db.transaction(() => {
      const sRes = db.prepare(
        "INSERT INTO session (name, scheduled_date, workout_template_id) VALUES (?,?,?)"
      ).run(data.name.trim(), data.scheduled_date, data.workout_template_id || null)
      const sid = sRes.lastInsertRowid

      for (let i = 0; i < (data.exercises || []).length; i++) {
        const ex = data.exercises[i]
        db.prepare(
          'INSERT INTO session_exercise (session_id, exercise_id, order_index) VALUES (?,?,?)'
        ).run(sid, ex.exercise_id, i)
      }

      return { type: 'session', id: sid, name: data.name.trim() }
    })()
  }

  if (type === 'update_program') {
    // Appliquer chaque changement de la liste data.changes
    return db.transaction(() => {
      const results = []
      for (const change of (data.changes || [])) {
        if (change.type === 'update_program_meta') {
          const UPDATABLE = ['name','goal','level','days_per_week','progression_model']
          const updates = []; const values = []
          for (const f of UPDATABLE) {
            if (change.fields?.[f] !== undefined) {
              updates.push(`${f} = ?`); values.push(change.fields[f])
            }
          }
          if (updates.length > 0) {
            values.push(data.program_id)
            db.prepare(`UPDATE program SET ${updates.join(', ')} WHERE id = ?`).run(...values)
            results.push(`meta updated`)
          }
        }
        // Autres types de changements peuvent être ajoutés ici
      }
      return { type: 'program_update', program_id: data.program_id, changes: results.length }
    })()
  }

  throw new Error(`Action "${type}" non implémentée.`)
}
```

import express from 'express'
import { getDb }         from '../db/database.js'
import { getAiSettings } from './settings.js'
import { streamChat }    from '../services/aiProvider.js'

const router = express.Router()
const db     = getDb()

// buildSystemPrompt, validateAction, applyAction définis ci-dessus dans ce fichier

// ── POST /ai/chat ─────────────────────────────────────────────
router.post('/chat', async (req, res) => {
  const { message, history = [] } = req.body

  if (!message?.trim()) {
    return res.status(400).json({ error: 'MISSING_MESSAGE' })
  }

  const aiSettings = getAiSettings(db)
  if (!aiSettings.base_url || !aiSettings.model) {
    return res.status(503).json({ error: 'AI_NOT_CONFIGURED' })
  }

  // Headers SSE
  res.writeHead(200, {
    'Content-Type':      'text/event-stream',
    'Cache-Control':     'no-cache',
    'Connection':        'keep-alive',
    'X-Accel-Buffering': 'no',
  })

  // Construire les messages
  const systemPrompt = buildSystemPrompt(db)
  const messages = [
    { role: 'system', content: systemPrompt },
    ...history.slice(-10).map(m => ({ role: m.role, content: m.content })),
    { role: 'user', content: message.trim() },
  ]

  const sendEvent = (data) => {
    res.write(`data: ${JSON.stringify(data)}\n\n`)
  }

  // Buffer pour détecter le bloc action
  let fullContent = ''
  let actionBuffer = ''
  let inActionBlock = false

  try {
    for await (const event of streamChat(messages, aiSettings)) {

      if (event.type === 'error') {
        sendEvent({ type: 'error', message: event.message })
        break
      }

      if (event.type === 'done') {
        sendEvent({ type: 'done', usage: event.usage })
        break
      }

      if (event.type === 'token') {
        fullContent += event.content

        // Détection du bloc ```action ... ```
        if (!inActionBlock && fullContent.includes('```action')) {
          inActionBlock = true
          // Envoyer le texte avant le bloc action
          const beforeAction = fullContent.split('```action')[0]
          if (beforeAction.trim()) {
            sendEvent({ type: 'token', content: event.content })
          }
          continue
        }

        if (inActionBlock) {
          actionBuffer += event.content
          // Détecter la fin du bloc
          if (actionBuffer.includes('```')) {
            const jsonStr = actionBuffer.split('```')[0].trim()
            try {
              const parsed = JSON.parse(jsonStr)
              sendEvent({ type: 'action', action: parsed })
            } catch {
              sendEvent({ type: 'error', message: 'Réponse du provider illisible. Essayez un autre modèle.' })
            }
            inActionBlock = false
            actionBuffer  = ''
          }
          continue
        }

        // Token normal
        sendEvent({ type: 'token', content: event.content })
      }
    }
  } catch (err) {
    sendEvent({ type: 'error', message: err.message })
  } finally {
    res.end()
  }
})

// ── GET /ai/advice ────────────────────────────────────────────
router.get('/advice', (req, res) => {
  try {
    const rows = db.prepare(`
      SELECT * FROM equipment_advice
      WHERE status = 'active'
      ORDER BY created_at DESC
    `).all().map(a => ({
      ...a,
      signal_data: JSON.parse(a.signal_data || '{}'),
    }))
    res.json({ advice: rows })
  } catch (err) {
    console.error('[GET /ai/advice]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── POST /ai/advice/:id/dismiss ───────────────────────────────
router.post('/advice/:id/dismiss', (req, res) => {
  try {
    const row = db.prepare('SELECT id FROM equipment_advice WHERE id = ?').get(req.params.id)
    if (!row) return res.status(404).json({ error: 'ADVICE_NOT_FOUND' })

    db.prepare(`
      UPDATE equipment_advice
      SET status = 'dismissed', dismissed_at = datetime('now')
      WHERE id = ?
    `).run(req.params.id)

    res.json({ id: parseInt(req.params.id), status: 'dismissed' })
  } catch (err) {
    console.error('[POST /ai/advice/:id/dismiss]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── POST /ai/advice/:id/purchased ─────────────────────────────
router.post('/advice/:id/purchased', (req, res) => {
  try {
    const row = db.prepare('SELECT id FROM equipment_advice WHERE id = ?').get(req.params.id)
    if (!row) return res.status(404).json({ error: 'ADVICE_NOT_FOUND' })

    db.prepare(`
      UPDATE equipment_advice
      SET status = 'purchased', purchased_at = datetime('now')
      WHERE id = ?
    `).run(req.params.id)

    res.json({ id: parseInt(req.params.id), status: 'purchased' })
  } catch (err) {
    console.error('[POST /ai/advice/:id/purchased]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── POST /ai/apply-action ─────────────────────────────────────
router.post('/apply-action', (req, res) => {
  try {
    const { action } = req.body
    if (!action) return res.status(400).json({ error: 'MISSING_ACTION' })

    const errors = validateAction(action, db)
    if (errors.length > 0) {
      return res.status(400).json({
        error:   'INVALID_ACTION_SCHEMA',
        details: errors,
      })
    }

    const result = applyAction(action, db)
    res.status(201).json({ applied: true, result })
  } catch (err) {
    console.error('[POST /ai/apply-action]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

// ── POST /ai/generate-equipment-advice ────────────────────────
router.post('/generate-equipment-advice', async (req, res) => {

    console.error('[POST /ai/generate-equipment-advice]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

## Étapes d'implémentation

1. Créer `src/routes/ai.js` avec toutes les fonctions dans un seul fichier :
   `buildSystemPrompt`, `validateAction`, `applyAction`, routes.
2. Modifier `server.js`.

**Points d'attention :**
- `buildSystemPrompt` et `validateAction` et `applyAction` sont des fonctions
  locales dans `ai.js` — pas des modules séparés. Cela évite les imports
  circulaires avec `settings.js`.
- `getAiSettings` est importé depuis `settings.js` :
  ```javascript
  import { getAiSettings } from './settings.js'
  ```
  S'assurer que cette fonction est bien exportée nommément en T5.1.
- Pour le SSE, **ne pas utiliser** `res.json()` après `res.writeHead()` —
  utiliser uniquement `res.write()` et `res.end()`.
- Le bloc `inActionBlock` accumule les tokens dans `actionBuffer` jusqu'à
  trouver un ` ``` ` fermant. Si le LLM ne ferme jamais le bloc (timeout),
  l'event `done` ou `error` terminera le stream proprement.
- `applyAction` est dans une `db.transaction()` — si une partie échoue,
  tout est rollbacké automatiquement.
