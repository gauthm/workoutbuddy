## Collecte non-streaming (helper)

```javascript
/**
 * Collecte tous les tokens d'un streamChat en une seule chaîne.
 * Timeout géré par streamChat lui-même.
 */
async function collectStreamToString(messages, aiSettings) {
  let text = ''
  for await (const event of streamChat(messages, aiSettings)) {
    if (event.type === 'token') text += event.content
    if (event.type === 'error') throw new Error(event.message)
    if (event.type === 'done')  break
  }
  return text.trim()
}
```

## Génération LLM des conseils en attente

```javascript
// Fonction à appeler dans POST /ai/generate-equipment-advice
// ET après POST /sessions/:id/complete (de manière asynchrone)

async function generatePendingAdviceTexts(db) {
  const aiSettings = getAiSettings(db)
  if (!aiSettings.base_url || !aiSettings.model) return 0

  // Récupérer les conseils sans texte
  const pending = db.prepare(`
    SELECT * FROM equipment_advice
    WHERE status = 'active' AND (advice_text IS NULL OR advice_text = '')
    ORDER BY created_at ASC
    LIMIT 5
  `).all()

  let generated = 0
  for (const advice of pending) {
    try {
      const signalData = JSON.parse(advice.signal_data || '{}')
      const prompt     = buildAdvicePrompt(signalData)
      const messages   = [{ role: 'user', content: prompt }]
      const text       = await collectStreamToString(messages, aiSettings)

      if (text) {
        db.prepare(`
          UPDATE equipment_advice
          SET advice_text = ?, last_generated_at = datetime('now')
          WHERE id = ?
        `).run(text, advice.id)
        generated++
      }
    } catch (err) {
      console.warn(`[advice] Génération échouée pour advice ${advice.id}:`, err.message)
      // Non bloquant : continuer avec les suivants
    }
  }
  return generated
}
```

  } catch (err) {

## `public/js/components/coach/AdviceCard.js` — implémentation complète

```javascript
import { inject } from '/vendor/vue.esm-browser.prod.js'
import { coach as api } from '../../api.js'
import { t } from '../../i18n.js'
import ConfirmModal from '../common/ConfirmModal.js'

// Icône par type de signal
const SIGNAL_ICONS = {
  max_load_reached:        '🏋️',
  min_increment_too_large: '📏',
  no_exercise_available:   '❌',
  no_rotation_possible:    '🔄',
  persistent_plateau:      '📉',
}

export default {
  name: 'AdviceCard',
  props: {
    advice: { type: Object, required: true }
    // { id, signal_type, signal_data, advice_text, status, exercise_id, muscle_group }
  },
  emits: ['dismissed', 'purchased'],
  components: { ConfirmModal },

  setup(props, { emit }) {
    const store  = inject('store')
    const router = inject('router')

    // État local
    const showPurchasedConfirm     = { value: false }
    const showInventoryConfirm     = { value: false }

    async function dismiss() {
      try {
        await api.dismissAdvice(props.advice.id)
        emit('dismissed', props.advice.id)
        store.showToast('Conseil masqué.', 'info')
      } catch (err) {
        store.showToast(err.message, 'error')
      }
    }

    async function markPurchased() {
      try {
        await api.purchasedAdvice(props.advice.id)
        emit('purchased', props.advice.id)
        showInventoryConfirm.value = true
      } catch (err) {
        store.showToast(err.message, 'error')
      }
    }

    function goToEquipment() {
      showInventoryConfirm.value = false
      router.push('#/equipment')
    }

    const icon = SIGNAL_ICONS[props.advice.signal_type] || '⚠️'

    return {
      dismiss, markPurchased, goToEquipment,
      showPurchasedConfirm, showInventoryConfirm,
      icon, t,
    }
  },

  template: `
    <div class="advice-card">
      <div class="advice-card__header">
        <span class="advice-card__icon">{{ icon }}</span>
        <span class="advice-card__title">{{ t('advice.title') }}</span>
      </div>

      <div class="advice-card__body">
        <template v-if="advice.advice_text">
          <p class="advice-card__text">{{ advice.advice_text }}</p>
        </template>
        <template v-else>
          <p class="advice-card__text advice-card__text--muted">
            {{ t('advice.generating') }}
          </p>
        </template>
      </div>

      <div class="advice-card__actions">
        <button class="btn btn--secondary advice-card__btn" @click="dismiss()">
          {{ t('advice.dismiss') }}
        </button>
        <button class="btn btn--primary advice-card__btn" @click="markPurchased()">
          {{ t('advice.purchased') }}
        </button>
      </div>

      <!-- Confirmation mise à jour inventaire -->
      <ConfirmModal
        v-if="showInventoryConfirm.value"
        :title="t('advice.update_inventory')"
        :body="t('advice.update_inventory_body')"
        :confirm-text="t('common.yes')"
        :cancel-text="t('common.no')"
        @confirm="goToEquipment()"
        @cancel="showInventoryConfirm.value = false"
      />
    </div>
  `
}
```

   `collectStreamToString` et `generatePendingAdviceTexts`.
2. Modifier `src/routes/ai.js` :
   - Importer `generatePendingAdviceTexts` depuis `adviceTextGenerator.js`.
   - Remplacer `POST /generate-equipment-advice`.
3. Modifier `src/routes/sessions.js` :
   - Importer `generatePendingAdviceTexts`.
   - Mettre à jour le bloc signaux dans `POST /:id/complete`.
4. Créer `public/js/components/coach/AdviceCard.js`.
5. Ajouter le CSS dans `components.css`.
6. Ajouter les clés i18n.

**Points d'attention :**
- La génération de texte dans `POST /sessions/:id/complete` est
  **asynchrone et non bloquante** : on ne `await` pas `generatePendingAdviceTexts`.
  La réponse HTTP est envoyée immédiatement. Le texte est généré en
  arrière-plan.
- Si le provider IA n'est pas configuré, `generatePendingAdviceTexts`
  retourne `0` silencieusement (contrôle en début de fonction).
- `AdviceCard` utilise un objet réactif local `showInventoryConfirm`
  avec `.value` (reactive ref-like) car `reactive({})` ne peut pas
  wraper un booléen directement. Utiliser :
  ```javascript
  import { ref } from '/vendor/vue.esm-browser.prod.js'
  const showInventoryConfirm = ref(false)
  // Dans le template : v-if="showInventoryConfirm"
  // Dans setup : showInventoryConfirm.value = true
  ```
  Corriger la déclaration en utilisant `ref` dans l'implémentation finale.
- `AdviceCard` est utilisé dans `StatsDashboardView` (T4.3) ET dans
  `CoachView` (T5.5). L'importer dans les deux.

## Tests à écrire

Fichier : `tests/integration/advice.test.js`

```javascript
import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import request from 'supertest'
import { app } from '../../server.js'

describe('T5.4 — Conseils d\'achat', () => {
  let adviceId

  it('POST /ai/generate-equipment-advice crée des signaux', async () => {
    const res = await request(app).post('/api/v1/ai/generate-equipment-advice')
    assert.equal(res.status, 200)
    assert.ok(typeof res.body.generated === 'number')
  })

  it('GET /ai/advice retourne les advice actifs', async () => {
    const res = await request(app).get('/api/v1/ai/advice')
    assert.equal(res.status, 200)
    assert.ok(Array.isArray(res.body.advice))
    if (res.body.advice.length > 0) {
      adviceId = res.body.advice[0].id
      assert.ok('signal_type' in res.body.advice[0])
      assert.ok('signal_data' in res.body.advice[0])
    }
  })

  it('POST /ai/advice/:id/dismiss change status → dismissed', async () => {
    if (!adviceId) return  // skip si pas d'advice
    const res = await request(app).post(`/api/v1/ai/advice/${adviceId}/dismiss`)
    assert.equal(res.status, 200)
    assert.equal(res.body.status, 'dismissed')
  })

  it('Advice dismissed n\'apparaît plus dans GET /ai/advice', async () => {
    const res = await request(app).get('/api/v1/ai/advice')
    assert.equal(res.status, 200)
    const found = res.body.advice.find(a => a.id === adviceId)
    assert.ok(!found, 'L\'advice dismissed ne doit pas apparaître')
  })
})
```
