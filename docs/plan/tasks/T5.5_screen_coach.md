## Note sur le streaming SSE côté client

Le navigateur reçoit des chunks SSE du format :
```
data: {"type":"token","content":"mot"}\n\n
```

Lire avec `fetch` + `ReadableStream` :

```javascript
const res = await fetch('/api/v1/ai/chat', {
  method:  'POST',
  headers: { 'Content-Type': 'application/json' },
  body:    JSON.stringify({ message, history }),
})
const reader  = res.body.getReader()
const decoder = new TextDecoder()
let   buffer  = ''

while (true) {
  const { done, value } = await reader.read()
  if (done) break
  buffer += decoder.decode(value, { stream: true })

  const lines = buffer.split('\n')
  buffer = lines.pop()  // ligne incomplète

  for (const line of lines) {
    if (!line.startsWith('data: ')) continue
    const raw = line.slice(6).trim()
    if (!raw) continue
    try {
      const event = JSON.parse(raw)
      // traiter l'event
    } catch { /* ignorer les lignes non-JSON */ }
  }
}
```

## `ActionPreview.js`

```javascript
import { ref } from '/vendor/vue.esm-browser.prod.js'
import { t } from '../../i18n.js'

const ACTION_TYPE_KEYS = {
  create_program: 'coach.action_type_create_program',
  update_program: 'coach.action_type_update_program',
  create_session: 'coach.action_type_create_session',
}

export default {
  name: 'ActionPreview',
  props: {
    action: { type: Object, required: true }
    // { action: string, explanation: string, data: object }
  },
  emits: ['confirm', 'cancel'],

  setup(props) {
    const showJson = ref(false)
    const actionLabel = ACTION_TYPE_KEYS[props.action.action] || props.action.action

    // Résumé lisible
    function buildSummary(action) {
      const d = action.data || {}
      if (action.action === 'create_program') {
        const tplCount = (d.templates || []).length
        const exCount  = (d.templates || []).reduce((sum, t) => sum + (t.exercises || []).length, 0)
        return `"${d.name}" — ${d.goal}, ${d.level}, ${d.days_per_week}j/sem, ${tplCount} template(s), ${exCount} exercice(s)`
      }
      if (action.action === 'create_session') {
        return `"${d.name}" le ${d.scheduled_date}`
      }
      if (action.action === 'update_program') {
        return `Programme ${d.program_id} — ${(d.changes || []).length} modification(s)`
      }
      return JSON.stringify(d).slice(0, 100)
    }

    return { showJson, actionLabel, buildSummary, t }
  },

  template: `
    <div class="action-preview">
      <div class="action-preview__header">
        <span class="action-preview__icon">⚡</span>
        <span class="action-preview__type">{{ t(actionLabel) }}</span>
      </div>

      <p class="action-preview__explanation">{{ action.explanation }}</p>
      <p class="action-preview__summary">{{ buildSummary(action) }}</p>

      <!-- Toggle JSON -->
      <button class="action-preview__toggle" @click="showJson = !showJson">
        {{ showJson ? '▲ Masquer les détails' : '▼ Voir les détails' }}
      </button>
      <pre v-if="showJson" class="action-preview__json">{{ JSON.stringify(action.data, null, 2) }}</pre>

      <div class="action-preview__actions">
        <button class="btn btn--secondary" @click="$emit('cancel')">
          {{ t('coach.action_cancel') }}
        </button>
        <button class="btn btn--primary" @click="$emit('confirm')">
          {{ t('coach.action_apply') }}
        </button>
      </div>
    </div>
  `
}
```

## `CoachView.js` — implémentation complète

```javascript
import { reactive, ref, onMounted, inject, nextTick }
  from '/vendor/vue.esm-browser.prod.js'
import { coach as coachApi } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader     from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner    from '../common/ErrorBanner.js'
import ChatMessage    from './ChatMessage.js'
import ActionPreview  from './ActionPreview.js'
import AdviceCard     from './AdviceCard.js'

export default {
  name: 'CoachView',
  components: {
    PageHeader, LoadingSpinner, ErrorBanner,
    ChatMessage, ActionPreview, AdviceCard
  },

  setup() {
    const router = inject('router')
    const store  = inject('store')

    const state = reactive({
      messages:      [],   // [{ role, content, isStreaming?, isError? }]
      advice:        [],   // EquipmentAdvice[]
      loadingAdvice: false,
      input:         '',
      isStreaming:   false,
      pendingAction: null, // { action, explanation, data }
      applying:      false,
      notConfigured: false,
    })

    const inputRef       = ref(null)
    const messagesEndRef = ref(null)

    // Historique pour le contexte : 10 derniers échanges
    function getHistory() {
      return state.messages
        .filter(m => !m.isError && m.role !== 'system')
        .slice(-10)
        .map(m => ({ role: m.role, content: m.content }))
    }

    async function scrollToBottom() {
      await nextTick()
      messagesEndRef.value?.scrollIntoView({ behavior: 'smooth', block: 'end' })
    }

    // ── Chargement des conseils ───────────────────────────────
    async function loadAdvice() {
      state.loadingAdvice = true
      try {
        const data = await coachApi.advice()
        state.advice = data.advice
      } catch (_) {}
      finally { state.loadingAdvice = false }
    }

    // ── Envoi d'un message ────────────────────────────────────
    async function sendMessage() {
      const text = state.input.trim()
      if (!text || state.isStreaming) return

      state.input        = ''
      state.pendingAction = null

      // Ajouter le message utilisateur
      state.messages.push({ role: 'user', content: text })
      await scrollToBottom()

      // Préparer le message assistant en cours de streaming
      const assistantMsg = reactive({ role: 'assistant', content: '', isStreaming: true })
      state.messages.push(assistantMsg)
      state.isStreaming = true

      try {
        const res = await fetch('/api/v1/ai/chat', {
          method:  'POST',
          headers: { 'Content-Type': 'application/json' },
          body:    JSON.stringify({ message: text, history: getHistory() }),
        })

        // Erreur non-SSE (ex: 503 provider non configuré)
        if (!res.ok) {
          const data = await res.json().catch(() => ({}))
          if (data.error === 'AI_NOT_CONFIGURED') {
            state.notConfigured = true
          }
          assistantMsg.content    = data.error === 'AI_NOT_CONFIGURED'
            ? t('coach.not_configured')
            : t('coach.error')
          assistantMsg.isStreaming = false
          assistantMsg.isError    = true
          return
        }

        const reader  = res.body.getReader()
        const decoder = new TextDecoder()
        let   buffer  = ''

        while (true) {
          const { done, value } = await reader.read()
          if (done) break
          buffer += decoder.decode(value, { stream: true })

          const lines = buffer.split('\n')
          buffer = lines.pop()

          for (const line of lines) {
            if (!line.startsWith('data: ')) continue
            const raw = line.slice(6).trim()
            if (!raw) continue
            let event
            try { event = JSON.parse(raw) } catch { continue }

            if (event.type === 'token') {
              assistantMsg.content += event.content
              await scrollToBottom()
            }
            if (event.type === 'action') {
              state.pendingAction = event.action
              // Ne pas afficher le bloc action dans le texte
            }
            if (event.type === 'error') {
              assistantMsg.content   = event.message
              assistantMsg.isError   = true
              assistantMsg.isStreaming = false
            }
            if (event.type === 'done') {
              assistantMsg.isStreaming = false
            }
          }
        }

      } catch (err) {
        assistantMsg.content    = `${t('coach.error')} : ${err.message}`
        assistantMsg.isError    = true
        assistantMsg.isStreaming = false
      } finally {
        assistantMsg.isStreaming = false
        state.isStreaming = false
        await scrollToBottom()
      }
    }

    // ── Confirmation d'action ─────────────────────────────────
    async function applyAction() {
      if (!state.pendingAction || state.applying) return
      state.applying = true
      try {
        const res = await fetch('/api/v1/ai/apply-action', {
          method:  'POST',
          headers: { 'Content-Type': 'application/json' },
          body:    JSON.stringify({ action: state.pendingAction }),
        })
        const data = await res.json()

        if (!res.ok) {
          const details = (data.details || []).join('; ')
          store.showToast(`Erreur : ${details || data.error}`, 'error')
          return
        }

        store.showToast(t('coach.action_applied'), 'success')

        // Rediriger si programme créé
        if (data.result?.type === 'program') {
          router.push('#/programs/' + data.result.id)
        }
        if (data.result?.type === 'session') {
          router.push('#/session/history')
        }
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        state.applying     = false
        state.pendingAction = null
      }
    }

    function cancelAction() {
      state.pendingAction = null
      state.messages.push({
        role: 'assistant',
        content: t('coach.action_cancelled'),
        isError: false,
      })
    }

    function onAdviceDismissed(id) {
      state.advice = state.advice.filter(a => a.id !== id)
    }

    function onAdvicePurchased(id) {
      state.advice = state.advice.filter(a => a.id !== id)
    }

    // Touches clavier : Entrée = envoyer, Shift+Entrée = saut de ligne
    function onKeydown(e) {
      if (e.key === 'Enter' && !e.shiftKey) {
        e.preventDefault()
        sendMessage()
      }
    }

    onMounted(loadAdvice)

    return {
      state, inputRef, messagesEndRef,
      sendMessage, applyAction, cancelAction,
      onAdviceDismissed, onAdvicePurchased,
      onKeydown, t,
    }
  },

  template: `
    <div class="view coach-view">
      <PageHeader :title="t('coach.title')" />

      <div class="coach-view__content">

        <!-- Conseils d'achat actifs -->
        <div v-if="state.advice.length" class="coach-view__advice-section">
          <div class="detail-section__title" style="padding:0 1rem;margin-top:1rem">
            {{ t('coach.advice_section') }}
          </div>
          <AdviceCard
            v-for="adv in state.advice" :key="adv.id"
            :advice="adv"
            @dismissed="onAdviceDismissed"
            @purchased="onAdvicePurchased"
          />
        </div>

        <!-- Messages du chat -->
        <div class="coach-view__messages">
          <div v-if="!state.messages.length && !state.loadingAdvice"
            class="coach-view__empty">
            <div style="font-size:3rem;margin-bottom:0.75rem">🤖</div>
            <p style="color:var(--color-text-muted);text-align:center;padding:0 2rem">
              {{ state.notConfigured ? t('coach.not_configured') : 'Posez une question sur votre entraînement.' }}
            </p>
            <button v-if="state.notConfigured"
              class="btn btn--primary" style="margin-top:1rem"
              @click="router.push('#/settings')">
              Configurer le Coach IA
            </button>
          </div>

          <ChatMessage
            v-for="(msg, i) in state.messages" :key="i"
            :message="msg"
          />

          <!-- Action en attente de confirmation -->
          <ActionPreview
            v-if="state.pendingAction"
            :action="state.pendingAction"
            @confirm="applyAction()"
            @cancel="cancelAction()"
          />

          <div ref="messagesEndRef" style="height:1px" />
        </div>
      </div>

      <!-- Zone de saisie fixe en bas -->
      <div class="coach-view__input-area">
        <textarea
          ref="inputRef"
          class="coach-view__input"
          v-model="state.input"
          :placeholder="t('coach.placeholder')"
          :disabled="state.isStreaming"
          rows="1"
          @keydown="onKeydown"
        />
        <button
          class="coach-view__send"
          :disabled="!state.input.trim() || state.isStreaming"
          @click="sendMessage()">
          {{ state.isStreaming ? '…' : '↑' }}
        </button>
      </div>
    </div>
  `
}
```

**Points d'attention :**
- `textarea` auto-resize : le composant utilise `rows="1"` et `max-height: 120px`.
  Pour l'auto-resize, ajouter dans l'event `@input` :
  ```javascript
  function autoResize(e) {
    e.target.style.height = 'auto'
    e.target.style.height = e.target.scrollHeight + 'px'
  }
  // @input="e => { state.input = e.target.value; autoResize(e) }"
  ```
- `scrollToBottom` utilise `await nextTick()` pour s'assurer que le DOM
  est mis à jour avant de scroller.
- `state.messages` est un tableau réactif. Les objets message sont
  `reactive({ ... })` pour permettre la mise à jour du `content` pendant
  le streaming sans recréer l'élément DOM.
- `state.isStreaming = false` dans le `finally` garantit que le bouton
  d'envoi est réactivé même si une erreur se produit.
- Si `state.notConfigured = true` (503 reçu), l'état est persistant
  dans la session — l'utilisateur voit le message de config jusqu'à
  navigation.
- `AdviceCard` est importé ici ET utilisé dans `StatsDashboardView`
  (T4.3). Si T4.3 n'a pas encore d'`AdviceCard`, l'ajouter maintenant
  dans le dashboard.
