## `AIProviderForm.js` — formulaire provider IA

```javascript
import { reactive, onMounted, inject } from '/vendor/vue.esm-browser.prod.js'
import { settings as settingsApi } from '../../api.js'
import { t } from '../../i18n.js'

export default {
  name: 'AIProviderForm',
  props: {
    currentSettings: { type: Object, required: true }
    // { ai_base_url, ai_model, ai_api_key_set }
  },
  emits: ['saved'],

  setup(props, { emit }) {
    const store = inject('store')

    const state = reactive({
      base_url:    props.currentSettings.ai_base_url || '',
      model:       props.currentSettings.ai_model    || '',
      api_key:     '',       // toujours vide en affichage
      models:      [],
      testing:     false,
      testResult:  null,     // { ok: bool, message: string }
      listingModels: false,
      savingUrl:   false,
      savingModel: false,
      savingKey:   false,
    })

    async function saveField(key, value) {
      if (!value && key !== 'ai_api_key') return
      try {
        await settingsApi.update({ [key]: value })
        emit('saved', { [key]: value })
        if (key === 'ai_api_key') {
          state.api_key = ''
          store.showToast(
            value ? t('settings.ai_api_key_set') : t('settings.ai_api_key_not_set'),
            'success'
          )
        } else {
          store.showToast(t('settings.saved'), 'success')
        }
      } catch (err) {
        store.showToast(err.message, 'error')
      }
    }

    async function testConnection() {
      state.testing    = true
      state.testResult = null
      try {
        const res = await settingsApi.testAI()
        state.testResult = {
          ok:      true,
          message: t('settings.ai_test_ok', { ms: res.latency_ms }),
        }
      } catch (err) {
        state.testResult = {
          ok:      false,
          message: t('settings.ai_test_fail', { error: err.message }),
        }
      } finally {
        state.testing = false
      }
    }

    async function listModels() {
      state.listingModels = true
      state.models        = []
      try {
        const res    = await settingsApi.listModels()
        state.models = res.models
        if (state.models.length === 0) {
          store.showToast('Aucun modèle retourné par le provider.', 'info')
        }
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        state.listingModels = false
      }
    }

    function selectModel(modelId) {
      state.model = modelId
      saveField('ai_model', modelId)
    }

    return {
      state, saveField, testConnection, listModels, selectModel, t,
    }
  },

  template: `
    <div class="ai-form">
      <!-- Base URL -->
      <div class="form-group">
        <label class="form-label">{{ t('settings.ai_base_url') }}</label>
        <div style="display:flex;gap:0.5rem">
          <input
            class="form-input"
            type="url"
            v-model="state.base_url"
            :placeholder="t('settings.ai_url_hint')"
            @blur="saveField('ai_base_url', state.base_url)"
          />
        </div>
        <div style="font-size:0.75rem;color:var(--color-text-muted);margin-top:0.25rem">
          {{ t('settings.ai_url_hint') }}
        </div>
      </div>

      <!-- Modèle -->
      <div class="form-group">
        <label class="form-label">{{ t('settings.ai_model') }}</label>
        <div style="display:flex;gap:0.5rem">
          <input
            class="form-input"
            type="text"
            v-model="state.model"
            placeholder="gpt-4o-mini"
            @blur="saveField('ai_model', state.model)"
            style="flex:1"
          />
          <button class="btn btn--secondary"
            style="flex-shrink:0;padding:0.6rem 0.75rem;font-size:0.8rem"
            :disabled="!state.base_url || state.listingModels"
            @click="listModels()">
            {{ state.listingModels ? '…' : t('settings.ai_list_models') }}
          </button>
        </div>

        <!-- Liste des modèles disponibles -->
        <div v-if="state.models.length" class="ai-models-list">
          <button
            v-for="m in state.models" :key="m.id"
            class="ai-model-item"
            :class="{ 'ai-model-item--active': state.model === m.id }"
            @click="selectModel(m.id)">
            {{ m.name || m.id }}
          </button>
        </div>
      </div>

      <!-- Clé API -->
      <div class="form-group">
        <label class="form-label">{{ t('settings.ai_api_key') }}</label>
        <div style="display:flex;gap:0.5rem;align-items:center">
          <input
            class="form-input"
            type="password"
            autocomplete="off"
            v-model="state.api_key"
            :placeholder="currentSettings.ai_api_key_set
              ? t('settings.ai_api_key_set')
              : 'sk-...'"
            style="flex:1"
          />
          <button class="btn btn--secondary"
            style="flex-shrink:0;padding:0.6rem 0.75rem;font-size:0.8rem"
            :disabled="!state.api_key"
            @click="saveField('ai_api_key', state.api_key)">
            {{ t('common.save') }}
          </button>
        </div>
        <div style="font-size:0.75rem;margin-top:0.25rem"
          :style="{ color: currentSettings.ai_api_key_set
            ? 'var(--color-success)' : 'var(--color-text-muted)' }">
          {{ currentSettings.ai_api_key_set
            ? t('settings.ai_api_key_set')
            : t('settings.ai_api_key_not_set') }}
        </div>
      </div>

      <!-- Test de connexion -->
      <button class="btn btn--primary btn--full"
        :disabled="!state.base_url || !state.model || state.testing"
        @click="testConnection()">
        {{ state.testing ? t('settings.ai_testing') : t('settings.ai_test') }}
      </button>

      <!-- Résultat du test -->
      <div v-if="state.testResult" class="ai-test-result"
        :class="state.testResult.ok ? 'ai-test-result--ok' : 'ai-test-result--fail'">
        {{ state.testResult.message }}
      </div>
    </div>
  `
}
```

## `SettingsView.js` — vue principale

```javascript
import { reactive, onMounted, inject, watch }
  from '/vendor/vue.esm-browser.prod.js'
import { settings as settingsApi } from '../../api.js'
import { t, i18n } from '../../i18n.js'
import PageHeader       from '../layout/PageHeader.js'
import LoadingSpinner   from '../common/LoadingSpinner.js'
import ErrorBanner      from '../common/ErrorBanner.js'
import SettingsSection  from './SettingsSection.js'
import AIProviderForm   from './AIProviderForm.js'
import ExportImport     from './ExportImport.js'

export default {
  name: 'SettingsView',
  components: {
    PageHeader, LoadingSpinner, ErrorBanner,
    SettingsSection, AIProviderForm, ExportImport,
  },

  setup() {
    const store = inject('store')

    const state = reactive({
      loading:  false,
      error:    null,
      settings: null,
      saving:   {},    // { key: bool } — indique quelle clé est en cours de sauvegarde
      networkChanged: false,
    })

    // ── Chargement ───────────────────────────────────────────
    async function load() {
      state.loading = true; state.error = null
      try {
        state.settings = await settingsApi.get()
        store.setSettings(state.settings)
      } catch (err) {
        state.error = err.message
      } finally {
        state.loading = false
      }
    }

    // ── Sauvegarde d'un paramètre ────────────────────────────
    async function saveSetting(key, value) {
      state.saving[key] = true
      try {
        const res = await settingsApi.update({ [key]: value })
        state.settings = { ...state.settings, ...res }
        store.setSettings(state.settings)
        store.showToast(t('settings.saved'), 'success')

        // Appliquer immédiatement les changements UI
        if (key === 'theme') {
          document.documentElement.setAttribute('data-theme', value)
        }
        if (key === 'language') {
          await i18n.setLocale(value)
        }
        if (res.restart_required) {
          state.networkChanged = true
        }
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        state.saving[key] = false
      }
    }

    // ── Sauvegarde depuis AIProviderForm ─────────────────────
    async function onAiSaved(changes) {
      const res = await settingsApi.get()
      state.settings = res
      store.setSettings(res)
    }

    onMounted(load)
    return { state, load, saveSetting, onAiSaved, t }
  },

  template: `
    <div class="view">
      <PageHeader :title="t('settings.title')" />

      <div class="view__content">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner v-else-if="state.error" :message="state.error" :retry="load" />

        <template v-else-if="state.settings">

          <!-- ① Général ───────────────────────────────────── -->
          <SettingsSection :title="t('settings.section_general')" :default-open="true">
            <div class="setting-row">
              <label class="setting-row__label">{{ t('settings.unit') }}</label>
              <select class="form-select setting-row__control"
                :value="state.settings.unit"
                @change="saveSetting('unit', $event.target.value)">
                <option value="kg">{{ t('settings.unit_kg') }}</option>
                <option value="lb">{{ t('settings.unit_lb') }}</option>
              </select>
            </div>
            <div class="setting-row">
              <label class="setting-row__label">{{ t('settings.rm_formula') }}</label>
              <select class="form-select setting-row__control"
                :value="state.settings.rm_formula"
                @change="saveSetting('rm_formula', $event.target.value)">
                <option value="epley">Epley</option>
                <option value="brzycki">Brzycki</option>
                <option value="lander">Lander</option>
                <option value="oconner">O'Conner</option>
                <option value="lombardi">Lombardi</option>
              </select>
            </div>
            <div class="setting-row">
              <label class="setting-row__label">{{ t('settings.theme') }}</label>
              <select class="form-select setting-row__control"
                :value="state.settings.theme"
                @change="saveSetting('theme', $event.target.value)">
                <option value="dark">{{ t('settings.theme_dark') }}</option>
                <option value="light">{{ t('settings.theme_light') }}</option>
              </select>
            </div>
            <div class="setting-row">
              <label class="setting-row__label">{{ t('settings.language') }}</label>
              <select class="form-select setting-row__control"
                :value="state.settings.language"
                @change="saveSetting('language', $event.target.value)">
                <option value="fr">{{ t('settings.language_fr') }}</option>
                <option value="en">{{ t('settings.language_en') }}</option>
              </select>
            </div>
          </SettingsSection>

          <!-- ② Entraînement ──────────────────────────────── -->
          <SettingsSection :title="t('settings.section_training')">
            <div class="setting-row">
              <label class="setting-row__label">{{ t('settings.default_rest') }}</label>
              <input class="form-input setting-row__control setting-row__control--sm"
                type="number" min="10" max="600" step="10"
                :value="state.settings.default_rest_seconds"
                @change="saveSetting('default_rest_seconds', parseInt($event.target.value))" />
            </div>
            <div class="setting-row">
              <label class="setting-row__label">{{ t('settings.linear_increment') }}</label>
              <input class="form-input setting-row__control setting-row__control--sm"
                type="number" min="0.5" max="10" step="0.5"
                :value="state.settings.linear_increment_kg"
                @change="saveSetting('linear_increment_kg', parseFloat($event.target.value))" />
            </div>
            <div class="setting-row">
              <label class="setting-row__label">{{ t('settings.dumbbell_increment') }}</label>
```

## Étapes d'implémentation

1. Ajouter toutes les clés i18n dans `fr.json` et `en.json`.
2. Créer `SettingsSection.js`.
3. Créer `AIProviderForm.js`.
4. Créer `ExportImport.js` (placeholder — sera remplacé en T6.3).
5. Créer `SettingsView.js`.
6. Ajouter le CSS dans `components.css`.
7. Modifier `AppShell.js`.

**Points d'attention :**
- Chaque paramètre est sauvegardé **immédiatement** au `@change` ou
  `@blur` — pas de bouton "Sauvegarder global". C'est le comportement
  UX attendu sur mobile.
- Le changement de thème `dark/light` est appliqué immédiatement :
  `document.documentElement.setAttribute('data-theme', value)`.
  Vérifier que `public/css/themes.css` ou `main.css` utilise
  `[data-theme="light"]` pour le thème clair. Si ce fichier n'existe
  pas encore, créer des variables CSS minimales pour le thème clair :
  ```css
  [data-theme="light"] {
    --color-bg:        #f5f5f5;
    --color-surface:   #ffffff;
    --color-surface-2: #f0f0f0;
    --color-border:    #ddd;
    --color-text:      #1a1a1a;
    --color-text-muted: #666;
  }
  ```
- Le changement de langue appelle `await i18n.setLocale(value)` —
  s'assurer que `i18n` est importé dans `SettingsView.js` :
  ```javascript
  import { t, i18n } from '../../i18n.js'
  ```
- `AIProviderForm` sauvegarde le champ `ai_api_key` seulement quand
  l'utilisateur clique "Sauvegarder" — pas au `@blur`. Le champ est
  `type="password"` avec `autocomplete="off"`.
- `ExportImport` est un placeholder dans cette tâche — il sera complété
  en T6.3.
