## Objectif

1. **Premier essai** : si `progression_state.current_weight_kg === 0`, calculer une
   suggestion de poids et reps via un algorithme déterministe basé sur le profil
   utilisateur. Afficher cette suggestion pré-remplie dans SetRow avec une explication.
2. **Essais suivants** : pré-remplir avec le dernier poids/reps + afficher une bulle `ℹ️`
   qui explique la cible et la logique de progression du moteur.
3. **Colonne 10RM** : afficher le 10RM calculé (lecture seule) sur chaque série validée.
4. **Layout SetRow** : `# | KG | REPS | 10RM`

```javascript
// Reps pour exercices au poids du corps (pas de charge externe)
const BODYWEIGHT_REPS = {
  strength:        { beginner: 3,  intermediate: 5,  advanced: 8,  athlete: 12 },
  hypertrophy:     { beginner: 5,  intermediate: 8,  advanced: 12, athlete: 15 },
  endurance:       { beginner: 10, intermediate: 15, advanced: 20, athlete: 25 },
  weight_loss:     { beginner: 8,  intermediate: 12, advanced: 15, athlete: 20 },
  general_fitness: { beginner: 6,  intermediate: 10, advanced: 15, athlete: 20 },
  sport_performance:{ beginner: 5, intermediate: 8,  advanced: 12, athlete: 15 },
}

/**
 * Calcule la suggestion de poids et reps pour un exercice effectué pour la première fois.
 *
 * Algorithme DÉTERMINISTE — aucun appel LLM.
 *
 * @param {object} exercise      - { id, mechanic, equipment_required, is_unilateral }
 * @param {object} userProfile   - { weight_kg, fitness_level, primary_goal }
 * @param {object} templateExercise - { target_reps_min, target_reps_max }
 * @param {object[]} equipmentItems - inventaire complet
 * @returns {{
 *   weight_kg: number,
 *   reps: number,
 *   reasoning: string     // texte affiché dans la bulle
 * }}
 */
export function suggestFirstTime(exercise, userProfile, templateExercise, equipmentItems) {
  const bodyweightKg = userProfile?.weight_kg || 70
  const level        = userProfile?.fitness_level || 'beginner'
  const goal         = userProfile?.primary_goal  || 'general_fitness'
  const mechanic     = exercise.mechanic === 'compound' ? 'compound' : 'isolation'

  // ── Exercice sans charge externe (poids du corps) ───────────
  const isBodyweight = !exercise.equipment_required ||
                       exercise.equipment_required.length === 0

  if (isBodyweight) {
    const reps = (BODYWEIGHT_REPS[goal] || BODYWEIGHT_REPS.general_fitness)[level] || 8
    // Utiliser les reps de la fourchette du template si disponibles
    const templateReps = templateExercise?.target_reps_min || reps
    return {
      weight_kg: 0,
      reps:      templateReps,
      reasoning: buildReasoning(goal, level, mechanic, 0, templateReps, bodyweightKg, true),
    }
  }

  // ── Exercice avec charge ─────────────────────────────────────
  const table = (FIRST_TIME_TABLE[goal] || FIRST_TIME_TABLE.general_fitness)[level]
  const config = table?.[mechanic] || { pct: 0.25, reps: 10 }

  const rawWeight   = bodyweightKg * config.pct
  const loadsResult = calculateAchievableLoads(equipmentItems, exercise)

  let weight
  if (loadsResult.loads_kg.length === 0) {
    // Aucune charge disponible (matériel manquant)
    weight = 0
  } else {
    weight = roundDownToAchievable(rawWeight, loadsResult.loads_kg)
    // Si la charge suggérée est la charge minimale, l'utiliser quand même
    if (weight <= 0) weight = loadsResult.loads_kg[0]
  }

  // Utiliser les reps de la fourchette du template si disponibles
  const reps = templateExercise?.target_reps_min || config.reps

  return {
    weight_kg: weight,
    reps,
    reasoning: buildReasoning(goal, level, mechanic, weight, reps, bodyweightKg, false),
  }
}

/**
 * Construit le texte d'explication pour la bulle.
 */
function buildReasoning(goal, level, mechanic, weightKg, reps, bodyweightKg, isBodyweight) {
  const goalLabels = {
    strength: 'force', hypertrophy: 'hypertrophie', endurance: 'endurance',
    weight_loss: 'perte de poids', general_fitness: 'forme générale',
    sport_performance: 'performance sportive',
  }
  const levelLabels = {
    beginner: 'débutant', intermediate: 'intermédiaire',
    advanced: 'avancé', athlete: 'athlète',
  }
  const mechanicLabel = mechanic === 'compound' ? 'composé' : 'isolation'

  if (isBodyweight) {
    return `Premier essai · Niveau ${levelLabels[level] || level}, objectif ${goalLabels[goal] || goal}. ` +
           `Exercice au poids du corps : commencez avec ${reps} répétitions propres. ` +
           `Ajustez selon votre ressenti.`
  }

  const pct = Math.round((weightKg / bodyweightKg) * 100)
  return `Premier essai · Niveau ${levelLabels[level] || level}, objectif ${goalLabels[goal] || goal}. ` +
         `Exercice ${mechanicLabel} : ${weightKg} kg (≈ ${pct}% de votre poids de corps) × ${reps} reps. ` +
         `C'est une charge prudente pour trouver votre niveau — ajustez à votre ressenti.`
}
```

## Modification de `src/routes/sessions.js`

### Remplacer entièrement `GET /:id/target/:teId`

```javascript
import { suggestFirstTime } from '../services/firstTimeSuggestion.js'
import { estimate1RM }      from '../services/rmCalculator.js'

router.get('/:id/target/:teId', (req, res) => {
  try {
    const session = db.prepare('SELECT * FROM session WHERE id = ?').get(req.params.id)
    if (!session) return res.status(404).json({ error: 'SESSION_NOT_FOUND' })

    const te = db.prepare('SELECT * FROM template_exercise WHERE id = ?').get(req.params.teId)
    if (!te) return res.status(404).json({ error: 'TEMPLATE_EXERCISE_NOT_FOUND' })

    const state = db.prepare(
      'SELECT * FROM progression_state WHERE template_exercise_id = ?'
    ).get(req.params.teId)

    const exercise = db.prepare('SELECT * FROM exercise WHERE id = ?').get(te.exercise_id)
    const exerciseForCalc = {
      equipment_required: JSON.parse(exercise?.equipment_required || '[]'),
      is_unilateral:      exercise?.is_unilateral || 0,
      mechanic:           exercise?.mechanic,
    }

    const equipmentItems = db.prepare('SELECT * FROM equipment_item').all()
    const loadsResult    = calculateAchievableLoads(equipmentItems, exerciseForCalc)

    // ── Détection premier essai ───────────────────────────────
    const isFirstTime = !state || state.current_weight_kg === 0

    let weight, targetReps, explanation, suggestion

    if (isFirstTime) {
      // Charger le profil utilisateur
      const profileRow = db.prepare('SELECT * FROM user_profile WHERE id = 1').get()
      const profile = profileRow ? {
        weight_kg:     profileRow.weight_kg,
        fitness_level: profileRow.fitness_level,
        primary_goal:  profileRow.primary_goal,
      } : null

      const parsed = {
        ...exerciseForCalc,
        id:      exercise?.id,
        mechanic: exercise?.mechanic,
      }

      suggestion = suggestFirstTime(parsed, profile, te, equipmentItems)
      weight      = suggestion.weight_kg
      targetReps  = suggestion.reps
      explanation = suggestion.reasoning

    } else {
      // Essais suivants : lire depuis progression_state
      weight      = state.current_weight_kg
      targetReps  = state.current_target_reps || te.target_reps_min

      // Récupérer la dernière performance réelle
      const lastSet = db.prepare(`
        SELECT sl.weight_kg, sl.reps, s.scheduled_date
        FROM set_log sl
        JOIN session_exercise se ON sl.session_exercise_id = se.id
        JOIN session s ON se.session_id = s.id
        WHERE se.exercise_id = ?
          AND se.template_exercise_id = ?
          AND s.status = 'completed'
          AND sl.is_warmup = 0
          AND sl.is_completed = 1
        ORDER BY s.completed_at DESC, sl.set_number DESC
        LIMIT 1
      `).get(te.exercise_id, req.params.teId)

      const program = session.program_id
        ? db.prepare('SELECT * FROM program WHERE id = ?').get(session.program_id)
        : null

      const { resolveProgressionModel } = await import('../services/progressionEngine.js')
      const model = resolveProgressionModel(te, program)

      const modelLabels = {
        linear:             'progression linéaire',
        double_progression: 'double progression',
        rpe_based:          'progression RPE',
      }

      if (lastSet) {
        const failureInfo = state.consecutive_failures > 0
          ? ` (${state.consecutive_failures} échec(s) consécutif(s))`
          : ''
        explanation = `Dernière séance (${lastSet.scheduled_date}) : ` +
                      `${lastSet.weight_kg} kg × ${lastSet.reps} reps. ` +
                      `Cible : ${weight} kg × ${targetReps} reps ` +
                      `[${modelLabels[model] || model}]${failureInfo}.`
      } else {
        explanation = `Charge initiale : ${weight} kg × ${targetReps} reps [${modelLabels[model] || model}].`
      }
    }

    // ── Calcul du 10RM estimé pour la charge cible ────────────
    // 10RM = 1RM / (1 + 10/30) = 1RM / 1.333
    // Si reps = 10 → 10RM ≈ weight
    const formulaRow = db.prepare("SELECT value FROM settings WHERE key = 'rm_formula'").get()
    const formula    = formulaRow ? JSON.parse(formulaRow.value) : 'epley'
    const est1rm     = weight > 0 && targetReps > 0
      ? estimate1RM(weight, targetReps, formula)
      : 0
    const est10rm    = est1rm > 0
      ? Math.round((est1rm / (1 + 10 / 30)) * 10) / 10
      : null

    res.json({
      weight_kg:           weight,
      reps_min:            targetReps,
      reps_max:            te.target_reps_max,
      explanation,
      achievable_loads_kg: loadsResult.loads_kg,
      is_first_time:       isFirstTime,
      // Bulle "suggestion premier essai" — null si essai suivant
      first_time_suggestion: isFirstTime ? {
        weight_kg: suggestion.weight_kg,
        reps:      suggestion.reps,
        reasoning: suggestion.reasoning,
      } : null,
      // 10RM estimé de la charge cible
      estimated_10rm_kg: est10rm,
    })
  } catch (err) {
    console.error('[GET /sessions/:id/target/:teId]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})
```

## Mise à jour de `public/js/components/session/SetRow.js`

```javascript
import { reactive, inject, computed } from '/vendor/vue.esm-browser.prod.js'
import { sets as api }  from '../../api.js'
import { estimate1RM }  from '../../utils/rm.js'
import { t }            from '../../i18n.js'
import AITooltip        from './AITooltip.js'

export default {
  name: 'SetRow',
  components: { AITooltip },
  props: {
    set:               { type: Object,  required: true },
    sessionExerciseId: { type: Number,  required: true },
    targetWeight:      { type: Number,  default: 0 },
    targetReps:        { type: Number,  default: 8 },
    setNumber:         { type: Number,  required: true },
    isNew:             { type: Boolean, default: false },
    restSeconds:       { type: Number,  default: 90 },
    // Champs bulle / suggestion
    isFirstTime:       { type: Boolean, default: false },
    tooltipText:       { type: String,  default: '' },  // texte pour la bulle ℹ️
  },
  emits: ['validated', 'rest-start'],

  setup(props, { emit }) {
    const store = inject('store')

    const local = reactive({
      weight_kg: props.isNew ? props.targetWeight : (props.set.weight_kg ?? props.targetWeight),
      reps:      props.isNew ? props.targetReps   : (props.set.reps      ?? props.targetReps),
      rpe:       props.set.rpe  || '',
      saving:    false,
    })

    const showRpe = () => store.getSetting('show_rpe_field', false)

    // ── Calcul du 10RM en temps réel ─────────────────────────
    const formula = () => store.getSetting('rm_formula', 'epley')

    const live10rm = computed(() => {
      const w = parseFloat(local.weight_kg) || 0
      const r = parseInt(local.reps)        || 0
      if (w <= 0 || r <= 0) return null
      const rm1 = estimate1RM(w, r, formula())
      if (!rm1) return null
      return Math.round((rm1 / (1 + 10 / 30)) * 10) / 10
    })

    // 10RM affiché pour une série déjà validée
    const saved10rm = computed(() => {
      if (!props.set.is_completed || !props.set.weight_kg || !props.set.reps) return null
      const rm1 = estimate1RM(props.set.weight_kg, props.set.reps, formula())
      if (!rm1) return null
      return Math.round((rm1 / (1 + 10 / 30)) * 10) / 10
    })

    async function validate() {
      if (local.saving) return
      local.saving = true
      try {
        let result
        if (props.isNew) {
          result = await api.create(props.sessionExerciseId, {
            set_number:   props.setNumber,
            weight_kg:    parseFloat(local.weight_kg) || 0,
            reps:         parseInt(local.reps)         || 0,
            rpe:          local.rpe ? parseFloat(local.rpe) : null,
            is_completed: true,
            is_warmup:    false,
          })
        } else {
          result = await api.update(props.set.id, {
            weight_kg:    parseFloat(local.weight_kg) || 0,
            reps:         parseInt(local.reps)         || 0,
            rpe:          local.rpe ? parseFloat(local.rpe) : null,
            is_completed: true,
          })
        }
        if (result._new_pr) {
          store.showToast(
            t('session.pr_toast', {
              weight: result.weight_kg,
              reps:   result.reps,
              rm:     result._estimated_1rm,
            }),
            'success',
            5000
          )
        }
        emit('validated', result)
        emit('rest-start', props.restSeconds)
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        local.saving = false
      }
    }

    return { local, validate, showRpe, live10rm, saved10rm, t }
  },

  template: `
    <div class="set-row" :class="{ 'set-row--done': set.is_completed && !isNew }">

      <!-- # -->
      <span class="set-row__num">{{ setNumber }}</span>

      <!-- KG + bulle tooltip -->
      <div class="set-row__kg-wrap">
        <input
          class="set-row__input"
          type="number" step="0.5" min="0"
          v-model="local.weight_kg"
          :disabled="set.is_completed && !isNew"
        />
        <!-- Bulle : visible si texte disponible -->
        <AITooltip
          v-if="tooltipText"
          :text="tooltipText"
          :is-first="isFirstTime"
        />
      </div>

      <!-- REPS -->
      <input
        class="set-row__input set-row__input--reps"
        type="number" min="0"
        v-model="local.reps"
        :disabled="set.is_completed && !isNew"
      />

      <!-- 10RM (calculé en temps réel si non validé, depuis les valeurs enregistrées sinon) -->
      <span class="set-row__10rm">
        <template v-if="set.is_completed && !isNew">
          {{ saved10rm !== null ? saved10rm + ' kg' : '—' }}
        </template>
        <template v-else>
          {{ live10rm !== null ? live10rm + ' kg' : '—' }}
        </template>
      </span>

      <!-- RPE optionnel -->
      <input v-if="showRpe() && (isNew || !set.is_completed)"
        class="set-row__input set-row__input--rpe"
        type="number" step="0.5" min="6" max="10"
        placeholder="RPE"
        v-model="local.rpe"
      />

      <!-- Bouton valider / check -->
      <button
        v-if="isNew || !set.is_completed"
        class="set-row__btn"
        :disabled="local.saving"
        @click="validate()">
        {{ local.saving ? '…' : '✓' }}
      </button>
      <span v-else class="set-row__done">✓</span>
    </div>
  `
}
```

**Points d'attention :**
- `import { resolveProgressionModel }` dans `sessions.js` : c'est un import dynamique
  (`await import(...)`) car le fichier est déjà importé statiquement ailleurs.
  Mieux : exporter `resolveProgressionModel` comme named export de `progressionEngine.js`

- `suggestFirstTime` ne doit jamais lever d'exception — wrapper dans try/catch dans la route,
  retourner `{ weight_kg: 0, reps: 8, reasoning: 'Aucun profil configuré. Commencez léger.' }` en cas d'erreur.

### `tests/unit/firstTimeSuggestion.test.js`

```javascript
import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import { suggestFirstTime } from '../../src/services/firstTimeSuggestion.js'

const BENCH = {
  id: 'bench_press_barbell', mechanic: 'compound',
  equipment_required: ['barbell','bench'], is_unilateral: 0,
}
const CURL = {
  id: 'curl_barbell', mechanic: 'isolation',
  equipment_required: ['barbell'], is_unilateral: 0,
}
const PULLUP = {
  id: 'pullup_bodyweight', mechanic: 'compound',
  equipment_required: [], is_unilateral: 0,
}

const BARBELL = { category: 'barbell', weight_kg: 20, count: 1 }
const PLATES  = [
  { category: 'plate', weight_kg: 5,  count: 4 },
  { category: 'plate', weight_kg: 10, count: 4 },
  { category: 'plate', weight_kg: 20, count: 2 },
]
const EQUIPMENT = [BARBELL, ...PLATES,
  { category: 'bench', name: 'Banc', count: 1 }]

const PROFILE_BEGINNER_HYPER = {
  weight_kg: 80, fitness_level: 'beginner', primary_goal: 'hypertrophy'
}
const PROFILE_INTER_STRENGTH = {
  weight_kg: 80, fitness_level: 'intermediate', primary_goal: 'strength'
}

describe('suggestFirstTime', () => {
  it('débutant hypertrophie bench → charge > 0 et reps 10', () => {
    const r = suggestFirstTime(BENCH, PROFILE_BEGINNER_HYPER, { target_reps_min: 10 }, EQUIPMENT)
    assert.ok(r.weight_kg > 0, 'Charge doit être > 0')
    assert.equal(r.reps, 10)
    assert.ok(typeof r.reasoning === 'string' && r.reasoning.length > 10)
  })

  it('charge arrondie à une valeur atteignable avec l\'équipement', () => {
    const r = suggestFirstTime(BENCH, PROFILE_BEGINNER_HYPER, { target_reps_min: 10 }, EQUIPMENT)
    // Charges disponibles : 20, 30, 40, 50, 60, 70, 80...
    // 80 * 0.25 = 20 → 20 est atteignable (barre seule)
    const loads = [20, 30, 40, 50, 60, 70, 80]
    assert.ok(loads.includes(r.weight_kg), `${r.weight_kg} doit être dans les charges atteignables`)
  })

  it('exercice bodyweight → weight_kg = 0', () => {
    const r = suggestFirstTime(PULLUP, PROFILE_BEGINNER_HYPER, { target_reps_min: 5 }, [])
    assert.equal(r.weight_kg, 0)
    assert.ok(r.reps > 0)
  })

  it('intermédiaire force → plus de reps que débutant pour isolation', () => {
    // Non — pour strength, moins de reps sont normales
    const rBeg  = suggestFirstTime(CURL, PROFILE_BEGINNER_HYPER,  { target_reps_min: 12 }, EQUIPMENT)
    const rInt  = suggestFirstTime(CURL, PROFILE_INTER_STRENGTH,  { target_reps_min: 8  }, EQUIPMENT)
    // Intermédiaire strength isolation : 8 reps
    assert.equal(rInt.reps, 8)
    // Débutant hypertrophie isolation : 12 reps
    assert.equal(rBeg.reps, 12)
  })

  it('profil null → valeur par défaut sans crash', () => {
    assert.doesNotThrow(() => {
      suggestFirstTime(BENCH, null, { target_reps_min: 10 }, EQUIPMENT)
    })
  })

  it('équipement vide → weight_kg = charge minimale dispo ou 0', () => {
    const r = suggestFirstTime(BENCH, PROFILE_BEGINNER_HYPER, { target_reps_min: 10 }, [])
    assert.equal(r.weight_kg, 0)
  })
})
```
