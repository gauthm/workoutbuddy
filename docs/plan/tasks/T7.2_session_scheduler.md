## Objectif

Générer automatiquement les séances planifiées pour la **semaine courante +
la semaine suivante** à chaque événement clé (activation d'un programme,
complétion d'une séance, démarrage du serveur). L'utilisateur voit toujours
ses 2 prochaines semaines pré-planifiées dans le calendrier.

## `src/services/sessionScheduler.js` — implémentation complète

```javascript
/**
 * Génère les séances planifiées pour la semaine courante + la semaine suivante.
 *
 * Règles :
 * - Ne crée PAS de séance si une séance 'planned' ou 'in_progress' existe déjà
 *   pour ce template sur ce jour ou cette semaine.
 * - Ne crée PAS de séance dans le passé (avant aujourd'hui).
 * - Distribue les séances sur la semaine selon days_per_week et les templates.
 * - Respecte l'ordre des templates (A → B → C → A → ...).
 * - Séances au format : status='planned', scheduled_date=YYYY-MM-DD.
 *
 * @param {object} db       - instance better-sqlite3
 * @param {number} programId - ID du programme actif (optionnel : lit l'actif si absent)
 * @returns {number} nombre de séances créées
 */
export function scheduleWeeks(db, programId = null) {
  try {
    // Trouver le programme actif
    const program = programId
      ? db.prepare('SELECT * FROM program WHERE id = ? AND is_active = 1').get(programId)
      : db.prepare('SELECT * FROM program WHERE is_active = 1 LIMIT 1').get()

    if (!program) return 0

    // Ne pas planifier si le programme est en deload
    // (le deload dure une semaine — on planifie quand même pour que l'utilisateur voie)

    const templates = db.prepare(
      'SELECT * FROM workout_template WHERE program_id = ? ORDER BY order_index ASC'
    ).all(program.id)

    if (templates.length === 0) return 0

    const today       = todayIso()
    const endOfRange  = endOfWeek(addWeeks(today, 1))  // fin de la semaine suivante

    // Séances déjà planifiées dans la plage
    const existing = db.prepare(`
      SELECT scheduled_date, workout_template_id
      FROM session
      WHERE program_id = ?
        AND status IN ('planned','in_progress')
        AND scheduled_date BETWEEN ? AND ?
    `).all(program.id, today, endOfRange)

    const existingDates = new Set(existing.map(s => s.scheduled_date))
    const existingTemplateIds = new Set(existing.map(s => s.workout_template_id))

    // Déterminer quel template utiliser en premier
    const lastCompleted = db.prepare(`
      SELECT workout_template_id FROM session
      WHERE program_id = ? AND status = 'completed'
      ORDER BY completed_at DESC LIMIT 1
    `).get(program.id)

    let nextTemplateIdx = 0
    if (lastCompleted?.workout_template_id) {
      const idx = templates.findIndex(t => t.id === lastCompleted.workout_template_id)
      if (idx !== -1) nextTemplateIdx = (idx + 1) % templates.length
    }

    // Calculer les jours disponibles dans la plage
    const daysInRange = getDaysInRange(today, endOfRange)

    // Distribuer les séances : days_per_week réparties sur chaque semaine
    const slots = distributeSlots(daysInRange, program.days_per_week, existingDates)

    let created = 0
    for (const date of slots) {
      const template = templates[nextTemplateIdx % templates.length]
      nextTemplateIdx++

      db.prepare(`
        INSERT INTO session (name, scheduled_date, workout_template_id, program_id, status)
        VALUES (?, ?, ?, ?, 'planned')
      `).run(template.name, date, template.id, program.id)

      created++
    }

    return created

  } catch (err) {
    console.warn('[sessionScheduler] Erreur :', err.message)
    return 0  // Non bloquant
  }
}

// ── Helpers date ──────────────────────────────────────────────

/** YYYY-MM-DD d'aujourd'hui */
function todayIso() {
  return new Date().toISOString().slice(0, 10)
}

/** Ajoute N semaines à une date ISO */
function addWeeks(dateIso, n) {
  const d = new Date(dateIso + 'T12:00:00Z')
  d.setUTCDate(d.getUTCDate() + n * 7)
  return d.toISOString().slice(0, 10)
}

/** Dernier jour (dimanche) de la semaine contenant dateIso */
function endOfWeek(dateIso) {
  const d   = new Date(dateIso + 'T12:00:00Z')
  const dow = d.getUTCDay()  // 0=dimanche
  const diff = dow === 0 ? 0 : 7 - dow
  d.setUTCDate(d.getUTCDate() + diff)
  return d.toISOString().slice(0, 10)
}

/** Liste tous les jours YYYY-MM-DD entre start et end (inclus) */
function getDaysInRange(start, end) {
  const days = []
  const current = new Date(start + 'T12:00:00Z')
  const endDate = new Date(end   + 'T12:00:00Z')
  while (current <= endDate) {
    days.push(current.toISOString().slice(0, 10))
    current.setUTCDate(current.getUTCDate() + 1)
  }
  return days
}

/**
 * Choisit les jours où planifier des séances.
 *
 * Stratégie : répartir les jours de façon uniforme sur chaque semaine.
 * - Pour days_per_week = 3 : Lun, Mer, Ven (indices 0, 2, 4 de la semaine)
 * - Pour days_per_week = 4 : Lun, Mar, Jeu, Ven
 * - Pour days_per_week = 5 : Lun, Mar, Mer, Jeu, Ven
 * - Pour days_per_week = 6 : Lun, Mar, Mer, Jeu, Ven, Sam
 * - Pour days_per_week = 2 : Lun, Jeu
 *
 * Ignorer les jours déjà dans existingDates.
 *
 * @param {string[]} daysInRange   - tous les jours de la plage
 * @param {number}   daysPerWeek   - séances par semaine (2–6)
 * @param {Set}      existingDates - jours déjà planifiés (à ignorer)
 * @returns {string[]} dates des séances à créer
 */
function distributeSlots(daysInRange, daysPerWeek, existingDates) {
  // Indices de jours de semaine (0=Lun, 6=Dim) selon days_per_week
  const PREFERRED_DAYS = {
    2: [0, 3],           // Lun, Jeu
    3: [0, 2, 4],        // Lun, Mer, Ven
    4: [0, 1, 3, 4],     // Lun, Mar, Jeu, Ven
    5: [0, 1, 2, 3, 4],  // Lun au Ven
    6: [0, 1, 2, 3, 4, 5], // Lun au Sam
    7: [0, 1, 2, 3, 4, 5, 6],
  }

  const preferred = PREFERRED_DAYS[daysPerWeek] || PREFERRED_DAYS[3]
  const slots     = []

  // Regrouper les jours par semaine ISO
  const byWeek = {}
  for (const day of daysInRange) {
    const weekKey = isoWeekKey(day)
    if (!byWeek[weekKey]) byWeek[weekKey] = []
    byWeek[weekKey].push(day)
  }

  // Pour chaque semaine, sélectionner les jours préférés non encore planifiés
  for (const [, days] of Object.entries(byWeek).sort()) {
    let addedThisWeek = 0
    for (const day of days) {
      if (addedThisWeek >= daysPerWeek) break
      if (existingDates.has(day)) { addedThisWeek++; continue }

      const d   = new Date(day + 'T12:00:00Z')
      const dow = (d.getUTCDay() + 6) % 7  // 0=Lun, 6=Dim

      if (preferred.includes(dow)) {
        slots.push(day)
        existingDates.add(day)  // éviter les doublons
        addedThisWeek++
      }
    }
  }

  return slots
}

/** Retourne "YYYY-Www" pour une date ISO */
function isoWeekKey(dateIso) {
  const d = new Date(dateIso + 'T12:00:00Z')
  d.setUTCDate(d.getUTCDate() + 3 - ((d.getUTCDay() + 6) % 7))
  const week1 = new Date(Date.UTC(d.getUTCFullYear(), 0, 4))
  const weekNum = 1 + Math.round(
    ((d - week1) / 86400000 - 3 + (week1.getUTCDay() + 6) % 7) / 7
  )
  return `${d.getUTCFullYear()}-W${String(weekNum).padStart(2, '0')}`
}
```

## Exemples chiffrés

### Exemple 1 — Programme PPL 3j/sem, activation lundi 7 juillet

```
Semaine courante (7–13 juillet) :
  Jours préférés : Lun 7, Mer 9, Ven 11
  Aucune séance existante
  → Créer : Jour A (Push) le 7, Jour B (Pull) le 9, Jour C (Legs) le 11

Semaine suivante (14–20 juillet) :
  Jours préférés : Lun 14, Mer 16, Ven 18
  → Créer : Jour A (Push) le 14, Jour B (Pull) le 16, Jour C (Legs) le 18

Total créé : 6 séances
```

### Exemple 2 — Séance complétée mercredi, déjà 2 séances planifiées cette semaine

```
Séance du mer 9 → complétée
Déjà planifiés : ven 11 (cette semaine), lun 14 et mer 16 (semaine suivante)
→ scheduleWeeks vérifie la plage aujourd'hui → fin semaine suivante
→ existingDates = {11/07, 14/07, 16/07}
→ Slots manquants semaine suivante : ven 18 → créer 1 séance
Total créé : 1 séance
```

### Exemple 3 — Programme 4j/sem, Upper/Lower

```
Templates : Upper A, Lower A, Upper B, Lower B
Jours préférés 4j : Lun, Mar, Jeu, Ven
lastCompleted = Lower A → nextTemplateIdx = 2 (Upper B)

Semaine courante : Upper B (Lun), Lower B (Mar), Upper A (Jeu), Lower A (Ven)
Semaine suivante : Upper B (Lun), Lower B (Mar), Upper A (Jeu), Lower A (Ven)
Total créé : 8 séances (si aucune n'existe)
```

## Étapes d'implémentation

1. Créer `src/services/sessionScheduler.js`.
2. Modifier `programs.js` : import + appel dans `POST /:id/activate`.
3. Modifier `sessions.js` : import + appel dans `POST /:id/complete`.
4. Modifier `server.js` : import + appel dans `bootstrap()`.

**Points d'attention :**
- `scheduleWeeks` est **toujours non bloquant** — encapsuler les appels dans `try/catch`.
- Si aucun programme actif → retourne `0` silencieusement.
- La fonction vérifie `status IN ('planned','in_progress')` — ne compte pas
  les séances skipped ou completed comme "déjà planifiées".
- `distributeSlots` opère par semaine ISO — un programme 3j/sem crée
  3 séances par semaine, pas 3 sur toute la plage.
- Si le programme est activé en milieu de semaine (ex: jeudi), la semaine
  courante aura moins de séances que prévu (les jours passés sont ignorés).
- `endOfWeek` retourne le dimanche de la semaine contenant `addWeeks(today, 1)` —
  soit au maximum 16 jours dans le futur (aujourd'hui = lundi → fin semaine suivante = dimanche +14).
- Ne jamais créer deux séances le même jour pour le même programme.

### `tests/unit/sessionScheduler.test.js`

```javascript
import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import Database from 'better-sqlite3'
import { migrate } from '../../src/db/migrate.js'
import { seedExercises } from '../../src/seed/seed.js'
import { scheduleWeeks } from '../../src/services/sessionScheduler.js'

function createDb() {
  const db = new Database(':memory:')
  db.pragma('foreign_keys = ON')
  migrate(db)
  seedExercises(db)
  return db
}

function createActiveProgram(db, daysPerWeek = 3) {
  const pRes = db.prepare(`
    INSERT INTO program (name, goal, level, days_per_week, progression_model, is_active)
    VALUES ('Test PPL', 'hypertrophy', 'intermediate', ?, 'double_progression', 1)
  `).run(daysPerWeek)
  const pid = pRes.lastInsertRowid

  const names = ['Push', 'Pull', 'Legs', 'Upper', 'Lower']
  for (let i = 0; i < daysPerWeek; i++) {
    db.prepare('INSERT INTO workout_template (program_id, name, order_index) VALUES (?,?,?)')
      .run(pid, names[i] || `Jour ${i+1}`, i)
  }
  return pid
}

describe('sessionScheduler', () => {
  it('pas de programme actif → retourne 0', () => {
    const db = createDb()
    assert.equal(scheduleWeeks(db), 0)
  })

  it('crée des séances pour 2 semaines', () => {
    const db  = createDb()
    const pid = createActiveProgram(db, 3)
    const n   = scheduleWeeks(db, pid)
    // 3j/sem × 2 semaines max = 6, mais peut être moins si semaine entamée
    assert.ok(n >= 1 && n <= 6, `Attendu 1-6, obtenu ${n}`)
  })

  it('idempotent : 2e appel ne crée rien', () => {
    const db  = createDb()
    const pid = createActiveProgram(db, 3)
    scheduleWeeks(db, pid)
    const n2 = scheduleWeeks(db, pid)
    assert.equal(n2, 0, 'Le 2e appel ne doit créer aucune séance')
  })

  it('les séances créées sont toutes status=planned', () => {
    const db  = createDb()
    const pid = createActiveProgram(db, 3)
    scheduleWeeks(db, pid)
    const notPlanned = db.prepare(
      "SELECT COUNT(*) as n FROM session WHERE program_id = ? AND status != 'planned'"
    ).get(pid)
    assert.equal(notPlanned.n, 0)
  })

  it('les dates créées sont >= aujourd\'hui', () => {
    const db    = createDb()
    const pid   = createActiveProgram(db, 3)
    scheduleWeeks(db, pid)
    const today = new Date().toISOString().slice(0, 10)
    const pastSessions = db.prepare(
      'SELECT COUNT(*) as n FROM session WHERE program_id = ? AND scheduled_date < ?'
    ).get(pid, today)
    assert.equal(pastSessions.n, 0)
  })

  it('les templates alternent dans l\'ordre', () => {
    const db  = createDb()
    const pid = createActiveProgram(db, 3)
    scheduleWeeks(db, pid)
    const sessions = db.prepare(
      'SELECT name FROM session WHERE program_id = ? ORDER BY scheduled_date ASC'
    ).all(pid)
    // Les 3 premiers doivent être Push, Pull, Legs
    const names = sessions.slice(0, 3).map(s => s.name)
    assert.deepEqual(names, ['Push', 'Pull', 'Legs'])
  })

  it('programme 4j/sem crée jusqu\'à 8 séances sur 2 semaines', () => {
    const db  = createDb()
    const pid = createActiveProgram(db, 4)
    const n   = scheduleWeeks(db, pid)
    assert.ok(n >= 1 && n <= 8, `Attendu 1-8, obtenu ${n}`)
  })
})
```
