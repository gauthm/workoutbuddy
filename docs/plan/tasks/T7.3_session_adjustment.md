## Objectif

Quand l'utilisateur lance une séance (`POST /sessions/:id/start`), calculer
les cibles actualisées pour **tous** les exercices de cette séance en une
seule passe et retourner un résumé des ajustements. Si les cibles ont changé
depuis la création de la séance (ex : la séance précédente a été complétée
depuis), afficher un bandeau "Votre coach a ajusté cette séance" avec le
détail des changements.

**Important** : le calcul est **entièrement déterministe** (moteur de
progression + suggestions premier essai). Aucun appel LLM dans ce flux.
L'IA intervient seulement via le chat coach si l'utilisateur veut plus de détail.

### Remplacer `POST /:id/start` entièrement

```javascript
import { computeNextState, resolveProgressionModel } from '../services/progressionEngine.js'
import { calculateAchievableLoads }                  from '../services/equipmentCalc.js'
import { suggestFirstTime }                          from '../services/firstTimeSuggestion.js'

router.post('/:id/start', (req, res) => {
  try {
    const session = db.prepare('SELECT * FROM session WHERE id = ?').get(req.params.id)
    if (!session) return res.status(404).json({ error: 'SESSION_NOT_FOUND' })
    if (session.status === 'completed') {
      return res.status(409).json({ error: 'SESSION_ALREADY_COMPLETED' })
    }

    // Vérifier qu'aucune autre séance n'est en cours
    const existing = db.prepare(
      "SELECT id FROM session WHERE status = 'in_progress' AND id != ?"
    ).get(req.params.id)
    if (existing) {
      return res.status(409).json({
        error:  'SESSION_ALREADY_IN_PROGRESS',
        detail: `Séance ${existing.id} déjà en cours.`,
      })
    }

    // Passer en in_progress
    db.prepare(`
      UPDATE session
      SET status = 'in_progress', started_at = datetime('now'), updated_at = datetime('now')
      WHERE id = ?
    `).run(req.params.id)

    // ── Calcul des ajustements ────────────────────────────────
    const adjustments = computeSessionAdjustments(db, session)

    const updated = db.prepare('SELECT * FROM session WHERE id = ?').get(req.params.id)
    res.json({ ...updated, adjustments })

  } catch (err) {
    console.error('[POST /sessions/:id/start]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})

/**
 * Calcule les cibles actualisées pour chaque exercice de la séance
 * et retourne la liste des ajustements par rapport aux valeurs de référence.
 *
 * Un ajustement est significatif si :
 * - Le poids a changé de plus de 0.1 kg
 * - Les reps cibles ont changé
 * - C'est un premier essai (weight_kg = 0 → suggestion)
 *
 * @returns {Adjustment[]}
 * Adjustment = {
 *   exercise_id,
 *   exercise_name_fr,
 *   template_exercise_id,
 *   is_first_time: bool,
 *   target_weight_kg: float,
 *   target_reps: int,
 *   explanation: string,
 *   change_type: 'increase'|'decrease'|'first_time'|'unchanged'
 * }
 */
function computeSessionAdjustments(db, session) {
  if (!session.workout_template_id) return []

  const sessionExercises = db.prepare(`
    SELECT se.*, e.name_fr, e.mechanic, e.equipment_required, e.is_unilateral
    FROM session_exercise se
    JOIN exercise e ON se.exercise_id = e.id
    WHERE se.session_id = ?
      AND se.template_exercise_id IS NOT NULL
    ORDER BY se.order_index
  `).all(session.id)

  if (sessionExercises.length === 0) return []

  const program        = session.program_id
    ? db.prepare('SELECT * FROM program WHERE id = ?').get(session.program_id)
    : null
  const equipmentItems = db.prepare('SELECT * FROM equipment_item').all()
  const profile        = db.prepare('SELECT * FROM user_profile WHERE id = 1').get()

  const settingsRows   = db.prepare('SELECT key, value FROM settings').all()
  const settings       = {}
  for (const r of settingsRows) settings[r.key] = JSON.parse(r.value)
  const progressionSettings = {
    progression_failure_threshold:    settings.progression_failure_threshold    || 2,
    progression_weight_reduction_pct: settings.progression_weight_reduction_pct || 10,
  }

  const adjustments = []

  for (const se of sessionExercises) {
    const te = db.prepare('SELECT * FROM template_exercise WHERE id = ?')
      .get(se.template_exercise_id)
    if (!te) continue

    const state = db.prepare(
      'SELECT * FROM progression_state WHERE template_exercise_id = ?'
    ).get(te.id)

    const exerciseForCalc = {
      equipment_required: JSON.parse(se.equipment_required || '[]'),
      is_unilateral:      se.is_unilateral || 0,
      mechanic:           se.mechanic,
    }
    const loadsResult = calculateAchievableLoads(equipmentItems, exerciseForCalc)
    const achievable  = loadsResult.loads_kg.length > 0
      ? loadsResult.loads_kg
      : [state?.current_weight_kg || 0]

    const isFirstTime = !state || state.current_weight_kg === 0

    let targetWeight, targetReps, explanation, changeType

    if (isFirstTime) {
      const profileForSuggestion = profile ? {
        weight_kg:     profile.weight_kg,
        fitness_level: profile.fitness_level,
        primary_goal:  profile.primary_goal,
      } : null

      const suggestion = suggestFirstTime(
        { ...exerciseForCalc, mechanic: se.mechanic },
        profileForSuggestion,
        te,
        equipmentItems
      )
      targetWeight = suggestion.weight_kg
      targetReps   = suggestion.reps
      explanation  = suggestion.reasoning
      changeType   = 'first_time'

    } else {
      // Récupérer les dernières séries de travail pour recalculer
      const lastSessionSets = db.prepare(`
        SELECT sl.*
        FROM set_log sl
        JOIN session_exercise se2 ON sl.session_exercise_id = se2.id
        JOIN session s ON se2.session_id = s.id
        WHERE se2.template_exercise_id = ?
          AND s.status = 'completed'
          AND sl.is_warmup = 0
          AND sl.is_completed = 1
        ORDER BY s.completed_at DESC, sl.set_number ASC
        LIMIT ?
      `).all(te.id, te.target_sets * 2)  // prendre les séries de la dernière séance

      // Grouper par séance et ne prendre que la dernière
      const lastSessionId = lastSessionSets.length > 0 ? lastSessionSets[0].session_exercise_id : null
      const relevantSets  = lastSessionSets.filter(s =>
        !lastSessionId || s.session_exercise_id === lastSessionId
      )

      const nextState = relevantSets.length > 0
        ? computeNextState(state, relevantSets, te, program, achievable, progressionSettings)
        : null

      // Charge cible = nextState si dispo, sinon current_weight_kg
      targetWeight = nextState?.current_weight_kg ?? state.current_weight_kg
      targetReps   = nextState?.current_target_reps ?? state.current_target_reps ?? te.target_reps_min
      explanation  = nextState?.explanation ?? `Charge maintenue : ${targetWeight} kg × ${targetReps} reps.`

      // Détecter le type de changement
      const prevWeight = state.current_weight_kg
      const prevReps   = state.current_target_reps ?? te.target_reps_min
      const weightDiff = targetWeight - prevWeight

      if (Math.abs(weightDiff) > 0.1) {
        changeType = weightDiff > 0 ? 'increase' : 'decrease'
      } else if (targetReps !== prevReps) {
        changeType = targetReps > prevReps ? 'increase' : 'decrease'
      } else {
        changeType = 'unchanged'
      }
    }

    adjustments.push({
      exercise_id:          se.exercise_id,
      exercise_name_fr:     se.name_fr,
      template_exercise_id: se.template_exercise_id,
      is_first_time:        isFirstTime,
      target_weight_kg:     targetWeight,
      target_reps:          targetReps,
      explanation,
      change_type:          changeType,
    })
  }

  // Ne retourner que les ajustements pertinents à afficher
  // (premier essai + changements réels — exclure 'unchanged')
  return adjustments.filter(a => a.change_type !== 'unchanged')
}
```

## Flux complet de bout en bout

```
1. Utilisateur complète Séance A (POST /sessions/:id/complete)
   → progression_state mis à jour (+2.5 kg)
   → scheduleWeeks() planifie la Séance B

2. Utilisateur clique "Démarrer" sur la Séance B planifiée
   → startNextSession() appelle POST /sessions/:id/start
   → computeSessionAdjustments() tourne :
      - Exo 1 : 80 kg → 82.5 kg (+2.5) [increase]
      - Exo 2 : premier essai → suggestion 60 kg × 10 [first_time]
      - Exo 3 : 2 échecs → 72 kg (-10%) [decrease]
   → Réponse : { ...session, adjustments: [ 3 entrées ] }

3. SessionActiveView monte
   → Lit store.pendingAdjustments → 3 ajustements
   → Affiche SessionAdjustmentBanner (accordéon fermé par défaut)
   → "🧠 3 ajustements coach pour cette séance ▼"

4. Utilisateur tape le bandeau
   → Accordéon s'ouvre :
      📈 Développé couché barre — 82.5 kg × 5 reps — Objectif atteint, +2.5 kg
      🤖 Curl haltère — 12 kg × 10 reps — Premier essai, niveau intermédiaire hypertrophie
      📉 Squat barre — 72 kg × 8 reps — 2 échecs, -10%

5. Utilisateur tape "OK, j'ai compris"
   → Bandeau disparaît
   → SetRow de chaque exercice pré-rempli avec les nouvelles cibles
   → Bulle ℹ️ disponible sur chaque SetRow pour revoir l'explication
```

1. Modifier `POST /:id/start` dans `sessions.js` :
   - Ajouter les imports en tête.

   - Ajouter la fonction `computeSessionAdjustments()` dans le même fichier.

   - Importer `SessionAdjustmentBanner`.
   - Lire `store.pendingAdjustments` au mount.
   - Afficher le bandeau dans le template.
5. Modifier `ProgramDetailView.js` (ou tout endroit qui appelle `sessionsApi.start()`) :
   - Stocker les adjustments dans le store avant de naviguer vers `#/session`.
6. Ajouter le CSS dans `components.css`.
7. Ajouter les clés i18n.

**Points d'attention :**
- `computeSessionAdjustments` est appelée dans `POST /sessions/:id/start` mais
  ne met PAS à jour `progression_state`. Elle lit uniquement pour calculer.
  La vraie mise à jour de `progression_state` reste dans `POST /sessions/:id/complete`.
- `computeNextState` dans `computeSessionAdjustments` est appelé avec les séries
  de la DERNIÈRE séance complétée — pas les séries en cours (la séance n'a pas encore
  commencé). Si `progression_state` est déjà à jour (mis à jour lors de la complétion),
  le calcul est redondant. Dans ce cas, retourner directement les valeurs de `progression_state`
  sans recalculer.
- **Simplification acceptable** : si `progression_state.last_session_id` correspond à la
  dernière séance complétée, utiliser directement `current_weight_kg` et
  `current_target_reps` de `progression_state` sans recalculer. C'est suffisant car
  `POST /sessions/:id/complete` a déjà fait le calcul.
- Le bandeau est **accordéon fermé par défaut** — l'utilisateur peut l'ignorer.
  `@dismiss` le masque définitivement pour cette session.
- `store.pendingAdjustments` est vidé dès que `SessionActiveView` le lit —
  il ne persiste pas entre les sessions.
- Si l'utilisateur navigue vers `#/session` sans passer par `startNextSession()`
  (ex: séance déjà en cours au redémarrage), `pendingAdjustments = []` → pas de bandeau.

```javascript
import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import request from 'supertest'
import { app } from '../../server.js'

describe('T7.3 — Ajustements au lancement de séance', () => {
  it('POST /sessions/:id/start retourne adjustments[]', async () => {
    // Créer programme → template → séance planifiée
    const pRes = await request(app).post('/api/v1/programs').send({
      name: 'Test', goal: 'hypertrophy', level: 'intermediate', days_per_week: 3
    })
    const tRes = await request(app)
      .post(`/api/v1/programs/${pRes.body.id}/templates`)
      .send({ name: 'Push' })
    await request(app).post(`/api/v1/templates/${tRes.body.id}/exercises`).send({
      exercise_id: 'bench_press_barbell', target_sets: 3, target_reps_min: 8, target_reps_max: 12
    })
    const sRes = await request(app).post('/api/v1/sessions').send({
      name: 'Push A', scheduled_date: new Date().toISOString().slice(0, 10),
      workout_template_id: tRes.body.id, program_id: pRes.body.id
    })
    const startRes = await request(app).post(`/api/v1/sessions/${sRes.body.id}/start`)

    assert.equal(startRes.status, 200)
    assert.ok('adjustments' in startRes.body, 'adjustments doit être dans la réponse')
    assert.ok(Array.isArray(startRes.body.adjustments))
  })

  it('premier essai → adjustment avec change_type=first_time', async () => {
    // Même setup — progression_state.current_weight_kg = 0 par défaut
    // Le premier essai doit produire un ajustement first_time
    const pRes  = await request(app).post('/api/v1/programs').send({
      name: 'Test2', goal: 'strength', level: 'beginner', days_per_week: 3
    })
    const tRes  = await request(app)
      .post(`/api/v1/programs/${pRes.body.id}/templates`)
      .send({ name: 'A' })
    await request(app).post(`/api/v1/templates/${tRes.body.id}/exercises`).send({
      exercise_id: 'squat_barbell', target_sets: 3, target_reps_min: 5, target_reps_max: 5
    })
    const sRes = await request(app).post('/api/v1/sessions').send({
      name: 'A', scheduled_date: new Date().toISOString().slice(0, 10),
      workout_template_id: tRes.body.id, program_id: pRes.body.id
    })
    const startRes = await request(app).post(`/api/v1/sessions/${sRes.body.id}/start`)

    assert.equal(startRes.status, 200)
    const firstAdj = startRes.body.adjustments.find(a => a.change_type === 'first_time')
    assert.ok(firstAdj, 'Doit avoir un ajustement first_time')
    assert.equal(firstAdj.exercise_id, 'squat_barbell')
    assert.ok(typeof firstAdj.explanation === 'string')
  })

  it('séance sans template → adjustments = []', async () => {
    const sRes = await request(app).post('/api/v1/sessions').send({
      name: 'Libre', scheduled_date: new Date().toISOString().slice(0, 10)
    })
    const startRes = await request(app).post(`/api/v1/sessions/${sRes.body.id}/start`)
    assert.equal(startRes.status, 200)
    assert.deepEqual(startRes.body.adjustments, [])
  })
})
```
