## Nouveau endpoint `POST /reset-stats`

### Dans `src/routes/exportImport.js`

```javascript
// ── POST /reset-stats ─────────────────────────────────────────
// Efface uniquement les données d'activité (séances, séries, poids, records,
// état de progression) sans toucher aux programmes, exercices, équipement,
// paramètres.
router.post('/reset-stats', (req, res) => {
  try {
    // Sauvegarder avant d'effacer
    autoBackup()

    db.transaction(() => {
      // Ordre FK-safe
      db.prepare('DELETE FROM set_log').run()
      db.prepare('DELETE FROM session_exercise').run()
      db.prepare('DELETE FROM session').run()
      db.prepare('DELETE FROM body_weight').run()
      db.prepare('DELETE FROM personal_record').run()
      db.prepare('DELETE FROM equipment_advice').run()

      // Remettre les états de progression à zéro
      // (garder les lignes — elles sont liées aux template_exercises)
      db.prepare(`
        UPDATE progression_state
        SET current_weight_kg    = 0,
            current_target_reps  = NULL,
            consecutive_failures = 0,
            last_session_id      = NULL,
            last_updated         = datetime('now')
      `).run()

      // Remettre la semaine courante des programmes à 1
      db.prepare(`
        UPDATE program
        SET current_week     = 1,
            is_in_deload     = 0,
            last_deload_date = NULL,
            updated_at       = datetime('now')
      `).run()
    })()

    res.json({
      message: 'Statistiques et historique réinitialisés.',
      cleared: ['sessions', 'set_logs', 'body_weights',
                'personal_records', 'equipment_advice',
                'progression_states (reset to 0)'],
    })
  } catch (err) {
    console.error('[POST /reset-stats]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})
```

```javascript
router.delete('/:id', (req, res) => {
  try {
    const session = db.prepare('SELECT * FROM session WHERE id = ?').get(req.params.id)
    if (!session) return res.status(404).json({ error: 'SESSION_NOT_FOUND' })

    // Seule restriction : impossible de supprimer une séance en cours
    if (session.status === 'in_progress') {
      return res.status(409).json({
        error:  'CANNOT_DELETE_IN_PROGRESS_SESSION',
        detail: 'Terminez ou abandonnez la séance avant de la supprimer.',
      })
    }

    // planned, completed, skipped → supprimable
    db.prepare('DELETE FROM session WHERE id = ?').run(req.params.id)
    res.status(204).end()
  } catch (err) {
    console.error('[DELETE /sessions/:id]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})
```

## `public/js/components/session/SessionHistoryView.js` — version enrichie

Remplacer entièrement le contenu du fichier :

```javascript
import { reactive, onMounted, inject, computed } from '/vendor/vue.esm-browser.prod.js'
import { sessions as api } from '../../api.js'
import { formatDuration, formatDate } from '../../utils/date.js'
import { t } from '../../i18n.js'
import PageHeader     from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner    from '../common/ErrorBanner.js'
import EmptyState     from '../common/EmptyState.js'
import ConfirmModal   from '../common/ConfirmModal.js'

export default {
  name: 'SessionHistoryView',
  components: {
    PageHeader, LoadingSpinner, ErrorBanner, EmptyState, ConfirmModal
  },

  setup() {
    const router = inject('router')
    const store  = inject('store')

    const state = reactive({
      loading:      false,
      error:        null,
      sessions:     [],
      // Filtres
      filterStatus: 'all',      // 'all' | 'completed' | 'skipped'
      filterQ:      '',         // recherche texte sur le nom
      // Suppression
      deleteTarget: null,       // session à supprimer
      deleting:     false,
    })

    // ── Chargement ────────────────────────────────────────────
    async function load() {
      state.loading = true; state.error = null
      try {
        const params = { limit: 200 }
        if (state.filterStatus !== 'all') params.status = state.filterStatus
        const data = await api.list(params)
        // Trier du plus récent au plus ancien
        state.sessions = data.sessions.sort((a, b) =>
          b.scheduled_date.localeCompare(a.scheduled_date)
        )
      } catch (err) {
        state.error = err.message
      } finally {
        state.loading = false
      }
    }

    // ── Filtrage local ─────────────────────────────────────────
    const filteredSessions = computed(() => {
      if (!state.filterQ.trim()) return state.sessions
      const q = state.filterQ.trim().toLowerCase()
      return state.sessions.filter(s => s.name.toLowerCase().includes(q))
    })

    // ── Stats globales ─────────────────────────────────────────
    const globalStats = computed(() => {
      const completed = state.sessions.filter(s => s.status === 'completed')
      const totalVol  = completed.reduce((sum, s) => sum + (s.total_volume_kg || 0), 0)
      const totalDur  = completed.reduce((sum, s) => sum + (s.duration_seconds || 0), 0)
      const avgRpe    = completed.filter(s => s.rpe_avg).length > 0
        ? completed.filter(s => s.rpe_avg).reduce((sum, s) => sum + s.rpe_avg, 0) /
          completed.filter(s => s.rpe_avg).length
        : null
      return {
        count:   completed.length,
        vol:     Math.round(totalVol),
        dur:     totalDur,
        avgRpe:  avgRpe ? Math.round(avgRpe * 10) / 10 : null,
      }
    })

    // ── Suppression ───────────────────────────────────────────
    function askDelete(session) {
      state.deleteTarget = session
    }

    async function confirmDelete() {
      if (!state.deleteTarget) return
      state.deleting = true
      try {
        await api.delete(state.deleteTarget.id)
        state.sessions    = state.sessions.filter(s => s.id !== state.deleteTarget.id)
        state.deleteTarget = null
        store.showToast(t('session.deleted'), 'success')
      } catch (err) {
        store.showToast(err.message, 'error')
      } finally {
        state.deleting = false
      }
    }

    // ── Couleur badge statut ──────────────────────────────────
    function statusColor(status) {
      return {
        completed:   'var(--color-success)',
        planned:     'var(--color-warning)',
        skipped:     'var(--color-text-muted)',
        in_progress: 'var(--color-primary)',
      }[status] || 'var(--color-border)'
    }

    onMounted(load)

    return {
      state, filteredSessions, globalStats,
      load, askDelete, confirmDelete,
      statusColor, formatDuration, formatDate, router, t,
    }
  },

  template: `
    <div class="view">
      <PageHeader :title="t('session.history_title')" :show-back="true" />

      <div class="view__content">
        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner   v-else-if="state.error" :message="state.error" :retry="load" />

        <template v-else>
          <!-- Stats globales -->
          <div v-if="globalStats.count > 0" class="history-stats">
            <div class="history-stats__item">
              <span class="history-stats__value">{{ globalStats.count }}</span>
              <span class="history-stats__label">séances</span>
            </div>
            <div class="history-stats__item">
              <span class="history-stats__value">{{ globalStats.vol.toLocaleString() }}</span>
              <span class="history-stats__label">kg soulevés</span>
            </div>
            <div class="history-stats__item">
              <span class="history-stats__value">{{ formatDuration(globalStats.dur) }}</span>
              <span class="history-stats__label">temps total</span>
            </div>
            <div v-if="globalStats.avgRpe" class="history-stats__item">
              <span class="history-stats__value">{{ globalStats.avgRpe }}</span>
              <span class="history-stats__label">RPE moy.</span>
            </div>
          </div>

          <!-- Filtres -->
          <div class="history-filters">
            <input
              class="form-input history-filters__search"
              type="search"
              :placeholder="t('session.history_search')"
              v-model="state.filterQ"
            />
            <div class="history-filters__tabs">
              <button
                v-for="s in ['all','completed','skipped']" :key="s"
                class="history-filter-tab"
                :class="{ 'history-filter-tab--active': state.filterStatus === s }"
                @click="state.filterStatus = s; load()">
                {{ t('session.filter_' + s) }}
              </button>
            </div>
          </div>

          <!-- Liste vide -->
          <EmptyState
            v-if="!filteredSessions.length"
            icon="📅"
            :message="state.filterQ ? t('session.history_no_results') : t('session.history_empty')"
          />

          <!-- Liste des séances -->
          <div v-else class="history-list">
            <div
              v-for="s in filteredSessions"
              :key="s.id"
              class="history-row">

              <!-- Zone principale → tap → détail -->
              <div class="history-row__main" @click="router.push('#/session/' + s.id)">
                <div class="history-row__top">
                  <span class="history-row__name">{{ s.name }}</span>
                  <span class="history-row__status-dot"
                    :style="{ background: statusColor(s.status) }" />
                </div>
                <div class="history-row__meta">
                  <span>{{ formatDate(s.scheduled_date) }}</span>
                  <span v-if="s.duration_seconds">
                    · {{ formatDuration(s.duration_seconds) }}
                  </span>
                  <span v-if="s.total_volume_kg">
                    · {{ s.total_volume_kg }} kg
                  </span>
                  <span v-if="s.rpe_avg">
                    · RPE {{ s.rpe_avg.toFixed(1) }}
                  </span>
                </div>
              </div>

              <!-- Bouton supprimer -->
              <button
                v-if="s.status !== 'in_progress'"
                class="history-row__delete"
                @click.stop="askDelete(s)"
                :title="t('session.delete')">
                🗑
              </button>
            </div>
          </div>
        </template>
      </div>

      <!-- Confirmation suppression -->
      <ConfirmModal
        v-if="state.deleteTarget"
        :title="t('session.delete_confirm_title')"
        :body="t('session.delete_confirm_body', { name: state.deleteTarget.name, date: formatDate(state.deleteTarget.scheduled_date) })"
        :danger="true"
        :confirm-text="t('common.delete')"
        @confirm="confirmDelete()"
        @cancel="state.deleteTarget = null"
      />
    </div>
  `
}
```

## Étapes d'implémentation

1. **Backend** — Ajouter `POST /reset-stats` dans `exportImport.js`.
2. **Backend** — Vérifier `DELETE /sessions/:id` dans `sessions.js`
   (seule restriction : `in_progress`). Si déjà correct → rien à faire.
3. **`api.js`** — Ajouter `resetStats: () => request('POST', '/reset-stats')`.
4. **`SessionHistoryView.js`** — Remplacer entièrement par la version enrichie.
5. **`SessionDetailView.js`** — Ajouter bouton delete + ConfirmModal.
6. **`ExportImport.js`** — Ajouter section reset stats + modale double confirmation.
7. **CSS** — Ajouter les styles dans `components.css`.
8. **i18n** — Ajouter toutes les clés dans `fr.json` et `en.json`.

**Points d'attention :**
- `POST /reset-stats` appelle `autoBackup()` **avant** de supprimer quoi que
  ce soit. Si le backup échoue (disque plein, etc.), le reset est quand même
  exécuté — c'est acceptable.
- `DELETE FROM set_log` entraîne la cascade sur rien (pas de table fille).
  `DELETE FROM session_exercise` via cascade de `session` suffit, mais
  l'ordre explicite garantit la compatibilité même si les FK sont désactivées.
- La double confirmation ("taper RESET") est côté **client uniquement** —
  ce n'est pas une validation serveur. L'endpoint `POST /reset-stats` lui-même
  n'exige pas de token. C'est acceptable pour une app locale mono-utilisateur.
- `formatDate` et `formatDuration` sont importés depuis `utils/date.js` — déjà
  disponibles depuis T1.6.
- Le filtre `filterStatus` recharge depuis l'API au clic (pas de filtre local)
  pour s'assurer que la liste est toujours fraîche.
- Après `confirmDelete()`, la session est retirée de `state.sessions` localement
  sans recharger depuis l'API — plus fluide sur mobile.
- `formatDate(state.session?.scheduled_date)` dans `SessionDetailView.js` :
  vérifier que `formatDate` gère `undefined` et retourne `'—'` sans planter.
- La modale de reset stats utilise `v-model` sur un `<input>` pour la saisie
  de "RESET". Le bouton de confirmation est `disabled` tant que la valeur n'est
  pas exactement `'RESET'` (case-sensitive).

```javascript
import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import request from 'supertest'
import { app } from '../../server.js'

describe('T7.4 — Historique, suppression, reset stats', () => {

  // ── GET /sessions ──────────────────────────────────────────
  it('GET /sessions retourne la liste triée par date DESC', async () => {
    // Créer 2 séances à des dates différentes
    await request(app).post('/api/v1/sessions')
      .send({ name: 'A', scheduled_date: '2025-07-01' })
    await request(app).post('/api/v1/sessions')
      .send({ name: 'B', scheduled_date: '2025-07-10' })

    const res = await request(app).get('/api/v1/sessions?limit=200')
    assert.equal(res.status, 200)
    assert.ok(Array.isArray(res.body.sessions))
    // La plus récente doit être en premier
    if (res.body.sessions.length >= 2) {
      const dates = res.body.sessions.map(s => s.scheduled_date)
      for (let i = 1; i < dates.length; i++) {
        assert.ok(dates[i] <= dates[i - 1],
          `Ordre DESC attendu : ${dates[i-1]} ≥ ${dates[i]}`)
      }
    }
  })

  // ── DELETE /sessions/:id ───────────────────────────────────
  it('DELETE /sessions/:id (planned) → 204', async () => {
    const s = await request(app).post('/api/v1/sessions')
      .send({ name: 'À supprimer', scheduled_date: '2025-08-01' })
    const res = await request(app).delete(`/api/v1/sessions/${s.body.id}`)
    assert.equal(res.status, 204)
  })

  it('DELETE /sessions/:id (completed) → 204', async () => {
    const s = await request(app).post('/api/v1/sessions')
      .send({ name: 'Complétée', scheduled_date: '2025-07-05' })
    await request(app).post(`/api/v1/sessions/${s.body.id}/start`)
    await request(app).post(`/api/v1/sessions/${s.body.id}/complete`)
    const res = await request(app).delete(`/api/v1/sessions/${s.body.id}`)
    assert.equal(res.status, 204)
  })

  it('DELETE /sessions/:id (in_progress) → 409', async () => {
    const s = await request(app).post('/api/v1/sessions')
      .send({ name: 'En cours', scheduled_date: new Date().toISOString().slice(0,10) })
    await request(app).post(`/api/v1/sessions/${s.body.id}/start`)
    const res = await request(app).delete(`/api/v1/sessions/${s.body.id}`)
    assert.equal(res.status, 409)
    assert.equal(res.body.error, 'CANNOT_DELETE_IN_PROGRESS_SESSION')
    // Nettoyer
    await request(app).post(`/api/v1/sessions/${s.body.id}/complete`)
  })

  it('DELETE /sessions/9999 → 404', async () => {
    const res = await request(app).delete('/api/v1/sessions/9999')
    assert.equal(res.status, 404)
  })

  // ── POST /reset-stats ──────────────────────────────────────
  it('POST /reset-stats efface séances et body_weight', async () => {
    // Créer des données
    await request(app).post('/api/v1/sessions')
      .send({ name: 'Test reset', scheduled_date: '2025-06-01' })
    await request(app).post('/api/v1/bodyweight')
      .send({ date: '2025-06-01', weight_kg: 75 })

    const res = await request(app).post('/api/v1/reset-stats')
    assert.equal(res.status, 200)
    assert.ok(res.body.cleared.includes('sessions'))

    // Vérifier que les séances sont effacées
    const sessionsRes = await request(app).get('/api/v1/sessions')
    assert.equal(sessionsRes.body.sessions.length, 0)

    // Vérifier que le body_weight est effacé
    const bwRes = await request(app).get('/api/v1/bodyweight')
    assert.equal(bwRes.body.entries.length, 0)
  })

  it('POST /reset-stats conserve les programmes et exercices', async () => {
    // Créer un programme
    await request(app).post('/api/v1/programs').send({
      name: 'PPL', goal: 'strength', level: 'beginner', days_per_week: 3
    })
    // Reset
    await request(app).post('/api/v1/reset-stats')

    // Le programme doit toujours exister
    const progRes = await request(app).get('/api/v1/programs')
    assert.ok(progRes.body.programs.length >= 1)

    // Les exercices aussi
    const exRes = await request(app).get('/api/v1/exercises')
    assert.ok(exRes.body.exercises.length > 0)
  })

  it('POST /reset-stats remet progression_state à 0', async () => {
    // Importer le db singleton pour vérifier directement
    const { getDb } = await import('../../src/db/database.js')
    const db = getDb()

    // Simuler une progression
    db.prepare('UPDATE progression_state SET current_weight_kg = 100').run()

    await request(app).post('/api/v1/reset-stats')

    const states = db.prepare(
      'SELECT current_weight_kg FROM progression_state WHERE current_weight_kg > 0'
    ).all()
    assert.equal(states.length, 0, 'Tous les progression_state doivent être à 0')
  })
})
```

## Critères d'acceptation

### Historique
- [ ] `#/session/history` affiche la liste des séances, du plus récent au plus ancien
- [ ] Stats globales visibles en haut : nombre de séances, volume total, temps total
- [ ] Barre de recherche filtre les séances par nom en temps réel
- [ ] Onglets "Toutes / Complétées / Sautées" fonctionnels
- [ ] Tap sur une séance → `#/session/:id` avec détail complet

### Suppression
- [ ] Chaque ligne de l'historique a un bouton 🗑
- [ ] Tap sur 🗑 → `ConfirmModal` avec nom et date de la séance
- [ ] Confirmer → séance supprimée, disparaît de la liste, Toast "Séance supprimée."
- [ ] Annuler → rien ne se passe
- [ ] `SessionDetailView` a aussi un bouton "Supprimer" en bas
- [ ] Séance `in_progress` → bouton absent (ou grisé)
- [ ] Séance inexistante → 404

### Reset stats
- [ ] Paramètres > Données → bouton "Réinitialiser stats et historique" visible
  (distinct du reset total)
- [ ] Cliquer → modale avec description précise de ce qui est effacé/conservé

- [ ] Confirmer avec "RESET" → POST /reset-stats → Toast + rechargement
- [ ] Après reset : `GET /sessions` → liste vide
- [ ] Après reset : `GET /programs` → programmes toujours là
- [ ] Après reset : `GET /exercises` → exercices toujours là
- [ ] Backup créé dans `data/backups/` avant le reset
