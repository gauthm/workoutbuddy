### 3. Ajouter `PUT /exercises/:id/toggle-exclusion`

```javascript
// IMPORTANT : déclarer cette route AVANT /:id dans le fichier
// pour éviter qu'Express interprète "toggle-exclusion" comme un :id

router.put('/:id/toggle-exclusion', (req, res) => {
  try {
    const exercise = db.prepare('SELECT * FROM exercise WHERE id = ?').get(req.params.id)
    if (!exercise) return res.status(404).json({ error: 'EXERCISE_NOT_FOUND' })

    const { is_excluded, exclusion_reason } = req.body
    if (is_excluded === undefined || is_excluded === null) {
      return res.status(400).json({ error: 'MISSING_IS_EXCLUDED' })
    }

    const newValue = (is_excluded === true || is_excluded === 1) ? 1 : 0

    db.prepare(`
      UPDATE exercise
      SET is_excluded      = ?,
          exclusion_reason = ?
      WHERE id = ?
    `).run(
      newValue,
      newValue === 1 ? (exclusion_reason || null) : null,
      req.params.id
    )

    const updated = db.prepare('SELECT * FROM exercise WHERE id = ?').get(req.params.id)
    res.json(parseExercise(updated))
  } catch (err) {
    console.error('[PUT /exercises/:id/toggle-exclusion]', err.message)
    res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
})
```

## `ExerciseListView.js` — version enrichie avec exclusion

```javascript
import { reactive, onMounted, inject, computed } from '/vendor/vue.esm-browser.prod.js'
import { exercises as api } from '../../api.js'
import { t } from '../../i18n.js'
import PageHeader     from '../layout/PageHeader.js'
import LoadingSpinner from '../common/LoadingSpinner.js'
import ErrorBanner    from '../common/ErrorBanner.js'
import EmptyState     from '../common/EmptyState.js'
import ExerciseFilter from './ExerciseFilter.js'

export default {
  name: 'ExerciseListView',
  components: { PageHeader, LoadingSpinner, ErrorBanner, EmptyState, ExerciseFilter },

  setup() {
    const router = inject('router')
    const store  = inject('store')

    const state = reactive({
      loading:      false,
      error:        null,
      exercises:    [],
      filters:      { muscle: '', equipment: '', movement: '', q: '' },
      showExcluded: true,
    })

    // ── Chargement ────────────────────────────────────────────
    async function load() {
      state.loading = true; state.error = null
      try {
        // Charger TOUS les exercices (exclus inclus) pour le toggle
        const params = {}
        if (state.filters.muscle)    params.muscle    = state.filters.muscle
        if (state.filters.equipment) params.equipment = state.filters.equipment
        if (state.filters.movement)  params.movement  = state.filters.movement
        if (state.filters.q)         params.q         = state.filters.q
        const data = await api.list(params)
        state.exercises = data.exercises
      } catch (err) { state.error = err.message }
      finally { state.loading = false }
    }

    function onFilterChange(f) { Object.assign(state.filters, f); load() }

    // ── Tri : exclus en bas, puis alphabétique ─────────────────
    const displayedExercises = computed(() => {
      let list = state.exercises
      if (!state.showExcluded) list = list.filter(e => !e.is_excluded)
      return [...list].sort((a, b) => {
        if (a.is_excluded !== b.is_excluded) return a.is_excluded ? 1 : -1
        return (a.name_fr || a.name_en || '').localeCompare(b.name_fr || b.name_en || '', 'fr')
      })
    })

    const excludedCount = computed(() => state.exercises.filter(e => e.is_excluded).length)

    // ── Toggle exclusion ──────────────────────────────────────
    async function toggleExclusion(ex, event) {
      event.stopPropagation()
      try {
        const updated = await api.toggleExclusion(ex.id, { is_excluded: !ex.is_excluded })
        const idx = state.exercises.findIndex(e => e.id === ex.id)
        if (idx !== -1) {
          state.exercises[idx] = {
            ...state.exercises[idx],
            is_excluded:      updated.is_excluded,
            exclusion_reason: updated.exclusion_reason,
          }
        }
        store.showToast(
          updated.is_excluded
            ? t('exercises.excluded_toast')
            : t('exercises.included_toast'),
          updated.is_excluded ? 'info' : 'success'
        )
      } catch (err) {
        store.showToast(err.message, 'error')
      }
    }

    onMounted(load)

    return {
      state, displayedExercises, excludedCount,
      load, onFilterChange, toggleExclusion, router, t,
    }
  },

  template: `
    <div class="view">
      <PageHeader :title="t('exercises.title')">
        <template #actions>
          <button class="btn btn--primary"
            style="padding:0.4rem 0.75rem;font-size:0.85rem"
            @click="router.push('#/exercises/new')">+</button>
        </template>
      </PageHeader>

      <div class="view__content">
        <ExerciseFilter :filters="state.filters" @change="onFilterChange" />

        <!-- Bandeau exercices exclus -->
        <div v-if="excludedCount > 0" class="exercise-excluded-bar">
          <span class="exercise-excluded-bar__count">
            🚫 {{ excludedCount }} exercice(s) désactivé(s)
          </span>
          <button class="exercise-excluded-bar__toggle"
            @click="state.showExcluded = !state.showExcluded">
            {{ state.showExcluded ? t('exercises.hide_excluded') : t('exercises.show_excluded') }}
          </button>
        </div>

        <LoadingSpinner v-if="state.loading" />
        <ErrorBanner   v-else-if="state.error" :message="state.error" :retry="load" />
        <EmptyState    v-else-if="!displayedExercises.length"
          icon="🔍" :message="t('exercises.no_results')" />

        <div v-else class="exercise-list">
          <div
            v-for="ex in displayedExercises"
            :key="ex.id"
            class="exercise-card"
            :class="{ 'exercise-card--excluded': ex.is_excluded }"
            @click="router.push('#/exercises/' + ex.id)">

            <div class="exercise-card__main">
              <span class="exercise-card__name">{{ ex.name_fr || ex.name_en }}</span>
              <div style="display:flex;gap:0.3rem;align-items:center;flex-shrink:0">
                <span v-if="ex.is_custom"   class="badge badge--custom">Custom</span>
                <span v-if="ex.is_excluded" class="badge badge--excluded">
                  {{ t('exercises.excluded_badge') }}
                </span>
              </div>
            </div>

            <div class="exercise-card__meta">
              <span class="exercise-card__muscles">
                {{ (ex.primary_muscles || []).map(m => t('muscles.' + m) || m).join(', ') }}
              </span>

              <!-- Bouton toggle exclusion -->
              <button
                class="exercise-card__exclude-btn"
                :title="ex.is_excluded ? t('exercises.include_btn') : t('exercises.exclude_btn')"
                @click.stop="toggleExclusion(ex, $event)">
                {{ ex.is_excluded ? '✅' : '🚫' }}
              </button>
            </div>

            <!-- Raison d'exclusion si présente -->
            <div v-if="ex.is_excluded && ex.exclusion_reason"
              class="exercise-card__exclusion-reason">
              {{ t('exercises.exclusion_reason', { reason: ex.exclusion_reason }) }}
            </div>
          </div>
        </div>
      </div>
    </div>
  `
}
```

import { describe, it } from 'node:test'
import assert from 'node:assert/strict'
import Database from 'better-sqlite3'
import { migrate } from '../../src/db/migrate.js'
import { seedExercises } from '../../src/seed/seed.js'
import { generateProgram } from '../../src/services/programGenerator.js'
import { findRotationCandidates } from '../../src/services/rotationEngine.js'

function createDb() {
  const db = new Database(':memory:')
  db.pragma('foreign_keys = ON')
  migrate(db)
  seedExercises(db)  // charge les ~800 exercices réels
  return db
}

function getRandomExercise(db, filters = '') {
  return db.prepare(
    `SELECT * FROM exercise WHERE is_custom=0 ${filters} LIMIT 1`
  ).get()
}

function parseEx(ex) {
  return {
    ...ex,
    primary_muscles:    JSON.parse(ex.primary_muscles    || '[]'),
    secondary_muscles:  JSON.parse(ex.secondary_muscles  || '[]'),
    equipment_required: JSON.parse(ex.equipment_required || '[]'),
    is_excluded:        ex.is_excluded === 1,
  }
}

describe('Exclusion d\'exercices — unité', () => {

  it('un exercice exclu n\'apparaît pas dans la génération de programme', () => {
    const db = createDb()

    // Récupérer un exercice barbell chest et l'exclure
    const target = db.prepare(`
      SELECT id FROM exercise
      WHERE primary_muscles LIKE '%"chest"%' AND equipment_required LIKE '%"barbell"%'
      AND is_custom = 0 LIMIT 1
    `).get()
    if (!target) return // skip si aucun exercice barbell chest dans la DB

    db.prepare('UPDATE exercise SET is_excluded = 1 WHERE id = ?').run(target.id)

    const allRaw = db.prepare('SELECT * FROM exercise').all()
    const all    = allRaw.map(parseEx)
    const equip  = [
      { category: 'barbell', weight_kg: 20, count: 1 },
      { category: 'bench',   name: 'Banc', count: 1 },
      { category: 'plate',   weight_kg: 5, count: 4 },
    ]

    const draft = generateProgram(
      { goal: 'hypertrophy', level: 'intermediate', days_per_week: 3,
        priority_muscles: [], name: 'Test' },
      all, equip
    )

    const allIds = draft.templates.flatMap(t => t.exercises.map(e => e.exercise_id))
    assert.ok(
      !allIds.includes(target.id),
      `L'exercice exclu "${target.id}" ne doit pas être dans le programme`
    )
  })

  it('un exercice exclu n\'est pas candidat pour la rotation', () => {
    const db = createDb()

    // Prendre un exercice push_horizontal comme base
    const base = db.prepare(
      "SELECT * FROM exercise WHERE movement_pattern='push_horizontal' AND is_custom=0 LIMIT 1"
    ).get()
    if (!base) return

    const parsedBase = parseEx(base)

    // Exclure tous les autres exercices push_horizontal
    db.prepare(`
      UPDATE exercise SET is_excluded = 1
      WHERE movement_pattern = 'push_horizontal' AND id != ?
    `).run(base.id)

    const allRaw = db.prepare('SELECT * FROM exercise').all()
    const all    = allRaw.map(parseEx)
    const equip  = [
      { category: 'barbell', weight_kg: 20, count: 1 },
      { category: 'dumbbell_fixed', weight_kg: 15, count: 2 },
      { category: 'bench', name: 'Banc', count: 1 },
    ]

    const candidates = findRotationCandidates(parsedBase, all, equip)
    const nonBase    = candidates.filter(c => c.id !== base.id)

    assert.ok(
      nonBase.every(c => !c.is_excluded),
      'Aucun exercice exclu ne doit apparaître comme candidat de rotation'
    )
  })

  it('réactiver un exercice le remet dans la sélection', () => {
    const db  = createDb()
    const ex  = getRandomExercise(db)
    assert.ok(ex, 'Doit avoir au moins un exercice en DB')

    db.prepare('UPDATE exercise SET is_excluded = 1 WHERE id = ?').run(ex.id)
    let row = db.prepare('SELECT is_excluded FROM exercise WHERE id = ?').get(ex.id)
    assert.equal(row.is_excluded, 1)

    db.prepare('UPDATE exercise SET is_excluded = 0, exclusion_reason = NULL WHERE id = ?').run(ex.id)
    row = db.prepare('SELECT is_excluded FROM exercise WHERE id = ?').get(ex.id)
    assert.equal(row.is_excluded, 0)
  })

  it('tous les exercices ont is_excluded = 0 par défaut', () => {
    const db = createDb()
    const excluded = db.prepare(
      'SELECT COUNT(*) as n FROM exercise WHERE is_excluded = 1'
    ).get()
    assert.equal(excluded.n, 0, 'Aucun exercice ne doit être exclu par défaut')
  })
})

  it('PUT /exercises/:id/toggle-exclusion → exclure un exercice réel', async () => {

    assert.equal(res.status, 200)
    assert.equal(res.body.is_excluded, true)
    assert.equal(res.body.exclusion_reason, 'Test douleur épaule')
    assert.equal(res.body.id, realExerciseId1)
  })

  it('GET /exercises?excluded=1 retourne seulement les exclus', async () => {
    if (!realExerciseId1) return
    const res = await request(app).get('/api/v1/exercises?excluded=1')
    assert.equal(res.status, 200)
    assert.ok(res.body.exercises.length >= 1)
    assert.ok(
      res.body.exercises.every(e => e.is_excluded === true),
      'Tous les exercices retournés doivent être exclus'
    )
  })

  it('GET /exercises?excluded=0 ne retourne pas les exclus', async () => {
    if (!realExerciseId1) return
    const res = await request(app).get('/api/v1/exercises?excluded=0')
    assert.equal(res.status, 200)
    assert.ok(
      res.body.exercises.every(e => e.is_excluded === false),
      'Aucun exercice exclu dans les résultats'
    )
    // L'exercice exclu ne doit pas apparaître
    const found = res.body.exercises.find(e => e.id === realExerciseId1)
    assert.ok(!found, 'L\'exercice exclu ne doit pas être dans les résultats ?excluded=0')

      .put(`/api/v1/exercises/${realExerciseId1}/toggle-exclusion`)

    assert.equal(res.status, 200)
    assert.equal(res.body.is_excluded, false)

  it('PUT toggle-exclusion sur ID inexistant → 404', async () => {
