## Backend — nouveau `POST /ai/chat` dans `src/routes/ai.js`

Remplacer le début du handler (de `router.post('/chat',` jusqu'à la
construction de `messages`) par la logique ci-dessous. **La boucle de
stream existante (détection blocs action, `sendEvent`, etc.) est
conservée telle quelle**, sauf le point `done` (voir plus bas) et le bloc
`finally` (persistance).

```javascript
router.post('/chat', async (req, res) => {
  const { message, conversation_id } = req.body
  // Le champ "history" du body est volontairement ignoré (contexte depuis la DB)
  if (!message?.trim()) return res.status(400).json({ error: 'MISSING_MESSAGE' })

  const aiSettings = getAiSettings(db)
  if (!aiSettings.base_url || !aiSettings.model) {
    return res.status(503).json({ error: 'AI_NOT_CONFIGURED' })  // AUCUNE écriture DB ici
  }

  // ── 1. Résolution de la conversation ──────────────────────
  let conversation
  try {
    conversation = db.transaction(() => {
      if (conversation_id !== undefined && conversation_id !== null) {
        const c = db.prepare('SELECT * FROM coach_conversation WHERE id = ?').get(conversation_id)
        if (!c) return null
        if (!c.is_active) {
          db.prepare('UPDATE coach_conversation SET is_active = 0').run()
          db.prepare("UPDATE coach_conversation SET is_active = 1, updated_at = datetime('now') WHERE id = ?").run(c.id)
          return db.prepare('SELECT * FROM coach_conversation WHERE id = ?').get(c.id)
        }
        return c
      }
      let c = db.prepare('SELECT * FROM coach_conversation WHERE is_active = 1 LIMIT 1').get()
      if (!c) {
        const n = db.prepare('SELECT COUNT(*) n FROM coach_conversation').get().n + 1
        db.prepare('INSERT INTO coach_conversation (title, is_active) VALUES (?,1)').run(`Conversation ${n}`)
        c = db.prepare('SELECT * FROM coach_conversation WHERE id = (SELECT last_insert_rowid())').get()
      }
      return c
    })()
  } catch (err) {
    console.error('[POST /ai/chat] résolution conversation', err.message)
    return res.status(500).json({ error: 'INTERNAL_ERROR' })
  }
  if (!conversation) return res.status(404).json({ error: 'CONVERSATION_NOT_FOUND' })

  // ── 2. Insertion du message utilisateur ───────────────────
  const userMsg = db.prepare('INSERT INTO coach_message (conversation_id, role, content) VALUES (?,\'user\',?)')
    .run(conversation.id, message.trim())

  // Titre dérivé de la première question (si jamais renommée)
  const msgCount = db.prepare('SELECT COUNT(*) n FROM coach_message WHERE conversation_id = ?').get(conversation.id).n
  if (msgCount === 1 && conversation.title === 'Conversation') {
    const derived = message.trim().slice(0, 60)
    db.prepare("UPDATE coach_conversation SET title = ? WHERE id = ?")
      .run(message.trim().length > 60 ? derived + '…' : derived, conversation.id)
  }

  // ── 3. Contexte LLM construit depuis la DB ────────────────
  const history = db.prepare(`
    SELECT role, content FROM coach_message
    WHERE conversation_id = ? AND id < ?
    ORDER BY id DESC LIMIT 14
  `).all(conversation.id, userMsg.lastInsertRowid).reverse()

  res.writeHead(200, {
    'Content-Type':      'text/event-stream',
    'Cache-Control':     'no-cache',
    'Connection':        'keep-alive',
    'X-Accel-Buffering': 'no',
  })

  const systemPrompt = buildSystemPrompt(db)
  const messages = [
    { role: 'system', content: systemPrompt },
    ...history,
    { role: 'user', content: message.trim() },
  ]
```

### Dans le `finally` (persistance assistant)

```javascript
} finally {
  // Persistance de la réponse (brute, blocs ```action inclus)
  try {
    if (fullContent.trim()) {
      db.prepare('INSERT INTO coach_message (conversation_id, role, content) VALUES (?,\'assistant\',?)')
        .run(conversation.id, fullContent)
    }
    db.prepare("UPDATE coach_conversation SET updated_at = datetime('now') WHERE id = ?")
      .run(conversation.id)
  } catch (err) {
    console.error('[POST /ai/chat] persistance assistant', err.message)
    // Non bloquant : le client a déjà reçu le stream
  }
  res.end()
}
```

**Points d'attention backend :**
- Ordre exact : 400 → 503 → résolution conversation (404 possible) →
  insertion user. Ainsi un échec de config n'écrit rien en DB.
- `fullContent` inclut les blocs ```action bruts. Au rechargement de
  l'historique, `ChatMessage` reçoit le même texte brut que pendant le
  stream → rendu identique. L'état "action confirmée" n'est PAS rejoué
  (limitation assumée, documentée en critère).
- Stream interrompu (erreur réseau LLM) : le partiel est persisté s'il
  est non vide. Acceptable pour une app locale.
- `history.slice(-12)` de l'ancienne version est remplacé par les 14
  messages précédents depuis la DB (ordre chronologique via `.reverse()`).
- `applyAction(s)` ne sont pas tracées dans le fil — hors périmètre.
- `done.conversation_id` : le front s'en sert pour mémoriser l'id de la
  conversation auto-créée.

```javascript
    async function refreshConversations() {
      try { state.conversations = (await coachApi.conversations()).conversations } catch (_) {}
    }

    async function newConversation() {
      try {
        const conv = await coachApi.createConversation()
        state.conversationId = conv.id
        state.messages       = []
        state.pendingActions = []
        state.showHistory    = false
        await refreshConversations()
      } catch (err) { store.showToast(err.message, 'error') }
    }

    async function openHistoryPanel() {
      state.showHistory = true
      state.loadingHistory = true
      await refreshConversations()
      state.loadingHistory = false
    }

    async function selectConversation(conv) {
      if (state.isStreaming) return
      try {
        await coachApi.activateConversation(conv.id)
        const detail = await coachApi.messages(conv.id)
        state.conversationId = conv.id
        state.messages       = detail.messages.map(m => ({ role: m.role, content: m.content }))
        state.pendingActions = []
        state.showHistory    = false
        await scrollToBottom()
      } catch (err) { store.showToast(err.message, 'error') }
    }

    async function renameConversation(conv) {
      const title = window.prompt(t('coach.rename_prompt'), conv.title)
      if (!title?.trim()) return
      try {
        await coachApi.renameConversation(conv.id, title.trim())
        await refreshConversations()
      } catch (err) { store.showToast(err.message, 'error') }
    }

    function askDeleteConversation(conv) { state.deleteTarget = conv }

    async function confirmDeleteConversation() {
      if (!state.deleteTarget) return
      try {
        await coachApi.deleteConversation(state.deleteTarget.id)
        if (state.deleteTarget.id === state.conversationId) {
          state.conversationId = null
          state.messages       = []
        }
        state.deleteTarget = null
        store.showToast(t('coach.conversation_deleted'), 'success')
        await refreshConversations()
      } catch (err) { store.showToast(err.message, 'error') }
    }
```
